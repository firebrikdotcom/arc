1use crate::FfiError;
2use std::path::{Path, PathBuf};
3
4/// Normalized database identity, independent of credentials and configuration.
5#[derive(Clone, Debug, Eq, Hash, PartialEq)]
6pub struct CoreKey(Identity);
7
8#[derive(Clone, Debug, Eq, Hash, PartialEq)]
9enum Identity {
10    Sqlite(PathBuf),
11    Postgres(String),
12}
13
14impl CoreKey {
15    /// Resolve filesystem identity synchronously; call off a UI thread.
16    pub fn sqlite_file(path: PathBuf) -> Result<Self, FfiError> {
17        let invalid = || FfiError::InvalidDatabasePath.logged("database_identity");
18        let input = path.to_str().ok_or_else(invalid)?;
19        // SQLite treats only `:memory:` and `file:` names specially (Diesel also rewrites
20        // `sqlite://` to `file:`); other colons, such as Windows drive letters, are literal.
21        let special =
22            input == ":memory:" || input.starts_with("file:") || input.starts_with("sqlite:");
23        if input.is_empty() || special || input.contains('?') || input.contains('#') {
24            return Err(invalid());
25        }
26        let resolved = match std::fs::symlink_metadata(&path) {
27            Ok(_) => {
28                let resolved = path.canonicalize().map_err(|_| invalid())?;
29                if !resolved.is_file() {
30                    return Err(invalid());
31                }
32                resolved
33            }
34            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
35                let name = path.file_name().ok_or_else(invalid)?;
36                let parent = path
37                    .parent()
38                    .filter(|p| !p.as_os_str().is_empty())
39                    .unwrap_or(Path::new("."));
40                let parent = parent.canonicalize().map_err(|_| invalid())?;
41                if !parent.is_dir() {
42                    return Err(invalid());
43                }
44                parent.join(name)
45            }
46            Err(_) => return Err(invalid()),
47        };
48        resolved.to_str().ok_or_else(invalid)?;
49        Ok(Self(Identity::Sqlite(resolved)))
50    }
51
52    /// App-assigned stable, nonsecret identity; DSNs are captured separately.
53    pub fn postgres(namespace: String) -> Result<Self, FfiError> {
54        if namespace.trim().is_empty() {
55            return Err(FfiError::InvalidConfiguration.logged("database_identity"));
56        }
57        Ok(Self(Identity::Postgres(namespace)))
58    }
59
60    pub fn sqlite_path(&self) -> Option<&Path> {
61        match &self.0 {
62            Identity::Sqlite(path) => Some(path),
63            Identity::Postgres(_) => None,
64        }
65    }
66}
