16/// Safe infrastructure categories; database inputs and panic payloads never cross FFI.
17#[derive(Clone, Debug, Eq, PartialEq, thiserror::Error)]
18#[cfg_attr(feature = "uniffi", derive(uniffi::Error))]
19pub enum FfiError {
20    #[error("invalid database path")]
21    InvalidDatabasePath,
22    #[error("invalid configuration identity")]
23    InvalidConfiguration,
24    #[error("core configuration conflicts with an existing registration")]
25    ConfigurationConflict,
26    #[error("native runtime unavailable")]
27    RuntimeUnavailable,
28    #[error("schema initialization failed")]
29    SchemaInitializationFailed,
30    #[error("stack initialization failed")]
31    StackInitializationFailed,
32    #[error("projection rebuild failed")]
33    ProjectionRebuildFailed,
34    #[error("core initialization panicked")]
35    InitializationPanicked,
36    #[error("core operation panicked")]
37    OperationPanicked,
38    #[error("native task cancelled")]
39    TaskCancelled,
40}
41
42impl FfiError {
43    pub(crate) fn logged(self, stage: &'static str) -> Self {
44        tracing::error!(stage, category = %self, "native adapter failure");
45        self
46    }
47}

49/// Delegate assembly to arc-stack. The caller prepares schema and chooses replay.
50pub async fn build_stack<A: Aggregate>(
51    driver: DatabaseDriver,
52    database_url: &str,
53    projectors: Vec<ProjectorReg>,
54    snapshot_policy: Option<SnapshotPolicy>,
55) -> Result<SharedStack<A>, FfiError> {
56    arc_stack::es_stack::build_with_driver(driver, database_url, projectors, snapshot_policy)
57        .await
58        .map(Into::into)
59        .map_err(|error| {
60            // Log only a fixed category; messages may carry paths, DSNs or SQL.
61            tracing::error!(
62                cause = stack_failure_cause(error.as_ref()),
63                "stack initialization cause"
64            );
65            FfiError::StackInitializationFailed.logged("stack_initialization")
66        })
67}
68
69fn stack_failure_cause(error: &(dyn std::error::Error + 'static)) -> &'static str {
70    use arc_core::{
71        event_bus::EventBusError, event_store::EventStoreError, read_model_store::ReadModelError,
72    };
73    if error.is::<EventStoreError>() {
74        "event_store"
75    } else if error.is::<ReadModelError>() {
76        "read_model_store"
77    } else if error.is::<EventBusError>() {
78        "event_bus"
79    } else {
80        "other"
81    }
82}

84#[cfg(test)]
85mod tests {
86    use super::stack_failure_cause;
87    use arc_core::{event_store::EventStoreError, read_model_store::ReadModelError};
88
89    #[test]
90    fn stack_failure_cause_is_a_fixed_category() {
91        let store: Box<dyn std::error::Error> = Box::new(EventStoreError::AggregateNotFound {
92            aggregate_id: "/private/path".into(),
93        });
94        let read: Box<dyn std::error::Error> =
95            Box::new(ReadModelError::other("postgres://secret@host/db"));
96        let other: Box<dyn std::error::Error> = "SELECT secret".into();
97        assert_eq!(stack_failure_cause(store.as_ref()), "event_store");
98        assert_eq!(stack_failure_cause(read.as_ref()), "read_model_store");
99        assert_eq!(stack_failure_cause(other.as_ref()), "other");
100    }
