11const EVENTS_SUBJECT: &str = "events.>";
12pub const DLQ_STREAM: &str = "ARC_DLQ";
13pub const DLQ_SUBJECT: &str = "dlq.>";
14const MSG_ID_HEADER: &str = "Nats-Msg-Id";
15
16/// Publishes full [`Event`] payloads to a NATS JetStream stream.
17pub struct NatsEventBus {
18    jetstream: jetstream::Context,
19    stream: String,
20}
21
22impl NatsEventBus {
23    /// Connect to NATS and idempotently ensure the event stream exists.
24    pub async fn new(url: &str, stream: &str) -> EventBusResult<Self> {
25        let client = async_nats::connect(url)
26            .await
27            .map_err(|error| EventBusError::other(format!("NATS connection failed: {error}")))?;
28        let jetstream = jetstream::new(client);
29        let stream = stream.to_string();
30
31        ensure_stream(&jetstream, stream.clone(), EVENTS_SUBJECT).await?;
32        ensure_stream(&jetstream, DLQ_STREAM.to_string(), DLQ_SUBJECT).await?;
33
34        Ok(Self { jetstream, stream })
35    }
36
37    /// The configured JetStream stream name.
38    pub fn stream(&self) -> &str {
39        &self.stream
40    }
41}
42
43#[async_trait]
44impl EventBus for NatsEventBus {
45    async fn publish(&self, events: Vec<Event>) -> EventBusResult<()> {
46        for event in events {
47            let subject = subject_for(&event);
48            let payload = serde_json::to_vec(&event).map_err(|error| {
49                EventBusError::other(format!("Event serialization failed: {error}"))
50            })?;
51            let mut headers = HeaderMap::new();
52            headers.insert(MSG_ID_HEADER, HeaderValue::from(event.event_id.to_string()));
53
54            let ack = self
55                .jetstream
56                .publish_with_headers(subject, headers, payload.into())
57                .await
58                .map_err(|error| {
59                    EventBusError::other(format!("JetStream publish failed: {error}"))
60                })?;
61
62            ack.await.map_err(|error| {
63                EventBusError::other(format!("JetStream publish ack failed: {error}"))
64            })?;
65        }
66
67        Ok(())
68    }
69
70    async fn subscribe(&mut self, _handler: Box<dyn EventHandler>) -> EventBusResult<()> {
71        Ok(())
72    }
73}
74
75async fn ensure_stream(
76    jetstream: &jetstream::Context,
77    name: String,
78    subject: &str,
79) -> EventBusResult<()> {
80    jetstream
81        .get_or_create_stream(jetstream::stream::Config {
82            name: name.clone(),
83            subjects: vec![subject.to_string()],
84            ..Default::default()
85        })
86        .await
87        .map_err(|error| {
88            EventBusError::other(format!("JetStream stream {name} setup failed: {error}"))
89        })?;
90
91    Ok(())
92}

94/// Map an event to `events.<aggregate_type>.<event_type>` using lowercase snake_case.
95pub fn subject_for(event: &Event) -> String {
96    format!(
97        "events.{}.{}",
98        to_snake_case(&event.aggregate_type),
99        to_snake_case(&event.event_type)
100    )
101}
102
103fn to_snake_case(input: &str) -> String {
104    let mut output = String::new();
105    let mut previous_kind = CharacterKind::Separator;
106    let mut characters = input.chars().peekable();
107
108    while let Some(character) = characters.next() {
109        if character.is_ascii_alphanumeric() {
110            let is_upper = character.is_ascii_uppercase();
111            let next_is_lower = characters
112                .peek()
113                .is_some_and(|next| next.is_ascii_lowercase());
114
115            let should_insert_separator = !output.is_empty()
116                && is_upper
117                && (matches!(previous_kind, CharacterKind::LowerOrDigit)
118                    || (next_is_lower && matches!(previous_kind, CharacterKind::Upper)));
119
120            if should_insert_separator {
121                output.push('_');
122            }
123
124            output.push(character.to_ascii_lowercase());
125            previous_kind = if is_upper {
126                CharacterKind::Upper
127            } else {
128                CharacterKind::LowerOrDigit
129            };
130        } else if !output.is_empty() && !matches!(previous_kind, CharacterKind::Separator) {
131            output.push('_');
132            previous_kind = CharacterKind::Separator;
133        }
134    }
135
136    while output.ends_with('_') {
137        output.pop();
138    }
139
140    output
141}
142
143#[derive(Clone, Copy)]
144enum CharacterKind {
145    Upper,
146    LowerOrDigit,
147    Separator,
148}

150#[cfg(test)]
151mod tests {
152    use super::*;
153    use serde_json::json;
154
155    #[test]
156    fn subject_for_uses_convention_shape() {
157        let event = Event::new(NewEvent {
158            aggregate_type: "User",
159            aggregate_id: "user-1",
160            sequence: 1,
161            event_type: "UserRegistered",
162            payload: json!({"email": "test@example.com"}),
163        });
164
165        assert_eq!(subject_for(&event), "events.user.user_registered");
166    }
167
168    #[test]
169    fn subject_for_normalizes_existing_separators() {
170        let event = Event::new(NewEvent {
171            aggregate_type: "Billing Account",
172            aggregate_id: "account-1",
173            sequence: 1,
174            event_type: "Payment-Method Updated",
175            payload: json!({}),
176        });
177
178        assert_eq!(
179            subject_for(&event),
180            "events.billing_account.payment_method_updated"
181        );
182    }
183
184    #[test]
185    fn snake_case_handles_digits() {
186        assert_eq!(to_snake_case("User2FAEnabled"), "user2_fa_enabled");
187    }
