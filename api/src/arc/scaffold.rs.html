129pub fn create_project(project: &NewProject) -> Result<PathBuf, String> {
130    validate_name(&project.name)?;
131    let root = project.destination.join(&project.name);
132    if root.exists() {
133        return Err(format!(
134            "destination `{}` already exists; Arc will not overwrite it",
135            root.display()
136        ));
137    }
138
139    fs::create_dir_all(&root)
140        .map_err(|error| format!("could not create `{}`: {error}", root.display()))?;
141
142    let result = write_project(project, &root);
143    if result.is_err() {
144        let _ = fs::remove_dir_all(&root);
145    }
146    result?;
147
148    if project.git {
149        initialize_git(&root)?;
150    }
151
152    Ok(root)
153}

155fn write_project(project: &NewProject, root: &Path) -> Result<(), String> {
156    for file in FILES {
157        if file.ui_only && !project.ui {
158            continue;
159        }
160        let path = root.join(file.path);
161        if let Some(parent) = path.parent() {
162            fs::create_dir_all(parent)
163                .map_err(|error| format!("could not create `{}`: {error}", parent.display()))?;
164        }
165        let contents = render(file.contents, project);
166        fs::write(&path, contents)
167            .map_err(|error| format!("could not write `{}`: {error}", path.display()))?;
168    }
169    Ok(())
170}

172fn render(template: &str, project: &NewProject) -> String {
173    let _api_requested = project.api;
174    let rendered = template
175        .replace("{{project-name}}", &project.name)
176        .replace("{{crate-name}}", &project.name.replace('-', "_"))
177        .replace("{{arc-version}}", ARC_VERSION)
178        .replace(
179            "// {{ui-module}}\n",
180            if project.ui { "mod ui;\n" } else { "" },
181        )
182        .replace(
183            "        // {{ui-host-registration}}",
184            if project.ui { "        // arc:ui-host-registration\n        .register_ui_host(ui::host())\n        .register_ui(ui::contribution())" } else { "" },
185        )
186        .replace(
187            "{{ui-routes}}",
188            if project.ui {
189                ".configure(crate::ui::config).service(actix_files::Files::new(\"/public\", \"public\"))"
190            } else {
191                ""
192            },
193        );
194
195    match std::env::var("ARC_CLI_TEST_LOCAL_ROOT") {
196        Ok(root) => rendered
197            .replace(
198                &format!("arc-core = \"{ARC_VERSION}\""),
199                &format!("arc-core = {{ path = \"{root}/crates/arc-core\" }}"),
200            )
201            .replace(
202                &format!("arc-web = \"{ARC_VERSION}\""),
203                &format!("arc-web = {{ path = \"{root}/crates/arc-web\" }}"),
204            ),
205        Err(_) => rendered,
206    }
207}

209pub(crate) fn validate_name(name: &str) -> Result<(), String> {
210    let mut chars = name.chars();
211    let valid_first = chars
212        .next()
213        .is_some_and(|character| character.is_ascii_alphabetic());
214    let valid_rest = chars.all(|character| character.is_ascii_alphanumeric() || character == '-');
215    if !valid_first || !valid_rest {
216        return Err(
217            "project name must start with an ASCII letter and contain only letters, numbers, or hyphens"
218                .to_string(),
219        );
220    }
221    Ok(())
222}

224/// Variables that make Git pick a repository other than the working directory, or inject
225/// per-invocation config from a parent `git -c` / hook. User and system config
226/// (`GIT_CONFIG_GLOBAL`, `GIT_CONFIG_SYSTEM`) and `GIT_CEILING_DIRECTORIES` are kept on purpose:
227/// they cannot redirect `git init` to another repository.
228const GIT_REPOSITORY_ENV: [&str; 10] = [
229    "GIT_DIR",
230    "GIT_WORK_TREE",
231    "GIT_INDEX_FILE",
232    "GIT_COMMON_DIR",
233    "GIT_OBJECT_DIRECTORY",
234    "GIT_ALTERNATE_OBJECT_DIRECTORIES",
235    "GIT_NAMESPACE",
236    "GIT_PREFIX",
237    "GIT_CONFIG_PARAMETERS",
238    "GIT_CONFIG_COUNT",
239];
240
241/// Builds a `git` command that cannot be redirected to a caller's repository.
242/// `inherited` simulates variables of the parent environment (used by tests).
243fn isolated_git_command(root: &Path, inherited: &[(&str, &Path)]) -> Command {
244    let mut command = Command::new("git");
245    for (name, value) in inherited {
246        command.env(name, value);
247    }
248    for name in GIT_REPOSITORY_ENV {
249        command.env_remove(name);
250    }
251    // `GIT_CONFIG_COUNT` pairs with numbered GIT_CONFIG_KEY_n / GIT_CONFIG_VALUE_n entries.
252    for (name, _) in std::env::vars_os() {
253        if let Some(name) = name.to_str() {
254            if name.starts_with("GIT_CONFIG_KEY_") || name.starts_with("GIT_CONFIG_VALUE_") {
255                command.env_remove(name);
256            }
257        }
258    }
259    command.current_dir(root);
260    command
261}

263pub(crate) fn initialize_git(root: &Path) -> Result<(), String> {
264    git_init(root, &[])
265}
266
267fn git_init(root: &Path, inherited: &[(&str, &Path)]) -> Result<(), String> {
268    let status = isolated_git_command(root, inherited)
269        .args(["init", "--quiet"])
270        .status()
271        .map_err(|error| format!("could not run `git init`: {error}; retry with --no-git"))?;
272    if !status.success() {
273        return Err("`git init` failed; retry with --no-git".to_string());
274    }
275    Ok(())
276}

291    #[test]
292    fn git_init_ignores_inherited_repository_environment() {
293        struct Cleanup(Vec<PathBuf>);
294        impl Drop for Cleanup {
295            fn drop(&mut self) {
296                for path in &self.0 {
297                    let _ = fs::remove_dir_all(path);
298                }
299            }
300        }
301        let outer = temp_root();
302        let project = temp_root().with_extension("project");
303        let _cleanup = Cleanup(vec![outer.clone(), project.clone()]);
304        fs::create_dir_all(&outer).unwrap();
305        fs::create_dir_all(&project).unwrap();
306        assert!(isolated_git_command(&outer, &[])
307            .args(["init", "--quiet"])
308            .status()
309            .unwrap()
310            .success());
311        let outer_git = outer.join(".git");
312        git_init(&project, &[("GIT_DIR", outer_git.as_path())]).unwrap();
313        assert!(project.join(".git").is_dir());
314        let bare = isolated_git_command(&outer, &[])
315            .args(["config", "--get", "core.bare"])
316            .output()
317            .unwrap();
318        assert_eq!(String::from_utf8_lossy(&bare.stdout).trim(), "false");
319    }
