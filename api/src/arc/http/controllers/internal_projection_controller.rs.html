8fn configured_token() -> Option<String> {
9    std::env::var("INTERNAL_PROJECTION_TOKEN")
10        .ok()
11        .filter(|token| !token.trim().is_empty())
12}
13
14fn bearer_token(req: &HttpRequest) -> Option<&str> {
15    req.headers()
16        .get("Authorization")
17        .and_then(|value| value.to_str().ok())
18        .and_then(|value| value.strip_prefix("Bearer "))
19}
20
21fn authorized(req: &HttpRequest) -> Result<(), Box<HttpResponse>> {
22    let Some(expected) = configured_token() else {
23        tracing::error!("INTERNAL_PROJECTION_TOKEN is not configured");
24        return Err(Box::new(
25            HttpResponse::ServiceUnavailable()
26                .json(json!({"error": "Projection handler is not configured"})),
27        ));
28    };
29
30    match bearer_token(req) {
31        Some(actual) if actual == expected => Ok(()),
32        _ => Err(Box::new(
33            HttpResponse::Unauthorized().json(json!({"error": "Unauthorized"})),
34        )),
35    }
36}

38#[post("/users/handle")]
39pub async fn handle_user_projection(
40    req: HttpRequest,
41    event: web::Json<Event>,
42    projection_engine: web::Data<ProjectionEngine>,
43) -> impl Responder {
44    if let Err(response) = authorized(&req) {
45        return *response;
46    }
47
48    if event.aggregate_type != "User" {
49        return HttpResponse::BadRequest()
50            .json(json!({"error": "Projection endpoint only accepts User events"}));
51    }
52
53    match projection_engine.process(&event).await {
54        Ok(()) => HttpResponse::NoContent().finish(),
55        Err(error) => {
56            tracing::error!(error = ?error, event_id = %event.event_id, event_type = event.event_type, "user projection failed");
57            HttpResponse::InternalServerError().json(json!({"error": "Projection failed"}))
58        }
59    }
60}
