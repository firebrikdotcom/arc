12fn main() -> ExitCode {
13    match run(env::args().skip(1)) {
14        Ok(()) => ExitCode::SUCCESS,
15        Err(message) => {
16            eprintln!("error: {message}");
17            ExitCode::FAILURE
18        }
19    }
20}

22fn run(args: impl Iterator<Item = String>) -> Result<(), String> {
23    let args: Vec<_> = args.collect();
24    // Resolve help before consuming positional names or invoking any command.
25    // This also handles help following an option that normally takes a value.
26    if print_requested_help(&args)? {
27        return Ok(());
28    }
29    let mut args = args.into_iter();
30    match args.next().as_deref() {
31        Some("new") => {
32            let name = args
33                .next()
34                .filter(|name| !name.starts_with('-'))
35                .ok_or_else(|| "usage: arc new <name> [--ui] [--api] [--no-git]".to_string())?;
36            let mut ui = false;
37            let mut api = false;
38            let mut git = true;
39
40            for argument in args {
41                match argument.as_str() {
42                    "--ui" => ui = true,
43                    "--api" => api = true,
44                    "--no-git" => git = false,
45                    unknown => return Err(format!("unknown option `{unknown}`")),
46                }
47            }
48
49            let project = NewProject {
50                name,
51                destination: PathBuf::from("."),
52                ui,
53                api,
54                git,
55            };
56            let path = create_project(&project)?;
57            println!();
58            println!("Created {} at {}", project.name, path.display());
59            println!();
60            println!("  cd {}", project.name);
61            println!("  make setup");
62            println!("  make dev");
63            Ok(())
64        }
65        Some("generate") => {
66            let kind = args.next().ok_or_else(|| {
67                "usage: arc generate resource <name> (alias: aggregate)".to_string()
68            })?;
69            if kind != "resource" && kind != "aggregate" {
70                return Err(format!(
71                    "unknown generator `{kind}`; expected `resource` or `aggregate`"
72                ));
73            }
74            let name = args.next().ok_or_else(|| {
75                "usage: arc generate resource <name> [--api] [--ui] (alias: aggregate)".to_string()
76            })?;
77            if name.starts_with('-') {
78                return Err("resource name is required; run `arc generate resource --help`".into());
79            }
80            let mut api = false;
81            let mut ui = false;
82            let mut api_auth = None;
83            let mut ui_auth = None;
84            let mut roles = Vec::new();
85            while let Some(argument) = args.next() {
86                match argument.as_str() {
87                    "--api" => api = true,
88                    "--ui" => ui = true,
89                    "--api-auth" => {
90                        api_auth = Some(args.next().ok_or("--api-auth requires jwt or none")?)
91                    }
92                    "--ui-auth" => {
93                        ui_auth = Some(args.next().ok_or("--ui-auth requires session or none")?)
94                    }
95                    "--roles" => {
96                        roles = args
97                            .next()
98                            .ok_or("--roles requires a comma-separated value")?
99                            .split(',')
100                            .map(str::to_owned)
101                            .collect()
102                    }
103                    _ => return Err(format!("unknown option `{argument}`")),
104                }
105            }
106
107            let resource = NewResource {
108                name,
109                root: PathBuf::from("."),
110                api,
111                ui,
112                api_auth,
113                ui_auth,
114                roles,
115            };
116            let created = create_resource(&resource)?;
117            println!();
118            println!("Generated {} resource:", created.type_name);
119            for path in created.files {
120                println!("  {}", path.display());
121            }
122            println!();
123            if let Some(api_path) = created.api_path {
124                println!("Run `make migrate`, then use the CRUD API at {api_path}.");
125            } else {
126                println!(
127                    "Run `make migrate` and add routes that dispatch {}Command.",
128                    created.type_name
129                );
130            }
131            Ok(())
132        }
133        Some("plugin") => {
134            let operation = args.next();
135            if operation.as_deref() == Some("migrate-auth-ui") {
136                if let Some(argument) = args.next() {
137                    return Err(format!("unexpected argument `{argument}`"));
138                }
139                plugin::migrate_auth_ui(PathBuf::from("."))?;
140                println!("Auth UI host migration complete.");
141                return Ok(());
142            }
143            if operation.as_deref() != Some("add") {
144                return Err("usage: arc plugin <add NAME|migrate-auth-ui>".into());
145            }
146            let name = args.next().ok_or("plugin name is required")?;
147            if name.starts_with('-') {
148                return Err("plugin name is required; run `arc plugin add --help`".into());
149            }
150            if let Some(argument) = args.next() {
151                return Err(format!("unexpected argument `{argument}`"));
152            }
153            plugin::add_plugin(PathBuf::from("."), &name)?;
154            println!("Installed {name}.");
155            Ok(())
156        }
157        Some("--version" | "-V") => {
158            println!("arc {}", env!("CARGO_PKG_VERSION"));
159            Ok(())
160        }
161        Some("--help" | "-h") | None => {
162            print_help();
163            Ok(())
164        }
165        Some(command) => Err(format!(
166            "unknown command `{command}`; run `arc --help` for usage"
167        )),
168    }
169}

171fn print_requested_help(args: &[String]) -> Result<bool, String> {
172    let Some(position) = args
173        .iter()
174        .position(|arg| matches!(arg.as_str(), "--help" | "-h"))
175    else {
176        return Ok(false);
177    };
178    let command = args.first().map(String::as_str);
179    let subcommand = args.get(1).filter(|_| position > 1).map(String::as_str);
180    match command {
181        Some("--help" | "-h") => print_help(),
182        Some("new") => print_new_help(),
183        Some("generate") => match subcommand {
184            None => println!("Generate application resources\n\nUsage:\n  arc generate <command>\n\nCommands:\n  resource    Generate an event-sourced CRUD resource\n  aggregate   Alias for resource\n\nOptions:\n  -h, --help  Show help"),
185            Some(kind @ ("resource" | "aggregate")) => print_generate_help(kind),
186            Some(kind) => return Err(format!("unknown generator `{kind}`; expected `resource` or `aggregate`")),
187        },
188        Some("plugin") => match subcommand {
189            None => println!("Manage Arc plugins\n\nUsage:\n  arc plugin <command>\n\nCommands:\n  add              Install an authentication plugin\n  migrate-auth-ui  Migrate a generated authentication UI host\n\nOptions:\n  -h, --help  Show help"),
190            Some("add") => println!("Install an Arc authentication plugin\n\nUsage:\n  arc plugin add <name>\n\nPlugins:\n  auth-db-session  Browser sessions, administration and RBAC\n  auth-db-jwt      JWT APIs and RBAC\n  auth-db          Database identity provider\n  auth-session     Browser session authentication\n  auth-admin       User and role administration\n  auth-jwt         JWT API authentication\n  auth-rbac        Role-based access control\n\nOptions:\n  -h, --help  Show help\n\nRun from the root of an application created by `arc new`."),
191            Some("migrate-auth-ui") => println!("Migrate an untouched generated UI to the authentication UI host\n\nUsage:\n  arc plugin migrate-auth-ui\n\nOptions:\n  -h, --help  Show help\n\nCustomized UI shells are refused with manual migration instructions."),
192            Some(operation) => return Err(format!("unknown plugin command `{operation}`")),
193        },
194        Some("--version" | "-V") => print_help(),
195        Some(command) => return Err(format!("unknown command `{command}`; run `arc --help` for usage")),
196        None => print_help(),
197    }
198    Ok(true)
199}

201fn print_help() {
202    println!(
203        "Arc application generator
204
205Usage:
206  arc new <name> [--ui] [--api] [--no-git]
207  arc plugin add <name>
208  arc plugin migrate-auth-ui
209  arc generate resource <name> [--api] [--ui] [--api-auth jwt] [--ui-auth session] [--roles admin,user]
210  arc generate aggregate <name> [--api] [--ui]
211
212Options:
213  --ui       Add Tera views and browser assets
214  --no-git   Do not initialize a Git repository
215  -h, --help Show help
216  -V, --version Show version"
217    );
218}

220fn print_generate_help(kind: &str) {
221    println!(
222        "Generate and register an event-sourced aggregate resource
223
224Usage:
225  arc generate {kind} <name> [--api] [--ui] [--api-auth jwt|none] [--ui-auth session|none] [--roles LIST]
226
227Options:
228  --api  Add public JSON CRUD endpoints
229  --ui   Add browser CRUD pages (session-protected when auth-session is installed)
230  --api-auth jwt      Explicitly protect this API resource
231  --ui-auth session   Explicitly protect this browser resource
232  --roles LIST        Require any listed role (requires auth)
233  -h, --help          Show help
234
235Run this command from the root of an application created by `arc new`."
236    );
237}
