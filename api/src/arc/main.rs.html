13fn main() -> ExitCode {
14    match run(env::args().skip(1)) {
15        Ok(()) => ExitCode::SUCCESS,
16        Err(message) => {
17            eprintln!("error: {message}");
18            ExitCode::FAILURE
19        }
20    }
21}

23fn run(args: impl Iterator<Item = String>) -> Result<(), String> {
24    let args: Vec<_> = args.collect();
25    // Resolve help before consuming positional names or invoking any command.
26    // This also handles help following an option that normally takes a value.
27    if print_requested_help(&args)? {
28        return Ok(());
29    }
30    let mut args = args.into_iter();
31    match args.next().as_deref() {
32        Some("new") => {
33            let name = args
34                .next()
35                .filter(|name| !name.starts_with('-'))
36                .ok_or_else(|| {
37                    "usage: arc new <name> [--ui] [--api] [--android] [--arc-path PATH] [--no-git]"
38                        .to_string()
39                })?;
40            let mut ui = false;
41            let mut api = false;
42            let mut git = true;
43            let mut android = false;
44            let mut arc_path = None;
45
46            while let Some(argument) = args.next() {
47                match argument.as_str() {
48                    "--ui" => ui = true,
49                    "--api" => api = true,
50                    "--no-git" => git = false,
51                    "--android" => android = true,
52                    "--arc-path" => {
53                        let path = args
54                            .next()
55                            .filter(|value| !value.is_empty() && !value.starts_with('-'))
56                            .ok_or("--arc-path requires a directory")?;
57                        arc_path = Some(PathBuf::from(path));
58                    }
59                    unknown => return Err(format!("unknown option `{unknown}`")),
60                }
61            }
62
63            if android && (ui || api) {
64                return Err("--android cannot be combined with --ui or --api".into());
65            }
66            if !android && arc_path.is_some() {
67                return Err("--arc-path requires --android".into());
68            }
69            if android {
70                let path =
71                    android::create_project(&name, &PathBuf::from("."), arc_path.as_deref(), git)?;
72                println!("Created {name} at {}\n\n  cd {name}\n  make bindings\n  make test\n  make build", path.display());
73                return Ok(());
74            }
75            let project = NewProject {
76                name,
77                destination: PathBuf::from("."),
78                ui,
79                api,
80                git,
81            };
82            let path = create_project(&project)?;
83            println!();
84            println!("Created {} at {}", project.name, path.display());
85            println!();
86            println!("  cd {}", project.name);
87            println!("  make setup");
88            println!("  make dev");
89            Ok(())
90        }
91        Some("generate") => {
92            let kind = args.next().ok_or_else(|| {
93                "usage: arc generate resource <name> [--ffi] [--api] [--ui] [--api-auth jwt|none] [--ui-auth session|none] [--roles LIST] (alias: aggregate; --ffi is incompatible with --api, --ui, --api-auth, --ui-auth, and --roles)".to_string()
94            })?;
95            if kind != "resource" && kind != "aggregate" {
96                return Err(format!(
97                    "unknown generator `{kind}`; expected `resource` or `aggregate`"
98                ));
99            }
100            let name = args.next().ok_or_else(|| {
101                "usage: arc generate resource <name> [--ffi] [--api] [--ui] [--api-auth jwt|none] [--ui-auth session|none] [--roles LIST] (alias: aggregate; --ffi is incompatible with --api, --ui, --api-auth, --ui-auth, and --roles)".to_string()
102            })?;
103            if name.starts_with('-') {
104                return Err("resource name is required; run `arc generate resource --help`".into());
105            }
106            if args.as_slice().iter().any(|arg| arg == "--ffi")
107                && args.as_slice().iter().any(|arg| {
108                    matches!(
109                        arg.as_str(),
110                        "--api" | "--ui" | "--api-auth" | "--ui-auth" | "--roles"
111                    )
112                })
113            {
114                return Err(
115                    "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles"
116                        .into(),
117                );
118            }
119            let mut api = false;
120            let mut ui = false;
121            let mut api_auth = None;
122            let mut ui_auth = None;
123            let mut roles = Vec::new();
124            let mut ffi = false;
125            let mut web_option = false;
126            while let Some(argument) = args.next() {
127                if matches!(
128                    argument.as_str(),
129                    "--api" | "--ui" | "--api-auth" | "--ui-auth" | "--roles"
130                ) {
131                    web_option = true;
132                }
133                match argument.as_str() {
134                    "--ffi" => ffi = true,
135                    "--api" => api = true,
136                    "--ui" => ui = true,
137                    "--api-auth" => {
138                        api_auth = Some(args.next().ok_or("--api-auth requires jwt or none")?)
139                    }
140                    "--ui-auth" => {
141                        ui_auth = Some(args.next().ok_or("--ui-auth requires session or none")?)
142                    }
143                    "--roles" => {
144                        roles = args
145                            .next()
146                            .ok_or("--roles requires a comma-separated value")?
147                            .split(',')
148                            .map(str::to_owned)
149                            .collect()
150                    }
151                    _ => return Err(format!("unknown option `{argument}`")),
152                }
153            }
154
155            if ffi && web_option {
156                return Err(
157                    "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles"
158                        .into(),
159                );
160            }
161            let android_project = android::read_manifest(std::path::Path::new("."))?;
162            if ffi && android_project.is_none() {
163                return Err(
164                    "--ffi requires an Arc Android project; run arc new <name> --android".into(),
165                );
166            }
167            if !ffi && android_project.is_some() {
168                return Err("Android resources require --ffi".into());
169            }
170            let resource = NewResource {
171                name,
172                root: PathBuf::from("."),
173                api,
174                ui,
175                api_auth,
176                ui_auth,
177                roles,
178            };
179            let created = create_resource(&resource)?;
180            println!();
181            println!("Generated {} resource:", created.type_name);
182            for path in created.files {
183                println!("  {}", path.display());
184            }
185            println!();
186            if ffi {
187                println!("Run `make test build`; schema is applied when the resource core opens.");
188            } else if let Some(api_path) = created.api_path {
189                println!("Run `make migrate`, then use the CRUD API at {api_path}.");
190            } else {
191                println!(
192                    "Run `make migrate` and add routes that dispatch {}Command.",
193                    created.type_name
194                );
195            }
196            Ok(())
197        }
198        Some("doctor") => {
199            if args.next().as_deref() != Some("--android") {
200                return Err("usage: arc doctor --android".into());
201            }
202            if let Some(argument) = args.next() {
203                return Err(format!("unknown option `{argument}`"));
204            }
205            android::doctor()
206        }
207        Some("plugin") => {
208            let operation = args.next();
209            if operation.as_deref() == Some("migrate-auth-ui") {
210                if let Some(argument) = args.next() {
211                    return Err(format!("unexpected argument `{argument}`"));
212                }
213                plugin::migrate_auth_ui(PathBuf::from("."))?;
214                println!("Auth UI host migration complete.");
215                return Ok(());
216            }
217            if operation.as_deref() != Some("add") {
218                return Err("usage: arc plugin <add NAME|migrate-auth-ui>".into());
219            }
220            let name = args.next().ok_or("plugin name is required")?;
221            if name.starts_with('-') {
222                return Err("plugin name is required; run `arc plugin add --help`".into());
223            }
224            if let Some(argument) = args.next() {
225                return Err(format!("unexpected argument `{argument}`"));
226            }
227            plugin::add_plugin(PathBuf::from("."), &name)?;
228            println!("Installed {name}.");
229            Ok(())
230        }
231        Some("--version" | "-V") => {
232            println!("arc {}", env!("CARGO_PKG_VERSION"));
233            Ok(())
234        }
235        Some("--help" | "-h") | None => {
236            print_help();
237            Ok(())
238        }
239        Some(command) => Err(format!(
240            "unknown command `{command}`; run `arc --help` for usage"
241        )),
242    }
243}

245fn print_requested_help(args: &[String]) -> Result<bool, String> {
246    let Some(position) = args
247        .iter()
248        .position(|arg| matches!(arg.as_str(), "--help" | "-h"))
249    else {
250        return Ok(false);
251    };
252    let command = args.first().map(String::as_str);
253    let subcommand = args.get(1).filter(|_| position > 1).map(String::as_str);
254    match command {
255        Some("--help" | "-h") => print_help(),
256        Some("new") => print_new_help(),
257        Some("doctor") => println!("Diagnose Android prerequisites without installing anything\n\nUsage:\n  arc doctor --android\n\nOptions:\n  --android  Check Rust, SDK, NDK, Java and emulator tooling\n  -h, --help Show help"),
258        Some("generate") => match subcommand {
259            None => println!("Generate application resources\n\nUsage:\n  arc generate <command>\n\nCommands:\n  resource    Generate an event-sourced CRUD resource\n  aggregate   Alias for resource\n\nOptions:\n  --ffi  Generate native resource adapters in an Arc Android project; incompatible with --api, --ui, --api-auth, --ui-auth, and --roles\n  -h, --help  Show help"),
260            Some(kind @ ("resource" | "aggregate")) => print_generate_help(kind),
261            Some(kind) => return Err(format!("unknown generator `{kind}`; expected `resource` or `aggregate`")),
262        },
263        Some("plugin") => match subcommand {
264            None => println!("Manage Arc plugins\n\nUsage:\n  arc plugin <command>\n\nCommands:\n  add              Install an authentication plugin\n  migrate-auth-ui  Migrate a generated authentication UI host\n\nOptions:\n  -h, --help  Show help"),
265            Some("add") => println!("Install an Arc authentication plugin\n\nUsage:\n  arc plugin add <name>\n\nPlugins:\n  auth-db-session  Browser sessions, administration and RBAC\n  auth-db-jwt      JWT APIs and RBAC\n  auth-db          Database identity provider\n  auth-session     Browser session authentication\n  auth-admin       User and role administration\n  auth-jwt         JWT API authentication\n  auth-rbac        Role-based access control\n\nOptions:\n  -h, --help  Show help\n\nRun from the root of an application created by `arc new`."),
266            Some("migrate-auth-ui") => println!("Migrate an untouched generated UI to the authentication UI host\n\nUsage:\n  arc plugin migrate-auth-ui\n\nOptions:\n  -h, --help  Show help\n\nCustomized UI shells are refused with manual migration instructions."),
267            Some(operation) => return Err(format!("unknown plugin command `{operation}`")),
268        },
269        Some("--version" | "-V") => print_help(),
270        Some(command) => return Err(format!("unknown command `{command}`; run `arc --help` for usage")),
271        None => print_help(),
272    }
273    Ok(true)
274}

336#[cfg(test)]
337mod android_options_tests {
338    use super::*;
339    fn invoke(args: &[&str]) -> Result<(), String> {
340        run(args.iter().map(|arg| (*arg).into()))
341    }
342    #[test]
343    fn android_options_fail_before_writes() {
344        for option in ["--api", "--ui"] {
345            assert_eq!(
346                invoke(&["new", "untouched", "--android", option]).unwrap_err(),
347                "--android cannot be combined with --ui or --api"
348            );
349        }
350        assert_eq!(
351            invoke(&["new", "untouched", "--arc-path", "."]).unwrap_err(),
352            "--arc-path requires --android"
353        );
354        for args in [
355            vec!["new", "untouched", "--android", "--arc-path"],
356            vec!["new", "untouched", "--arc-path", "--no-git"],
357        ] {
358            assert_eq!(
359                invoke(&args).unwrap_err(),
360                "--arc-path requires a directory"
361            );
362        }
363        for kind in ["resource", "aggregate"] {
364            for option in ["--api", "--ui", "--api-auth", "--ui-auth", "--roles"] {
365                let mut args = vec!["generate", kind, "Note", "--ffi", option];
366                if matches!(option, "--api-auth" | "--ui-auth" | "--roles") {
367                    assert_eq!(invoke(&args).unwrap_err(), "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles");
368                    args.push("none");
369                }
370                assert_eq!(
371                    invoke(&args).unwrap_err(),
372                    "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles"
373                );
374            }
375            assert_eq!(
376                invoke(&["generate", kind, "Note", "--ffi"]).unwrap_err(),
377                "--ffi requires an Arc Android project; run arc new <name> --android"
378            );
379        }
380    }
381    #[test]
382    fn help_precedes_new_android_validation_and_execution() {
383        for flag in ["--help", "-h"] {
384            for mut args in [
385                vec!["new", "--android", "--arc-path"],
386                vec!["new", "No-Writes", "--android", "--ui"],
387                vec!["generate", "resource", "Note", "--ffi", "--roles"],
388                vec!["generate", "aggregate", "Note", "--ffi"],
389                vec!["doctor", "--android"],
390            ] {
391                for index in 1..=args.len() {
392                    let mut candidate = args.clone();
393                    candidate.insert(index, flag);
394                    assert!(invoke(&candidate).is_ok(), "{candidate:?}");
395                }
396                args.push(flag);
397                assert!(invoke(&args).is_ok());
398            }
399        }
