14fn main() -> ExitCode {
15    match run(env::args().skip(1)) {
16        Ok(()) => ExitCode::SUCCESS,
17        Err(message) => {
18            eprintln!("error: {message}");
19            ExitCode::FAILURE
20        }
21    }
22}

24fn run(args: impl Iterator<Item = String>) -> Result<(), String> {
25    let args: Vec<_> = args.collect();
26    // Resolve help before consuming positional names or invoking any command.
27    // This also handles help following an option that normally takes a value.
28    if print_requested_help(&args)? {
29        return Ok(());
30    }
31    let mut args = args.into_iter();
32    match args.next().as_deref() {
33        Some("new") => {
34            let name = args
35                .next()
36                .filter(|name| !name.starts_with('-'))
37                .ok_or_else(|| {
38                    "usage: arc new <name> [--ui] [--api] [--android] [--arc-path PATH] [--no-git]"
39                        .to_string()
40                })?;
41            let mut ui = false;
42            let mut api = false;
43            let mut git = true;
44            let mut android = false;
45            let mut arc_path = None;
46
47            while let Some(argument) = args.next() {
48                match argument.as_str() {
49                    "--ui" => ui = true,
50                    "--api" => api = true,
51                    "--no-git" => git = false,
52                    "--android" => android = true,
53                    "--arc-path" => {
54                        let path = args
55                            .next()
56                            .filter(|value| !value.is_empty() && !value.starts_with('-'))
57                            .ok_or("--arc-path requires a directory")?;
58                        arc_path = Some(PathBuf::from(path));
59                    }
60                    unknown => return Err(format!("unknown option `{unknown}`")),
61                }
62            }
63
64            if android && (ui || api) {
65                return Err("--android cannot be combined with --ui or --api".into());
66            }
67            if !android && arc_path.is_some() {
68                return Err("--arc-path requires --android".into());
69            }
70            if android {
71                let path =
72                    android::create_project(&name, &PathBuf::from("."), arc_path.as_deref(), git)?;
73                println!("Created {name} at {}\n\n  cd {name}\n  make bindings\n  make test\n  make build", path.display());
74                return Ok(());
75            }
76            let project = NewProject {
77                name,
78                destination: PathBuf::from("."),
79                ui,
80                api,
81                git,
82            };
83            let path = create_project(&project)?;
84            println!();
85            println!("Created {} at {}", project.name, path.display());
86            println!();
87            println!("  cd {}", project.name);
88            println!("  make setup");
89            println!("  make dev");
90            Ok(())
91        }
92        Some("generate") => {
93            let kind = args.next().ok_or_else(|| {
94                "usage: arc generate resource <name> [--ffi] [--api] [--ui] [--api-auth jwt|none] [--ui-auth session|none] [--roles LIST] (alias: aggregate; --ffi is incompatible with --api, --ui, --api-auth, --ui-auth, and --roles)".to_string()
95            })?;
96            if kind != "resource" && kind != "aggregate" {
97                return Err(format!(
98                    "unknown generator `{kind}`; expected `resource` or `aggregate`"
99                ));
100            }
101            let name = args.next().ok_or_else(|| {
102                "usage: arc generate resource <name> [--ffi] [--api] [--ui] [--api-auth jwt|none] [--ui-auth session|none] [--roles LIST] (alias: aggregate; --ffi is incompatible with --api, --ui, --api-auth, --ui-auth, and --roles)".to_string()
103            })?;
104            if name.starts_with('-') {
105                return Err("resource name is required; run `arc generate resource --help`".into());
106            }
107            if args.as_slice().iter().any(|arg| arg == "--ffi")
108                && args.as_slice().iter().any(|arg| {
109                    matches!(
110                        arg.as_str(),
111                        "--api" | "--ui" | "--api-auth" | "--ui-auth" | "--roles"
112                    )
113                })
114            {
115                return Err(
116                    "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles"
117                        .into(),
118                );
119            }
120            let mut api = false;
121            let mut ui = false;
122            let mut api_auth = None;
123            let mut ui_auth = None;
124            let mut roles = Vec::new();
125            let mut ffi = false;
126            let mut web_option = false;
127            while let Some(argument) = args.next() {
128                if matches!(
129                    argument.as_str(),
130                    "--api" | "--ui" | "--api-auth" | "--ui-auth" | "--roles"
131                ) {
132                    web_option = true;
133                }
134                match argument.as_str() {
135                    "--ffi" => ffi = true,
136                    "--api" => api = true,
137                    "--ui" => ui = true,
138                    "--api-auth" => {
139                        api_auth = Some(args.next().ok_or("--api-auth requires jwt or none")?)
140                    }
141                    "--ui-auth" => {
142                        ui_auth = Some(args.next().ok_or("--ui-auth requires session or none")?)
143                    }
144                    "--roles" => {
145                        roles = args
146                            .next()
147                            .ok_or("--roles requires a comma-separated value")?
148                            .split(',')
149                            .map(str::to_owned)
150                            .collect()
151                    }
152                    _ => return Err(format!("unknown option `{argument}`")),
153                }
154            }
155
156            if ffi && web_option {
157                return Err(
158                    "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles"
159                        .into(),
160                );
161            }
162            let android_project = android::read_manifest(std::path::Path::new("."))?;
163            if ffi && android_project.is_none() {
164                return Err(
165                    "--ffi requires an Arc Android project; run arc new <name> --android".into(),
166                );
167            }
168            if !ffi && android_project.is_some() {
169                return Err("Android resources require --ffi".into());
170            }
171            let resource = NewResource {
172                name,
173                root: PathBuf::from("."),
174                api,
175                ui,
176                api_auth,
177                ui_auth,
178                roles,
179            };
180            let created = create_resource(&resource)?;
181            println!();
182            println!("Generated {} resource:", created.type_name);
183            for path in created.files {
184                println!("  {}", path.display());
185            }
186            println!();
187            if ffi {
188                println!("Run `make test build`; schema is applied when the resource core opens.");
189            } else if let Some(api_path) = created.api_path {
190                println!("Run `make migrate`, then use the CRUD API at {api_path}.");
191            } else {
192                println!(
193                    "Run `make migrate` and add routes that dispatch {}Command.",
194                    created.type_name
195                );
196            }
197            Ok(())
198        }
199        Some("console") => {
200            let mut options = console::ConsoleOptions {
201                app_modules: true,
202                irust_args: Vec::new(),
203            };
204            while let Some(argument) = args.next() {
205                match argument.as_str() {
206                    "--no-app-modules" => options.app_modules = false,
207                    "--" => options.irust_args.extend(args.by_ref()),
208                    unknown => return Err(format!("unknown option `{unknown}`")),
209                }
210            }
211            console::run(std::path::Path::new("."), &options)
212        }
213        Some("doctor") => {
214            if args.next().as_deref() != Some("--android") {
215                return Err("usage: arc doctor --android".into());
216            }
217            if let Some(argument) = args.next() {
218                return Err(format!("unknown option `{argument}`"));
219            }
220            android::doctor()
221        }
222        Some("plugin") => {
223            let operation = args.next();
224            if operation.as_deref() == Some("migrate-auth-ui") {
225                if let Some(argument) = args.next() {
226                    return Err(format!("unexpected argument `{argument}`"));
227                }
228                plugin::migrate_auth_ui(PathBuf::from("."))?;
229                println!("Auth UI host migration complete.");
230                return Ok(());
231            }
232            if operation.as_deref() != Some("add") {
233                return Err("usage: arc plugin <add NAME|migrate-auth-ui>".into());
234            }
235            let name = args.next().ok_or("plugin name is required")?;
236            if name.starts_with('-') {
237                return Err("plugin name is required; run `arc plugin add --help`".into());
238            }
239            if let Some(argument) = args.next() {
240                return Err(format!("unexpected argument `{argument}`"));
241            }
242            plugin::add_plugin(PathBuf::from("."), &name)?;
243            println!("Installed {name}.");
244            Ok(())
245        }
246        Some("--version" | "-V") => {
247            println!("arc {}", env!("CARGO_PKG_VERSION"));
248            Ok(())
249        }
250        Some("--help" | "-h") | None => {
251            print_help();
252            Ok(())
253        }
254        Some(command) => Err(format!(
255            "unknown command `{command}`; run `arc --help` for usage"
256        )),
257    }
258}

260fn print_requested_help(args: &[String]) -> Result<bool, String> {
261    // Arguments after `arc console --` belong to irust.
262    let scanned = match args.first().map(String::as_str) {
263        Some("console") => args
264            .iter()
265            .position(|arg| arg == "--")
266            .map_or(args, |end| &args[..end]),
267        _ => args,
268    };
269    let Some(position) = scanned
270        .iter()
271        .position(|arg| matches!(arg.as_str(), "--help" | "-h"))
272    else {
273        return Ok(false);
274    };
275    let command = args.first().map(String::as_str);
276    let subcommand = args.get(1).filter(|_| position > 1).map(String::as_str);
277    match command {
278        Some("--help" | "-h") => print_help(),
279        Some("new") => print_new_help(),
280        Some("console") => print_console_help(),
281        Some("doctor") => println!("Diagnose Android prerequisites without installing anything\n\nUsage:\n  arc doctor --android\n\nOptions:\n  --android  Check Rust, SDK, NDK, Java and emulator tooling\n  -h, --help Show help"),
282        Some("generate") => match subcommand {
283            None => println!("Generate application resources\n\nUsage:\n  arc generate <command>\n\nCommands:\n  resource    Generate an event-sourced CRUD resource\n  aggregate   Alias for resource\n\nOptions:\n  --ffi  Generate native resource adapters in an Arc Android project; incompatible with --api, --ui, --api-auth, --ui-auth, and --roles\n  -h, --help  Show help"),
284            Some(kind @ ("resource" | "aggregate")) => print_generate_help(kind),
285            Some(kind) => return Err(format!("unknown generator `{kind}`; expected `resource` or `aggregate`")),
286        },
287        Some("plugin") => match subcommand {
288            None => println!("Manage Arc plugins\n\nUsage:\n  arc plugin <command>\n\nCommands:\n  add              Install an authentication plugin\n  migrate-auth-ui  Migrate a generated authentication UI host\n\nOptions:\n  -h, --help  Show help"),
289            Some("add") => println!("Install an Arc authentication plugin\n\nUsage:\n  arc plugin add <name>\n\nPlugins:\n  auth-db-session  Browser sessions, administration and RBAC\n  auth-db-jwt      JWT APIs and RBAC\n  auth-db          Database identity provider\n  auth-session     Browser session authentication\n  auth-admin       User and role administration\n  auth-jwt         JWT API authentication\n  auth-rbac        Role-based access control\n\nOptions:\n  -h, --help  Show help\n\nRun from the root of an application created by `arc new`."),
290            Some("migrate-auth-ui") => println!("Migrate an untouched generated UI to the authentication UI host\n\nUsage:\n  arc plugin migrate-auth-ui\n\nOptions:\n  -h, --help  Show help\n\nCustomized UI shells are refused with manual migration instructions."),
291            Some(operation) => return Err(format!("unknown plugin command `{operation}`")),
292        },
293        Some("--version" | "-V") => print_help(),
294        Some(command) => return Err(format!("unknown command `{command}`; run `arc --help` for usage")),
295        None => print_help(),
296    }
297    Ok(true)
298}

384#[cfg(test)]
385mod android_options_tests {
386    use super::*;
387    fn invoke(args: &[&str]) -> Result<(), String> {
388        run(args.iter().map(|arg| (*arg).into()))
389    }
390    #[test]
391    fn android_options_fail_before_writes() {
392        for option in ["--api", "--ui"] {
393            assert_eq!(
394                invoke(&["new", "untouched", "--android", option]).unwrap_err(),
395                "--android cannot be combined with --ui or --api"
396            );
397        }
398        assert_eq!(
399            invoke(&["new", "untouched", "--arc-path", "."]).unwrap_err(),
400            "--arc-path requires --android"
401        );
402        for args in [
403            vec!["new", "untouched", "--android", "--arc-path"],
404            vec!["new", "untouched", "--arc-path", "--no-git"],
405        ] {
406            assert_eq!(
407                invoke(&args).unwrap_err(),
408                "--arc-path requires a directory"
409            );
410        }
411        for kind in ["resource", "aggregate"] {
412            for option in ["--api", "--ui", "--api-auth", "--ui-auth", "--roles"] {
413                let mut args = vec!["generate", kind, "Note", "--ffi", option];
414                if matches!(option, "--api-auth" | "--ui-auth" | "--roles") {
415                    assert_eq!(invoke(&args).unwrap_err(), "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles");
416                    args.push("none");
417                }
418                assert_eq!(
419                    invoke(&args).unwrap_err(),
420                    "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles"
421                );
422            }
423            assert_eq!(
424                invoke(&["generate", kind, "Note", "--ffi"]).unwrap_err(),
425                "--ffi requires an Arc Android project; run arc new <name> --android"
426            );
427        }
428    }
429    #[test]
430    fn help_precedes_new_android_validation_and_execution() {
431        for flag in ["--help", "-h"] {
432            for mut args in [
433                vec!["new", "--android", "--arc-path"],
434                vec!["new", "No-Writes", "--android", "--ui"],
435                vec!["generate", "resource", "Note", "--ffi", "--roles"],
436                vec!["generate", "aggregate", "Note", "--ffi"],
437                vec!["doctor", "--android"],
438            ] {
439                for index in 1..=args.len() {
440                    let mut candidate = args.clone();
441                    candidate.insert(index, flag);
442                    assert!(invoke(&candidate).is_ok(), "{candidate:?}");
443                }
444                args.push(flag);
445                assert!(invoke(&args).is_ok());
446            }
447        }
448    }
