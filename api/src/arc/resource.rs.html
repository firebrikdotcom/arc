39pub fn create_resource(resource: &NewResource) -> Result<CreatedResource, String> {
40    let names = parse_name(&resource.name)?;
41    if let Some(manifest) = crate::android::read_manifest(&resource.root)? {
42        return create_android_resource(resource, &names, manifest);
43    }
44    validate_project(&resource.root)?;
45    validate_auth(resource)?;
46
47    let domain_path = resource.root.join("src/domain.rs");
48    let main_path = resource.root.join("src/main.rs");
49    let domain = read_domain(&domain_path)?;
50    let main = read_main(&main_path)?;
51
52    let migration_version = next_migration_version(&resource.root.join("migrations"))?;
53    if resource.ui && !resource.root.join("src/ui.rs").is_file() {
54        return Err("resource UI requires an application created with `arc new --ui`".to_string());
55    }
56    let manifest = fs::read_to_string(resource.root.join("Cargo.toml"))
57        .map_err(|error| format!("could not read application manifest: {error}"))?;
58    let effective_ui_auth = if resource.ui && manifest.contains("arc-auth-session") {
59        Some("session")
60    } else {
61        resource.ui_auth.as_deref()
62    };
63    let files = generated_files(
64        &names,
65        migration_version,
66        resource.api,
67        resource.ui,
68        resource.api_auth.as_deref(),
69        effective_ui_auth,
70        &resource.roles,
71    );
72    let collisions = files
73        .iter()
74        .map(|file| resource.root.join(&file.relative))
75        .filter(|path| path.exists())
76        .collect::<Vec<_>>();
77    if !collisions.is_empty() {
78        return Err(format!(
79            "Arc will not overwrite existing resource files: {}",
80            collisions
81                .iter()
82                .map(|path| path.display().to_string())
83                .collect::<Vec<_>>()
84                .join(", ")
85        ));
86    }
87
88    let module_line = format!("pub mod {};", names.module);
89    if domain.lines().any(|line| line.trim() == module_line) {
90        return Err(format!(
91            "resource module `{}` is already registered",
92            names.module
93        ));
94    }
95
96    let import = format!(
97        "use crate::domain::{}::aggregate::{}Aggregate;\nuse crate::domain::{}::projector::{{{}Projector, {}_VIEW}};",
98        names.module, names.type_name, names.module, names.type_name, names.constant
99    );
100    let registration = format!(
101        "        .register_aggregate::<{}Aggregate>()\n        .register_projector({}Projector, {}_VIEW){}",
102        names.type_name, names.type_name, names.constant,
103        if resource.ui { format!("\n        .register_ui(crate::domain::{}::ui::contribution())", names.module) } else { String::new() }
104    );
105    let updated_domain = domain.replace(DOMAIN_MARKER, &format!("{DOMAIN_MARKER}\n{module_line}"));
106    let updated_main = main
107        .replace(IMPORT_MARKER, &format!("{IMPORT_MARKER}\n{import}"))
108        .replace(
109            REGISTRATION_MARKER,
110            &format!("{registration}\n{REGISTRATION_MARKER}"),
111        );
112
113    for file in &files {
114        let path = resource.root.join(&file.relative);
115        let parent = path.parent().expect("generated files always have a parent");
116        fs::create_dir_all(parent)
117            .map_err(|error| format!("could not create `{}`: {error}", parent.display()))?;
118        fs::write(&path, &file.contents)
119            .map_err(|error| format!("could not write `{}`: {error}", path.display()))?;
120    }
121    fs::write(&domain_path, updated_domain)
122        .map_err(|error| format!("could not update `{}`: {error}", domain_path.display()))?;
123    fs::write(&main_path, updated_main)
124        .map_err(|error| format!("could not update `{}`: {error}", main_path.display()))?;
125    if resource.api {
126        register_api_route(&resource.root, &names)?;
127    }
128    if resource.ui {
129        register_ui_route(&resource.root, &names)?;
130    }
131    format_project(&resource.root)?;
132
133    Ok(CreatedResource {
134        type_name: names.type_name,
135        files: files.into_iter().map(|file| file.relative).collect(),
136        api_path: resource.api.then(|| format!("/api/{}", names.view)),
137    })
138}

153fn validate_auth(resource: &NewResource) -> Result<(), String> {
154    if resource
155        .api_auth
156        .as_deref()
157        .is_some_and(|v| v != "none" && v != "jwt")
158    {
159        return Err("--api-auth must be jwt or none".into());
160    }
161    if resource
162        .ui_auth
163        .as_deref()
164        .is_some_and(|v| v != "none" && v != "session")
165    {
166        return Err("--ui-auth must be session or none".into());
167    }
168    if resource.api_auth.as_deref() == Some("jwt") && !resource.api {
169        return Err("--api-auth jwt requires --api".into());
170    }
171    if resource.ui_auth.as_deref() == Some("session") && !resource.ui {
172        return Err("--ui-auth session requires --ui".into());
173    }
174    if !resource.roles.is_empty()
175        && resource.api_auth.as_deref() != Some("jwt")
176        && resource.ui_auth.as_deref() != Some("session")
177    {
178        return Err("--roles requires an authenticated resource".into());
179    }
180    let manifest =
181        fs::read_to_string(resource.root.join("Cargo.toml")).map_err(|e| e.to_string())?;
182    if resource.api_auth.as_deref() == Some("jwt") && !manifest.contains("arc-auth-jwt") {
183        return Err("install JWT capability first: arc plugin add auth-db-jwt".into());
184    }
185    if resource.ui_auth.as_deref() == Some("session") && !manifest.contains("arc-auth-session") {
186        return Err("install session capability first: arc plugin add auth-db-session".into());
187    }
188    if !resource.roles.is_empty() && !manifest.contains("arc-auth-rbac") {
189        return Err("install RBAC capability first: arc plugin add auth-rbac".into());
190    }
191    Ok(())
192}

194fn register_ui_route(root: &Path, names: &Names) -> Result<(), String> {
195    let path = root.join("src/routes.rs");
196    let mut contents = read_file(&path)?;
197    let marker = "    // arc:browser-resource-routes";
198    if !contents.contains(marker) {
199        let anchor = "    cfg.service(health)";
200        if !contents.contains(anchor) {
201            return Err(format!(
202                "`{}` has no route configuration anchor",
203                path.display()
204            ));
205        }
206        contents = contents.replace(anchor, &format!("{marker}\n{anchor}"));
207    }
208    let registration = format!("    crate::domain::{}::ui::config(cfg);", names.module);
209    contents = contents.replace(marker, &format!("{registration}\n{marker}"));
210    fs::write(&path, contents)
211        .map_err(|error| format!("could not update `{}`: {error}", path.display()))?;
212
213    Ok(())
214}

216fn register_api_route(root: &Path, names: &Names) -> Result<(), String> {
217    let path = root.join("src/routes.rs");
218    let mut contents = read_file(&path)?;
219    if !contents.contains(API_ROUTE_MARKER) {
220        let closing = contents.rfind('}').ok_or_else(|| {
221            format!(
222                "`{}` has no route configuration closing brace",
223                path.display()
224            )
225        })?;
226        contents.insert_str(closing, &format!("    {API_ROUTE_MARKER}\n"));
227    }
228    let registration = format!("    crate::domain::{}::api::config(_cfg);", names.module);
229    if contents
230        .lines()
231        .any(|line| line.trim() == registration.trim())
232    {
233        return Err(format!(
234            "resource API `{}` is already registered",
235            names.module
236        ));
237    }
238    contents = contents.replace(
239        API_ROUTE_MARKER,
240        &format!("{registration}\n{API_ROUTE_MARKER}"),
241    );
242    fs::write(&path, contents)
243        .map_err(|error| format!("could not update `{}`: {error}", path.display()))
244}

272fn read_main(path: &Path) -> Result<String, String> {
273    let mut contents = read_file(path)?;
274    if !contents.contains(IMPORT_MARKER) {
275        let anchor = "use crate::domain::AppAggregate;";
276        if !contents.contains(anchor) {
277            return Err(format!(
278                "`{}` has no Arc resource import marker or generated AppAggregate import",
279                path.display()
280            ));
281        }
282        contents = contents.replace(anchor, &format!("{anchor}\n{IMPORT_MARKER}"));
283    }
284    if !contents.contains(REGISTRATION_MARKER) {
285        let anchor = "        .register_aggregate::<AppAggregate>()";
286        if !contents.contains(anchor) {
287            return Err(format!(
288                "`{}` has no Arc resource registration marker or generated AppAggregate registration",
289                path.display()
290            ));
291        }
292        contents = contents.replace(anchor, &format!("{anchor}\n{REGISTRATION_MARKER}"));
293    }
294    Ok(contents)

297fn parse_name(input: &str) -> Result<Names, String> {
298    if input.is_empty()
299        || !input
300            .chars()
301            .all(|character| character.is_ascii_alphanumeric() || matches!(character, '-' | '_'))
302    {
303        return Err(
304            "resource name must contain only ASCII letters, numbers, hyphens, or underscores"
305                .to_string(),
306        );
307    }
308
309    let mut words = Vec::<String>::new();
310    let mut current = String::new();
311    for character in input.chars() {
312        if character == '-' || character == '_' {
313            if !current.is_empty() {
314                words.push(current);
315                current = String::new();
316            }
317        } else if character.is_ascii_uppercase()
318            && current
319                .chars()
320                .last()
321                .is_some_and(|last| last.is_ascii_lowercase())
322        {
323            words.push(current);
324            current = character.to_ascii_lowercase().to_string();
325        } else {
326            current.push(character.to_ascii_lowercase());
327        }
328    }
329    if !current.is_empty() {
330        words.push(current);
331    }
332    if words.is_empty()
333        || !words[0]
334            .chars()
335            .next()
336            .is_some_and(|character| character.is_ascii_alphabetic())
337    {
338        return Err("resource name must start with an ASCII letter".to_string());
339    }
340
341    let module = words.join("_");
342    if is_rust_keyword(&module) {
343        return Err(format!(
344            "resource name `{input}` is a reserved Rust keyword"
345        ));
346    }
347    let type_name = words
348        .iter()
349        .map(|word| {
350            let mut characters = word.chars();
351            characters
352                .next()
353                .map(|first| first.to_ascii_uppercase().to_string() + characters.as_str())
354                .unwrap_or_default()
355        })
356        .collect::<String>();
357
358    Ok(Names {
359        type_name,
360        constant: pluralize(&module).to_ascii_uppercase(),
361        view: pluralize(&module),
362        module,
363    })
364}

366fn pluralize(name: &str) -> String {
367    if name.ends_with('y')
368        && name
369            .chars()
370            .rev()
371            .nth(1)
372            .is_some_and(|character| !matches!(character, 'a' | 'e' | 'i' | 'o' | 'u'))
373    {
374        format!("{}ies", &name[..name.len() - 1])
375    } else if name.ends_with('s')
376        || name.ends_with('x')
377        || name.ends_with('z')
378        || name.ends_with("ch")
379        || name.ends_with("sh")
380    {
381        format!("{name}es")
382    } else {
383        format!("{name}s")
384    }
385}

430fn next_migration_version(migrations: &Path) -> Result<String, String> {
431    let entries = fs::read_dir(migrations)
432        .map_err(|error| format!("could not read `{}`: {error}", migrations.display()))?;
433    let max = entries
434        .filter_map(Result::ok)
435        .filter_map(|entry| entry.file_name().into_string().ok())
436        .filter_map(|name| name.split('_').next()?.parse::<u64>().ok())
437        .max()
438        .unwrap_or(0);
439    Ok(format!("{:014}", max + 1))
440}

464fn render_files(
465    names: &Names,
466    migration_version: &str,
467    api: bool,
468    ui: bool,
469    api_auth: Option<&str>,
470    ui_auth: Option<&str>,
471    roles: &[String],
472    prefix: &Path,
473) -> Vec<GeneratedFile> {
474    let base = prefix.join(format!("src/domain/{}", names.module));
475    let migration = prefix.join(format!(
476        "migrations/{}_{}_view",
477        migration_version, names.view
478    ));
479    let replacements = |template: &str| {
480        let role_wrap = format!(
481            ".wrap(RequireRoles::new(&[{}]))",
482            roles
483                .iter()
484                .map(|r| format!("\"{r}\""))
485                .collect::<Vec<_>>()
486                .join(",")
487        );
488        let rendered = template
489            .replace("{{Type}}", &names.type_name)
490            .replace("{{module}}", &names.module)
491            .replace("{{view}}", &names.view)
492            .replace("{{CONSTANT}}", &names.constant)
493            .replace("{{api-module}}", if api { "pub mod api;\n" } else { "" })
494            .replace("{{ui-module}}", if ui { "pub mod ui;\n" } else { "" });
495        let rendered = rendered
496            .replace(
497                "{{api-auth-import}}",
498                if api_auth == Some("jwt") {
499                    "use arc_auth_jwt::RequireJwt;\n"
500                } else {
501                    ""
502                },
503            )
504            .replace(
505                "{{api-auth-wrap}}",
506                if api_auth == Some("jwt") {
507                    ".wrap(RequireJwt)"
508                } else {
509                    ""
510                },
511            )
512            .replace(
513                "{{api-role-import}}",
514                if !roles.is_empty() && api_auth == Some("jwt") {
515                    "use arc_auth_rbac::RequireRoles;\n"
516                } else {
517                    ""
518                },
519            )
520            .replace(
521                "{{api-role-wrap}}",
522                if !roles.is_empty() && api_auth == Some("jwt") {
523                    &role_wrap
524                } else {
525                    ""
526                },
527            )
528            .replace(
529                "{{ui-auth-import}}",
530                if ui_auth == Some("session") {
531                    "use arc_auth_session::RequireSession;\nuse arc_web::http::middlewares::idle_timeout_middleware::IdleTimeoutMiddleware;\n"
532                } else {
533                    ""
534                },
535            )
536            .replace(
537                "{{ui-auth-wrap}}",
538                if ui_auth == Some("session") {
539                    ".wrap(RequireSession).wrap(IdleTimeoutMiddleware::from_env())"
540                } else {
541                    ""
542                },
543            )
544            .replace(
545                "{{ui-role-import}}",
546                if !roles.is_empty() && ui_auth == Some("session") {
547                    "use arc_auth_rbac::RequireRoles;\n"
548                } else {
549                    ""
550                },
551            )
552            .replace(
553                "{{ui-role-wrap}}",
554                if !roles.is_empty() && ui_auth == Some("session") {
555                    &role_wrap
556                } else {
557                    ""
558                },
559            )
560            .replace("{{required-roles}}", &roles.join(","));
561        format!("{}\n", rendered.trim_end())
562    };
563    let mut templates = vec![
564        (
565            base.join("mod.rs"),
566            include_str!("../templates/resource/mod.rs.tpl"),
567        ),
568        (
569            base.join("commands.rs"),
570            include_str!("../templates/resource/commands.rs.tpl"),
571        ),
572        (
573            base.join("events.rs"),
574            include_str!("../templates/resource/events.rs.tpl"),
575        ),
576        (
577            base.join("aggregate.rs"),
578            include_str!("../templates/resource/aggregate.rs.tpl"),
579        ),
580        (
581            base.join("projector.rs"),
582            include_str!("../templates/resource/projector.rs.tpl"),
583        ),
584        (
585            migration.join("up.sql"),
586            include_str!("../templates/resource/up.sql.tpl"),
587        ),
588        (
589            migration.join("down.sql"),
590            include_str!("../templates/resource/down.sql.tpl"),
591        ),
592    ];
593    if api {
594        templates.push((
595            base.join("api.rs"),
596            include_str!("../templates/resource/api.rs.tpl"),
597        ));
598    }
599    if ui {
600        templates.push((
601            base.join("ui.rs"),
602            include_str!("../templates/resource/ui.rs.tpl"),
603        ));
604        templates.push((
605            PathBuf::from(format!(
606                "resources/views/resources/{}/collection.html",
607                names.module
608            )),
609            include_str!("../templates/resource/collection.html.tpl"),
610        ));
611        templates.push((
612            PathBuf::from(format!(
613                "resources/views/resources/{}/detail.html",
614                names.module
615            )),
616            include_str!("../templates/resource/detail.html.tpl"),
617        ));
618        templates.push((
619            PathBuf::from(format!(
620                "resources/views/resources/{}/form.html",
621                names.module
622            )),
623            include_str!("../templates/resource/form.html.tpl"),
624        ));
625    }
626    templates
627        .into_iter()
628        .map(|(relative, template)| GeneratedFile {
629            relative,
630            contents: replacements(template),
631        })
632        .collect()
633}

651// UniFFI's Kotlin oracle uses UpperCamelCase for exported record names.
652// Resource names are ASCII; retain the preceding letter's case across digits
653// and split acronym runs before the capital that begins a lowercase word.
654fn kotlin_record_name(type_name: &str) -> String {
655    let input = format!("{type_name}Record");
656    let bytes = input.as_bytes();
657    let mut starts = vec![0];
658    let mut previous_upper = None;
659    for index in 0..bytes.len().saturating_sub(1) {
660        let current = bytes[index];
661        let next = bytes[index + 1];
662        let current_upper = if current.is_ascii_uppercase() {
663            Some(true)
664        } else if current.is_ascii_lowercase() {
665            Some(false)
666        } else {
667            previous_upper
668        };
669        if current_upper == Some(false) && next.is_ascii_uppercase() {
670            starts.push(index + 1);
671            previous_upper = None;
672        } else if previous_upper == Some(true)
673            && current.is_ascii_uppercase()
674            && next.is_ascii_lowercase()
675        {
676            starts.push(index);
677            previous_upper = None;
678        } else {
679            previous_upper = current_upper;
680        }
681    }
682    starts.push(bytes.len());
683    starts
684        .windows(2)
685        .map(|range| {
686            let word = &input[range[0]..range[1]];
687            word[..1].to_ascii_uppercase() + &word[1..].to_ascii_lowercase()
688        })
689        .collect()
690}

692fn android_render(
693    template: &str,
694    names: &Names,
695    manifest: &crate::android::AndroidManifest,
696) -> String {
697    let lower_type = names.type_name[..1].to_ascii_lowercase() + &names.type_name[1..];
698    let plural_method = names
699        .view
700        .split('_')
701        .enumerate()
702        .map(|(i, w)| {
703            if i == 0 {
704                w.to_owned()
705            } else {
706                w[..1].to_ascii_uppercase() + &w[1..]
707            }
708        })
709        .collect::<String>();
710    template
711        .replace("{{RecordType}}", &kotlin_record_name(&names.type_name))
712        .replace("{{lowerType}}", &lower_type)
713        .replace(
714            "{{pluralMethod}}",
715            &(plural_method[..1].to_ascii_uppercase() + &plural_method[1..]),
716        )
717        .replace("{{Type}}", &names.type_name)
718        .replace("{{module}}", &names.module)
719        .replace("{{view}}", &names.view)
720        .replace("{{CONSTANT}}", &names.constant)
721        .replace("{{application-id}}", &manifest.application_id)
722        .replace("{{crate-name}}", &manifest.crate_name)
723}

800fn android_blocks(
801    names: &Names,
802    migration: &str,
803    manifest: &crate::android::AndroidManifest,
804) -> AndroidBlocks {
805    let t = &names.type_name;
806    let m = &names.module;
807    let v = &names.view;
808    let app = &manifest.application_id;
809    vec![
810        ("rust/src/domain.rs".into(), "domain-modules", format!("pub mod {m};")),
811        ("rust/src/ffi.rs".into(), "ffi-imports", format!("use crate::domain::{m}::{{aggregate::{t}Aggregate, ffi::{{{t}Record, {t}ResourceError}}}};")),
812        ("rust/src/ffi.rs".into(), "ffi-fields", format!("    pub(crate) {m}: Arc<arc_ffi::FfiCore<{t}Aggregate>>,")),
813        ("rust/src/ffi.rs".into(), "ffi-open-initializers", format!("        let {m} = crate::domain::{m}::ffi::open(arc_ffi::CoreRegistry::process()?, path.clone()).await?;")),
814        ("rust/src/ffi.rs".into(), "ffi-open-fields", format!("            {m},")),
815        ("rust/src/ffi.rs".into(), "ffi-methods", format!(r#"    pub async fn list_{v}(&self) -> Result<Vec<{t}Record>, {t}ResourceError> {{
816        crate::domain::{m}::ffi::list(&self.{m}).await
817    }}
818    pub async fn get_{m}(&self, id: String) -> Result<{t}Record, {t}ResourceError> {{
819        crate::domain::{m}::ffi::get(&self.{m}, id).await
820    }}
821    pub async fn create_{m}(&self, name: String) -> Result<String, {t}ResourceError> {{
822        crate::domain::{m}::ffi::create(&self.{m}, name).await
823    }}
824    pub async fn rename_{m}(&self, id: String, name: String) -> Result<(), {t}ResourceError> {{
825        crate::domain::{m}::ffi::rename(&self.{m}, id, name).await
826    }}
827    pub async fn delete_{m}(&self, id: String) -> Result<(), {t}ResourceError> {{
828        crate::domain::{m}::ffi::delete(&self.{m}, id).await
829    }}"#)),
830        ("rust/src/schema.rs".into(), "schema-migrations", format!("    (\"{migration}_{v}_view\", include_str!(\"../migrations/{migration}_{v}_view/up.sql\")),")),
831        (format!("app/src/main/java/{}/ResourceCatalog.kt", app.replace('.', "/")), "catalog-imports", format!("import {app}.{m}.{t}Screen")),
832        (format!("app/src/main/java/{}/ResourceCatalog.kt", app.replace('.', "/")), "catalog-entries", format!("    ResourceEntry(\"{m}\", \"{t}\") {{ back -> {t}Screen(onBack = back) }},")),
833    ].into_iter().map(|(path, marker, body)| {
834        // Whole-line delimiters permit checking exact ownership without touching
835        // handwritten text or other resource blocks in a shared file.
836        let block = format!("// arc:begin {m} {marker}\n{body}\n// arc:end {m} {marker}\n");
837        (path, marker, block)
838    }).collect()
839}

858fn create_android_resource(
859    resource: &NewResource,
860    names: &Names,
861    mut manifest: crate::android::AndroidManifest,
862) -> Result<CreatedResource, String> {
863    if resource.api
864        || resource.ui
865        || resource.api_auth.is_some()
866        || resource.ui_auth.is_some()
867        || !resource.roles.is_empty()
868    {
869        return Err(
870            "--ffi cannot be combined with --api, --ui, --api-auth, --ui-auth, or --roles".into(),
871        );
872    }
873    // Kotlin and UniFFI names occupy namespaces beyond the Rust module namespace.
874    if matches!(
875        names.module.as_str(),
876        "native_core"
877            | "resource_catalog"
878            | "main_activity"
879            | "when"
880            | "object"
881            | "in"
882            | "is"
883            | "fun"
884            | "val"
885            | "var"
886            | "by"
887            | "constructor"
888            | "init"
889            | "companion"
890            | "interface"
891            | "package"
892            | "typeof"
893            | "throw"
894            | "try"
895            | "catch"
896            | "finally"
897            | "import"
898            | "class"
899            | "this"
900            | "null"
901            | "typealias"
902    ) || matches!(
903        names.type_name.as_str(),
904        "String" | "Any" | "Unit" | "Nothing" | "NativeCore"
905    ) {
906        return Err(android_conflict([format!(
907            "reserved native resource name {}",
908            names.type_name
909        )]));
910    }
911    let root = &resource.root;
912    android_safe_path(root, Path::new(crate::android::MANIFEST_PATH))?;
913    let manifest_conflict = || android_conflict([crate::android::MANIFEST_PATH.into()]);
914    let legacy = manifest.format_version == 1;
915    let mut records = Vec::<AndroidRegistration>::new();
916    let mut occupied = std::collections::BTreeSet::new();
917    let mut conflicts = Vec::new();
918    // Only the manifest structure is validated for every registration. Other resources'
919    // files are never written while adding this one, so their customizations are kept.
920    for value in &manifest.resources {
921        let mut registration: AndroidRegistration =
922            serde_json::from_value(value.clone()).map_err(|_| manifest_conflict())?;
923        let registered_names = parse_name(&registration.name).map_err(|_| manifest_conflict())?;
924        if registered_names.module != registration.module
925            || registered_names.type_name != registration.type_name
926            || registration.migration.len() != 14
927            || !registration.migration.bytes().all(|b| b.is_ascii_digit())
928            || !occupied.insert(registration.module.clone())
929            || !registration.files.keys().all(|path| android_relative(path))
930            || !registration.blocks.keys().all(|key| {
931                key.split_once('#')
932                    .is_some_and(|(path, marker)| android_relative(path) && !marker.is_empty())
933            })
934        {
935            return Err(manifest_conflict());
936        }
937        if legacy {
938            // Format 1 recorded whole generated bodies; their hashes are the same identity.
939            for identity in registration
940                .files
941                .values_mut()
942                .chain(registration.blocks.values_mut())
943            {
944                *identity = crate::android::content_hash(identity.as_bytes());
945            }
946        } else if !registration
947            .files
948            .values()
949            .chain(registration.blocks.values())
950            .all(|identity| crate::android::is_content_hash(identity))
951        {
952            return Err(manifest_conflict());
953        }
954        records.push(registration);
955    }
956    // Validate the shared seam even for an exact repeat.
957    let blocks = android_blocks(names, "00000000000001", &manifest);
958    let mut shared = std::collections::BTreeMap::<String, String>::new();
959    for (path, marker, _) in &blocks {
960        android_safe_path(root, Path::new(path))?;
961        let source = read_file(&root.join(path))?;
962        check_android_marker(&source, path, marker)?;
963        shared.insert(path.clone(), source);
964    }
965    if let Some(record) = records.iter().find(|r| {
966        r.module == names.module
967            || r.type_name.eq_ignore_ascii_case(&names.type_name)
968            || parse_name(&r.name).is_ok_and(|n| n.view == names.view)
969    }) {
970        if record.name == resource.name && record.module == names.module {
971            // An exact repeat is a no-op only while everything it owns is unchanged.
972            for (path, identity) in &record.files {
973                android_safe_path(root, Path::new(path))?;
974                let target = root.join(path);
975                if !fs::symlink_metadata(&target).is_ok_and(|m| m.is_file())
976                    || fs::read(&target)
977                        .map(|bytes| crate::android::content_hash(&bytes))
978                        .ok()
979                        .as_ref()
980                        != Some(identity)
981                {
982                    conflicts.push(path.clone());
983                }
984            }
985            for (key, identity) in &record.blocks {
986                let (path, marker) = key.split_once('#').unwrap();
987                android_safe_path(root, Path::new(path))?;
988                let source = fs::read_to_string(root.join(path)).unwrap_or_default();
989                if owned_block(&source, &record.module, marker)
990                    .map(|block| crate::android::content_hash(block.as_bytes()))
991                    .as_ref()
992                    != Some(identity)
993                {
994                    conflicts.push(format!("{path} ({} {marker} block)", record.module));
995                }
996            }
997            if !conflicts.is_empty() {
998                return Err(android_conflict(conflicts));
999            }
1000            return Ok(CreatedResource {
1001                type_name: names.type_name.clone(),
1002                files: vec![],
1003                api_path: None,
1004            });
1005        }
1006        return Err(android_conflict([format!(
1007            "normalized resource {}",
1008            names.module
1009        )]));
1010    }
1011    let migration = next_migration_version(&root.join("rust/migrations"))?;
1012    let files = android_files(names, &migration, &manifest);
1013    let blocks = android_blocks(names, &migration, &manifest);
1014    let domain_dir = root.join(format!("rust/src/domain/{}", names.module));
1015    if fs::symlink_metadata(&domain_dir).is_ok() {
1016        conflicts.push(domain_dir.display().to_string());
1017    }
1018    let kotlin_dir = root.join(format!(
1019        "app/src/main/java/{}/{}",
1020        manifest.application_id.replace('.', "/"),
1021        names.module
1022    ));
1023    if fs::symlink_metadata(&kotlin_dir).is_ok() {
1024        conflicts.push(kotlin_dir.display().to_string());
1025    }
1026    for file in &files {
1027        if fs::symlink_metadata(root.join(&file.relative)).is_ok() {
1028            conflicts.push(file.relative.display().to_string());
1029        }
1030    }
1031    for (path, marker, _) in &blocks {
1032        let source = shared.get(path).unwrap();
1033        if source.contains(&format!("// arc:begin {} ", names.module))
1034            || (marker == &"domain-modules"
1035                && source
1036                    .lines()
1037                    .any(|l| l.trim() == format!("pub mod {};", names.module)))
1038        {
1039            conflicts.push(path.clone());
1040        }
1041    }
1042    if !conflicts.is_empty() {
1043        return Err(android_conflict(conflicts));
1044    }
1045    let mut identities = std::collections::BTreeMap::new();
1046    for (path, marker, block) in blocks {
1047        let source = shared.get_mut(&path).unwrap();
1048        let token = format!("// arc:{marker}");
1049        // Insert at the start of the marker line, preserving its indentation.
1050        let at = source
1051            .split_inclusive('\n')
1052            .scan(0, |offset, line| {
1053                let at = *offset;
1054                *offset += line.len();
1055                Some((at, line))
1056            })
1057            .find(|(_, line)| line.trim() == token)
1058            .unwrap()
1059            .0;
1060        source.insert_str(at, &block);
1061        identities.insert(
1062            format!("{path}#{marker}"),
1063            crate::android::content_hash(block.as_bytes()),
1064        );
1065    }
1066    records.push(AndroidRegistration {
1067        name: resource.name.clone(),
1068        module: names.module.clone(),
1069        type_name: names.type_name.clone(),
1070        migration,
1071        files: files
1072            .iter()
1073            .map(|f| {
1074                (
1075                    f.relative.to_string_lossy().into_owned(),
1076                    crate::android::content_hash(f.contents.as_bytes()),
1077                )
1078            })
1079            .collect(),
1080        blocks: identities,
1081    });
1082    // Format 1 manifests are upgraded in the same atomic write; skeleton files stay
1083    // without identities (unknown ownership) because none were recorded for them.
1084    manifest.format_version = crate::android::MANIFEST_FORMAT;
1085    manifest.resources = records
1086        .iter()
1087        .map(serde_json::to_value)
1088        .collect::<Result<_, _>>()
1089        .map_err(|e| e.to_string())?;
1090    let created = files.iter().map(|f| f.relative.clone()).collect();
1091    let mut writes = files;
1092    writes.extend(
1093        shared
1094            .into_iter()
1095            .map(|(relative, contents)| GeneratedFile {
1096                relative: relative.into(),
1097                contents,
1098            }),
1099    );
1100    writes.push(GeneratedFile {
1101        relative: crate::android::MANIFEST_PATH.into(),
1102        contents: serde_json::to_string_pretty(&manifest).map_err(|e| e.to_string())? + "\n",
1103    });
1104    atomic_android_write(root, &writes)?;
1105    Ok(CreatedResource {
1106        type_name: names.type_name.clone(),
1107        files: created,
1108        api_path: None,
1109    })
1110}

1119/// The whole-line `arc:begin`..`arc:end` block a resource owns in a shared file.
1120fn owned_block<'a>(source: &'a str, module: &str, marker: &str) -> Option<&'a str> {
1121    let begin = format!("// arc:begin {module} {marker}\n");
1122    let end = format!("// arc:end {module} {marker}\n");
1123    if source.matches(&begin).count() != 1 || source.matches(&end).count() != 1 {
1124        return None;
1125    }
1126    let start = source.find(&begin)?;
1127    let stop = source.find(&end)? + end.len();
1128    (start < stop && (start == 0 || source[..start].ends_with('\n'))).then(|| &source[start..stop])
1129}

1131fn android_safe_path(root: &Path, relative: &Path) -> Result<(), String> {
1132    let path = root.join(relative);
1133    let mut parent = Some(path.as_path());
1134    while let Some(p) = parent {
1135        if p == root {
1136            break;
1137        }
1138        if fs::symlink_metadata(p).is_ok_and(|m| m.file_type().is_symlink()) {
1139            return Err(android_conflict([p.display().to_string()]));
1140        }
1141        parent = p.parent();
1142    }
1143    Ok(())
1144}

1146/// Stage bytes before changing the tree; rename each file and restore originals on
1147/// any failure. Only newly created directories are removed during rollback.
1148fn atomic_android_write(root: &Path, writes: &[GeneratedFile]) -> Result<(), String> {
1149    use std::sync::atomic::{AtomicU64, Ordering};
1150    static NONCE: AtomicU64 = AtomicU64::new(0);
1151    let mut backups = Vec::new();
1152    for file in writes {
1153        let path = root.join(&file.relative);
1154        // No symlink traversal, including shared parent directories.
1155        let mut parent = Some(path.as_path());
1156        while let Some(p) = parent {
1157            if p == root {
1158                break;
1159            }
1160            if fs::symlink_metadata(p).is_ok_and(|m| m.file_type().is_symlink()) {
1161                return Err(android_conflict([p.display().to_string()]));
1162            }
1163            parent = p.parent();
1164        }
1165        let original = match fs::read(&path) {
1166            Ok(bytes) => Some(bytes),
1167            Err(e) if e.kind() == std::io::ErrorKind::NotFound => None,
1168            Err(e) => return Err(format!("could not stage {}: {e}", path.display())),
1169        };
1170        backups.push((path, original));
1171    }
1172    let stage = root.join(format!(
1173        ".arc-resource-stage-{}-{}",
1174        std::process::id(),
1175        NONCE.fetch_add(1, Ordering::Relaxed)
1176    ));
1177    fs::create_dir(&stage).map_err(|e| e.to_string())?;
1178    let mut directories = Vec::new();
1179    let mut installed = 0;
1180    let result: Result<(), String> = (|| {
1181        for (index, file) in writes.iter().enumerate() {
1182            fs::write(stage.join(index.to_string()), &file.contents).map_err(|e| e.to_string())?;
1183        }
1184        for (index, (path, _)) in backups.iter().enumerate() {
1185            let mut missing = Vec::new();
1186            let mut parent = path.parent().unwrap();
1187            while !parent.exists() {
1188                missing.push(parent.to_path_buf());
1189                parent = parent.parent().unwrap();
1190            }
1191            for directory in missing.into_iter().rev() {
1192                fs::create_dir(&directory).map_err(|e| e.to_string())?;
1193                directories.push(directory);
1194            }
1195            fs::rename(stage.join(index.to_string()), path)
1196                .map_err(|e| format!("could not install {}: {e}", path.display()))?;
1197            installed += 1;
1198        }
1199        Ok(())
1200    })();
1201    if let Err(error) = result {
1202        let mut rollback_errors = Vec::new();
1203        for (path, original) in backups[..installed].iter().rev() {
1204            let restored = match original {
1205                Some(bytes) => fs::write(path, bytes),
1206                None => fs::remove_file(path),
1207            };
1208            if let Err(e) = restored {
1209                rollback_errors.push(format!("{}: {e}", path.display()));
1210            }
1211        }
1212        for directory in directories.iter().rev() {
1213            if let Err(e) = fs::remove_dir(directory) {
1214                rollback_errors.push(format!("{}: {e}", directory.display()));
1215            }
1216        }
1217        let _ = fs::remove_dir_all(&stage);
1218        return Err(format!(
1219            "{error}; rollback errors: {}",
1220            rollback_errors.join(", ")
1221        ));
1222    }
1223    fs::remove_dir_all(&stage).map_err(|e| format!("could not remove staging directory: {e}"))
1224}

1468    fn tree(root: &Path) -> std::collections::BTreeMap<PathBuf, Vec<u8>> {
1469        fn visit(
1470            base: &Path,
1471            path: &Path,
1472            result: &mut std::collections::BTreeMap<PathBuf, Vec<u8>>,
1473        ) {
1474            for entry in fs::read_dir(path).unwrap() {
1475                let path = entry.unwrap().path();
1476                if path.is_dir() {
1477                    result.insert(path.strip_prefix(base).unwrap().to_path_buf(), vec![]);
1478                    visit(base, &path, result);
1479                } else {
1480                    result.insert(
1481                        path.strip_prefix(base).unwrap().to_path_buf(),
1482                        fs::read(&path).unwrap(),
1483                    );
1484                }
1485            }
1486        }
1487        let mut result = std::collections::BTreeMap::new();
1488        visit(root, root, &mut result);
1489        result
1490    }
