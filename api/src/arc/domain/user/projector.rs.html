1//! `UserProjector` — builds the `users_view` read model from `User` events.
2//!
3//! Pure event handler: receives an [`Event`] + a [`ReadModelStore`] and
4//! emits a single version-gated upsert, including a retained deletion tombstone.
5//! Engine delivery reconciles complete authoritative state, so partial events
6//! remain correct under reordered delivery and concurrent writers. Holds no
7//! mutable state. Idempotency comes from the store's version gate, not from
8//! the projector — replaying a known event sequence twice yields the same
9//! `users_view` because each upsert with a given `version` is rejected the
10//! second time. `_arc_deleted: true` rows are hidden by read-model stores.
11//!
12//! Wired into [`InProcessEventBus`](arc_core::event_bus::InProcessEventBus)
13//! at startup via a thin [`EventHandler`](arc_core::event_bus::EventHandler) adapter that calls
14//! [`ProjectionEngine::process`](arc_core::projection::ProjectionEngine::process).
15

43#[async_trait]
44impl Projector for UserProjector {
45    fn name(&self) -> &str {
46        "UserProjector"
47    }
48
49    fn handles(&self) -> Vec<String> {
50        vec![
51            "UserRegistered".to_string(),
52            "ProfileUpdated".to_string(),
53            "EmailChanged".to_string(),
54            "PasswordChanged".to_string(),
55            "UserDeleted".to_string(),
56        ]
57    }
58
59    async fn apply_authoritative(
60        &self,
61        event: &Event,
62        store: &dyn ReadModelStore,
63        event_store: &dyn EventStore,
64    ) -> ProjectionResult<()> {
65        // Fold the complete current stream, not the last projected row. A newer
66        // delivery can therefore never discard an older event's different field.
67        // Concurrent folds produce complete states; the store's version gate
68        // rejects the older result, including results racing with deletion.
69        let events = event_store
70            .load_stream("User", &event.aggregate_id)
71            .await
72            .map_err(|error| project_err(self, event, error.to_string()))?;
73        let mut row = None;
74        for stored in events {
75            match stored.event_type.as_str() {
76                "UserRegistered" => {
77                    row = Some(json!({
78                        "id": stored.aggregate_id,
79                        "name": payload_str(&stored.payload, "name")?,
80                        "email": payload_str(&stored.payload, "email")?,
81                        "password_hash": payload_str(&stored.payload, "password_hash")?,
82                        "version": stored.sequence,
83                    }))
84                }
85                "UserDeleted" => {
86                    row = Some(json!({
87                        "id": stored.aggregate_id, "version": stored.sequence, "_arc_deleted": true,
88                    }))
89                }
90                "ProfileUpdated" | "EmailChanged" | "PasswordChanged" => {
91                    let current = row.as_mut().ok_or_else(|| {
92                        project_err(self, event, "user stream is missing registration")
93                    })?;
94                    if current.get("_arc_deleted").and_then(Value::as_bool) == Some(true) {
95                        continue;
96                    }
97                    let field = match stored.event_type.as_str() {
98                        "ProfileUpdated" => "name",
99                        "EmailChanged" => "email",
100                        _ => "password_hash",
101                    };
102                    current[field] = json!(payload_str(&stored.payload, field)?);
103                    current["version"] = json!(stored.sequence);
104                }
105                _ => {}
106            }
107        }
108        let row =
109            row.ok_or_else(|| project_err(self, event, "authoritative user stream is empty"))?;
110        store
111            .upsert(Upsert::new(USERS_VIEW, &event.aggregate_id, row))
112            .await
113            .map_err(|error| project_err(self, event, error.to_string()))
114    }

116    async fn apply(&self, event: &Event, store: &dyn ReadModelStore) -> ProjectionResult<()> {
117        let id = &event.aggregate_id;
118
119        // For partial-update events we need the prior row so we can carry
120        // forward fields the event doesn't touch. UserRegistered seeds the
121        // row outright, so the lookup is skipped.
122        match event.event_type.as_str() {
123            "UserRegistered" => {
124                let row = json!({
125                    "id": id,
126                    "name": payload_str(&event.payload, "name")?,
127                    "email": payload_str(&event.payload, "email")?,
128                    "password_hash": payload_str(&event.payload, "password_hash")?,
129                    "version": event.sequence,
130                });
131                store
132                    .upsert(Upsert::new(USERS_VIEW, id, row))
133                    .await
134                    .map_err(|e| project_err(self, event, e.to_string()))?;
135            }
136
137            "UserDeleted" => {
138                store
139                    .upsert(Upsert::new(
140                        USERS_VIEW,
141                        id,
142                        json!({
143                            "id": id, "version": event.sequence, "_arc_deleted": true,
144                        }),
145                    ))
146                    .await
147                    .map_err(|e| project_err(self, event, e.to_string()))?;
148            }
149
150            "ProfileUpdated" | "EmailChanged" | "PasswordChanged" => {
151                let existing = store
152                    .get(USERS_VIEW, id)
153                    .await
154                    .map_err(|e| project_err(self, event, e.to_string()))?;
155
156                // No prior row to mutate. Either the row was already deleted
157                // or this projector is being driven without seeing the
158                // upstream UserRegistered event. Skip rather than guess.
159                let Some(mut row) = existing else {
160                    tracing::warn!(
161                        event_type = event.event_type,
162                        aggregate_id = id,
163                        "UserProjector: skipping update for unknown user"
164                    );
165                    return Ok(());
166                };
167
168                match event.event_type.as_str() {
169                    "ProfileUpdated" => {
170                        row["name"] = json!(payload_str(&event.payload, "name")?);
171                    }
172                    "EmailChanged" => {
173                        row["email"] = json!(payload_str(&event.payload, "email")?);
174                    }
175                    "PasswordChanged" => {
176                        row["password_hash"] = json!(payload_str(&event.payload, "password_hash")?);
177                    }
178                    _ => unreachable!(),
179                }
180                row["version"] = json!(event.sequence);
181
182                store
183                    .upsert(Upsert::new(USERS_VIEW, id, row))
184                    .await
185                    .map_err(|e| project_err(self, event, e.to_string()))?;
186            }
187
188            _ => {}
189        }
190
191        Ok(())
192    }
193}

195fn payload_str<'a>(payload: &'a Value, field: &str) -> ProjectionResult<&'a str> {
196    payload.get(field).and_then(Value::as_str).ok_or_else(|| {
197        ProjectionError::other(format!("event payload missing string field '{field}'"))
198    })
199}

201fn project_err(p: &UserProjector, event: &Event, message: impl Into<String>) -> ProjectionError {
202    ProjectionError::handle_failed(
203        p.name(),
204        &event.event_type,
205        event.event_id.to_string(),
206        message,
207    )

227    /// Suspend the older fold after it has built its complete v2 row but before
228    /// the version-gated write. The newer delivery must commit while it waits.
229    struct PausedUpsertStore {
230        inner: InMemoryReadModelStore,
231        pause_once: std::sync::atomic::AtomicBool,
232        entered: tokio::sync::Notify,
233        release: tokio::sync::Notify,
234    }
235
236    #[async_trait]
237    impl ReadModelStore for PausedUpsertStore {
238        async fn upsert(&self, op: Upsert) -> arc_core::read_model_store::ReadModelResult<()> {
239            if op.row["version"] == 2
240                && self
241                    .pause_once
242                    .swap(false, std::sync::atomic::Ordering::SeqCst)
243            {
244                self.entered.notify_one();
245                self.release.notified().await;
246            }
247            self.inner.upsert(op).await
248        }

296    async fn assert_paused_snapshot_cannot_overwrite_newer_delivery(newer: Event) {
297        use arc_core::event_store::{EventStore, InMemoryEventStore, VersionCheck};
298        use arc_core::projection::ProjectionEngine;
299        use std::sync::Arc;
300        use std::time::Duration;
301        let registration = ev(
302            "u1",
303            1,
304            "UserRegistered",
305            json!({
306                "id":"u1", "name":"Old", "email":"old@example.test", "password_hash":"old-hash"
307            }),
308        );
309        let profile = ev("u1", 2, "ProfileUpdated", json!({"name":"New"}));
310        let event_store = InMemoryEventStore::new();
311        event_store
312            .append_to(
313                "User",
314                "u1",
315                VersionCheck::New,
316                vec![registration, profile.clone()],
317            )
318            .await
319            .unwrap();
320        let store = Arc::new(PausedUpsertStore {
321            inner: InMemoryReadModelStore::new(),
322            pause_once: std::sync::atomic::AtomicBool::new(true),
323            entered: tokio::sync::Notify::new(),
324            release: tokio::sync::Notify::new(),
325        });
326        let mut engine = ProjectionEngine::new(Box::new(event_store.clone()));
327        engine.register_projector(Box::new(UserProjector), store.clone(), USERS_VIEW);
328        let engine = Arc::new(engine);
329        let older_engine = engine.clone();
330        let older = tokio::spawn(async move { older_engine.process(&profile).await });
331        tokio::time::timeout(Duration::from_secs(5), store.entered.notified())
332            .await
333            .expect("older authoritative fold reached its v2 upsert barrier");
334
335        // Only now append v3: the suspended fold necessarily contains stale v2
336        // state, rather than both tasks reconstructing the same complete history.
337        event_store
338            .append_to("User", "u1", VersionCheck::Expected(2), vec![newer.clone()])
339            .await
340            .unwrap();
341        engine
342            .process_external_delivery(&newer.with_stored_timestamp_precision())
343            .await
344            .unwrap();
345        let latest = store.inner.get_rows(USERS_VIEW);
346        assert_eq!(latest.len(), 1);
347        assert_eq!(latest[0]["version"], 3);
348        let expected = latest[0].clone();
349        if expected["_arc_deleted"] != true {
350            assert_eq!(expected["name"], "New");
351            assert_eq!(expected["email"], "new@example.test");
352            assert_eq!(expected["password_hash"], "old-hash");
353        }
354
355        store.release.notify_one();
356        tokio::time::timeout(Duration::from_secs(5), older)
357            .await
358            .expect("older projection finishes after releasing the barrier")
359            .expect("older projection task did not panic")
360            .expect("stale projection is safely ignored");
361        assert_eq!(store.inner.get_rows(USERS_VIEW), vec![expected.clone()]);
362        if expected["_arc_deleted"] == true {
363            assert!(store.get(USERS_VIEW, "u1").await.unwrap().is_none());
364            assert!(store.list(USERS_VIEW).await.unwrap().is_empty());
365        }
366    }

654        drop(conn);
655
656        let event_store = SqliteEventStore::new("file::memory:?cache=shared")
657            .await
658            .expect("event store");
659        let rm_store: Arc<dyn ReadModelStore> = Arc::new(
660            SqliteReadModelStore::new("file::memory:?cache=shared")
661                .await
662                .expect("rm store"),
663        );
664
665        // u1: register → update profile → change email → delete.
666        // u2: register only — survives.
667        event_store
668            .append(
669                "u1",
670                VersionCheck::New,
671                vec![
672                    ev(
673                        "u1",
674                        1,
675                        "UserRegistered",
676                        json!({"id":"u1","name":"Alice","email":"a@b.c","password_hash":"$argon$x"}),
677                    ),
678                    ev("u1", 2, "ProfileUpdated", json!({"name":"Alice2"})),
679                    ev("u1", 3, "EmailChanged", json!({"email":"a2@b.c"})),
680                    ev("u1", 4, "UserDeleted", json!({})),
681                ],
682            )
683            .await
684            .unwrap();
685        event_store
686            .append(
687                "u2",
688                VersionCheck::New,
689                vec![ev(
690                    "u2",
691                    1,
692                    "UserRegistered",
693                    json!({"id":"u2","name":"Bob","email":"b@b.c","password_hash":"$argon$y"}),
694                )],
695            )
696            .await
697            .unwrap();
698
699        // Build engine, register projector, rebuild from zero.
700        let mut engine = ProjectionEngine::new(Box::new(event_store));
701        engine.register_projector(Box::new(UserProjector::new()), rm_store.clone(), USERS_VIEW);
702        engine.rebuild_all().await.expect("rebuild_all");
703
704        // u1 should be absent (deleted), u2 present with correct fields.
705        assert!(rm_store.get(USERS_VIEW, "u1").await.unwrap().is_none());
706        let u2 = rm_store.get(USERS_VIEW, "u2").await.unwrap().unwrap();
707        assert_eq!(u2["name"], "Bob");
708        assert_eq!(u2["email"], "b@b.c");
709        assert_eq!(u2["version"], 1);
710
711        // Email lookup hits the JSON-extract index.
712        let by_email = rm_store
713            .find_by(USERS_VIEW, "email", &json!("b@b.c"))
714            .await
715            .unwrap();
716        assert_eq!(by_email.len(), 1);
717        assert_eq!(by_email[0]["id"], "u2");
718
719        // Idempotency: a second rebuild produces the same final state.
720        engine.rebuild_all().await.expect("rebuild_all again");
721        assert!(rm_store.get(USERS_VIEW, "u1").await.unwrap().is_none());
722        assert_eq!(
723            rm_store.get(USERS_VIEW, "u2").await.unwrap().unwrap()["version"],
724            1
725        );
726    }
