1//! # Access Logger
2//!
3//! Generic read-side audit logger. Where `EventStore` records *writes* with
4//! [`AuditMetadata`](crate::audit::AuditMetadata), `AccessLogger` records
5//! *reads* of sensitive data. These primitives do not establish compliance or
6//! provide durable storage by themselves.
7//!
8//! Reads do not go through the event store. Controllers must explicitly
9//! invoke [`AccessLogger::log_access`] before returning data classified as
10//! anything beyond [`Sensitivity::Public`].
11//!
12//! ## Why generic, not PHI-specific
13//!
14//! The mechanism is the same regardless of regime: log who looked at what,
15//! when, and why. [`Sensitivity`] classifies data so a downstream sink can
16//! apply deployment-specific routing, separate categories, drop
17//! [`Sensitivity::Public`] reads, and so on.
18//!
19//! ## Lifecycle
20//!
21//! 1. A read controller resolves the actor (typically the JWT-bound aggregate UUID).
22//! 2. Builds an [`AccessedResource`] describing what's about to be returned.
23//! 3. Calls `logger.log_access(actor, resource, purpose).await`.
24//! 4. Returns the data to the client.
25//!
26//! Default implementations in tests and non-regulated apps use
27//! [`NoOpAccessLogger`] which validates inputs but discards them. Real
28//! deployments can select the dedicated `arc-es-sqlite::SqliteAccessLogger`.
29//! Coverage remains opt-in at each disclosure boundary.

37/// Errors emitted by [`AccessLogger`] implementations.
38#[derive(Debug, Error)]
39pub enum AccessLogError {
40    #[error("access log validation failed: {0}")]
41    Validation(String),
42    #[error("access log sink failed: {0}")]
43    Sink(String),

46/// Identity of the entity performing the read.
47///
48/// `aggregate_id` is the framework-level handle (UUID, `"system"`,
49/// `"anonymous"`, or `"legacy-pre-hipaa"` — the same vocabulary as
50/// [`AuditMetadata::actor_id`](crate::audit::AuditMetadata::actor_id)).
51#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
52pub struct Identity {
53    pub actor_id: String,
54    pub session_id: Option<String>,
55    pub source_ip: Option<String>,
56    pub user_agent: Option<String>,
57}
58
59impl Identity {
60    pub fn new(actor_id: impl Into<String>) -> Self {
61        Self {
62            actor_id: actor_id.into(),
63            session_id: None,
64            source_ip: None,
65            user_agent: None,
66        }
67    }
68}

70/// What controllers should do when an [`AccessLogger`] sink fails.
71///
72/// Framework defaults: PHI and PCI refuse the response (HTTP 503); other
73/// classifications, including PII, warn and allow the response. These are
74/// implementation defaults, not a statement of regulatory requirements.
75/// See [`FailurePolicy::for_sensitivity`].
76#[derive(Debug, Clone, Copy, PartialEq, Eq)]
77pub enum FailurePolicy {
78    FailHard,
79    FailOpenWarn,
80}
81
82impl FailurePolicy {
83    /// Default per regime: PHI and PCI fail hard; everything else logs and continues.
84    pub fn for_sensitivity(s: Sensitivity) -> Self {
85        match s {
86            Sensitivity::Phi | Sensitivity::Pci => FailurePolicy::FailHard,
87            _ => FailurePolicy::FailOpenWarn,
88        }
89    }
90}
91
92/// Sensitivity tag used for failure policy and sink routing.
93///
94/// Audit sinks use this to decide retention, routing, and whether to record
95/// at all.
96#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
97#[serde(rename_all = "snake_case")]
98pub enum Sensitivity {
99    /// Protected Health Information (HIPAA §164).
100    Phi,
101    /// Payment card industry data (PCI-DSS).
102    Pci,
103    /// Personally identifiable information (GDPR/CCPA).
104    Pii,
105    /// Trade secrets / internal-only documents.
106    Confidential,
107    /// Non-public but unregulated business data.
108    Internal,
109    /// Anything intentionally public (logged for completeness; sinks may downsample).
110    Public,
111}
112
113impl Sensitivity {
114    /// True when reads should be auditable in regulated deployments. Sinks
115    /// MAY drop [`Sensitivity::Public`] events to control volume.
116    pub fn is_regulated(self) -> bool {
117        !matches!(self, Sensitivity::Public)
118    }
119}

121/// Reason the read happened. Maps to HIPAA "purpose of use" categories but is
122/// applicable to any regime.
123#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
124#[serde(rename_all = "snake_case")]
125pub enum PurposeOfUse {
126    Treatment,
127    Payment,
128    Operations,
129    Emergency,
130    UserInitiated,
131    AuditReview,
132    Other,
133}
134
135/// Description of the resource being read.
136///
137/// `kind` and `identifier` together name the row(s); `fields` enumerates the
138/// columns the response will expose; `sensitivity` triggers regime-specific
139/// routing in the sink.
140#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
141pub struct AccessedResource {
142    pub kind: String,
143    pub identifier: String,
144    pub fields: Vec<String>,
145    pub sensitivity: Sensitivity,
146}
147
148impl AccessedResource {
149    pub fn new(
150        kind: impl Into<String>,
151        identifier: impl Into<String>,
152        sensitivity: Sensitivity,
153    ) -> Self {
154        Self {
155            kind: kind.into(),
156            identifier: identifier.into(),
157            fields: Vec::new(),
158            sensitivity,
159        }
160    }
161
162    pub fn with_fields<I, S>(mut self, fields: I) -> Self
163    where
164        I: IntoIterator<Item = S>,
165        S: Into<String>,
166    {
167        self.fields = fields.into_iter().map(Into::into).collect();
168        self
169    }
170}

172/// Single record produced for every successful `log_access` call.
173///
174/// Sinks serialize this into their target medium; tests inspect it directly.
175#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
176pub struct AccessLogEntry {
177    pub access_id: Uuid,
178    pub actor: Identity,
179    pub resource: AccessedResource,
180    pub purpose: PurposeOfUse,
181    pub timestamp_utc_us: i64,
182    pub correlation_id: Option<Uuid>,
183}
184
185impl AccessLogEntry {
186    /// Construct an entry from caller inputs. Validates and stamps timestamp.
187    pub fn new(
188        actor: Identity,
189        resource: AccessedResource,
190        purpose: PurposeOfUse,
191        correlation_id: Option<Uuid>,
192    ) -> Result<Self, AccessLogError> {
193        if actor.actor_id.trim().is_empty() {
194            return Err(AccessLogError::Validation(
195                "actor_id must be non-empty (use 'anonymous' or 'system' explicitly)".into(),
196            ));
197        }
198        if resource.kind.trim().is_empty() {
199            return Err(AccessLogError::Validation(
200                "resource.kind must be non-empty".into(),
201            ));
202        }
203        if resource.identifier.trim().is_empty() {
204            return Err(AccessLogError::Validation(
205                "resource.identifier must be non-empty".into(),
206            ));
207        }
208        Ok(Self {
209            access_id: Uuid::new_v4(),
210            actor,
211            resource,
212            purpose,
213            timestamp_utc_us: now_us(),
214            correlation_id,
215        })
216    }
217}

219/// Sink for read-access audit events.
220///
221/// Implementations:
222/// - [`NoOpAccessLogger`] — validates and discards (default in tests, non-PHI apps)
223/// - `RecordingAccessLogger` — keeps entries in memory for assertions (test-utils)
224///
225/// - `arc-es-sqlite::SqliteAccessLogger` — dedicated durable SQLite journal
226#[async_trait]
227pub trait AccessLogger: Send + Sync {
228    /// Log a read. Implementations validate the input, record it (or not),
229    /// and return.
230    ///
231    /// Callers must apply [`FailurePolicy`] before releasing sensitive data.
232    /// The web `record_read` helper fails closed for PHI/PCI and warns while
233    /// continuing for other classifications. An `Ok(())` is only as durable
234    /// as the chosen implementation: the no-op implementation discards data.
235    /// Custom sinks must document their acknowledgement boundary, timeout,
236    /// cancellation and retry behavior. Error strings may contain private
237    /// implementation details; callers should log only a safe error category.
238    async fn log_access(
239        &self,
240        actor: Identity,
241        resource: AccessedResource,
242        purpose: PurposeOfUse,
243        correlation_id: Option<Uuid>,
244    ) -> Result<(), AccessLogError>;
245}
246
247/// Validates inputs and discards the entry. Default for test apps and any
248/// deployment that has not yet wired a real sink.
249#[derive(Debug, Default, Clone, Copy)]
250pub struct NoOpAccessLogger;
251
252#[async_trait]
253impl AccessLogger for NoOpAccessLogger {
254    async fn log_access(
255        &self,
256        actor: Identity,
257        resource: AccessedResource,
258        purpose: PurposeOfUse,
259        correlation_id: Option<Uuid>,
260    ) -> Result<(), AccessLogError> {
261        // Build the entry purely for validation side effects.
262        let _ = AccessLogEntry::new(actor, resource, purpose, correlation_id)?;
263        Ok(())
264    }
265}

272#[cfg(any(test, feature = "test-utils"))]
273mod recording {
274    use super::*;
275    use std::sync::Arc;
276    use tokio::sync::Mutex;
277
278    /// In-memory `AccessLogger` that records every successful entry.
279    /// Use `entries()` to assert in tests.
280    #[derive(Clone, Default)]
281    pub struct RecordingAccessLogger {
282        entries: Arc<Mutex<Vec<AccessLogEntry>>>,
283    }
284
285    impl RecordingAccessLogger {
286        pub fn new() -> Self {
287            Self::default()
288        }
289
290        pub async fn entries(&self) -> Vec<AccessLogEntry> {
291            self.entries.lock().await.clone()
292        }
293    }
294
295    #[async_trait]
296    impl AccessLogger for RecordingAccessLogger {
297        async fn log_access(
298            &self,
299            actor: Identity,
300            resource: AccessedResource,
301            purpose: PurposeOfUse,
302            correlation_id: Option<Uuid>,
303        ) -> Result<(), AccessLogError> {
304            let entry = AccessLogEntry::new(actor, resource, purpose, correlation_id)?;
305            self.entries.lock().await.push(entry);
306            Ok(())
307        }
308    }
309}
