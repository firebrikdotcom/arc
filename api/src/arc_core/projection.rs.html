1//! # Projection Module
2//!
3//! Three-trait architecture for building read models from event streams:
4//!
5//! - **[`Projector`]** — stateless event handler (the "machine"). Contains the pure
6//!   logic for transforming events into read model writes.
7//! - **[`Projection`]** — composed read model unit (the "output"). Ties a projector
8//!   to its storage backend.
9//! - **[`ReadModelStore`]** — persistence
10//!   layer for projections. Backend-agnostic storage (defined in `read_model_store` module).
11//!
12//! ## Design Principles
13//!
14//! - **Separation of concerns**: Handler logic (projector) is separate from storage
15//!   (read model store) and orchestration (projection engine)
16//! - **Stateless projectors**: Projectors take `&self`, not `&mut self`. All mutable
17//!   state lives in the `ReadModelStore` via interior mutability.
18//! - **Rebuildable**: Projections can be rebuilt from scratch by replaying events
19//! - **Idempotent**: Handling the same event multiple times should be safe
20//! - **Composable**: One projector per read model concern; swap backends freely
21//!
22//! ## Example
23//!
24//! ```rust,ignore
25//! use arc_core::projection::{Projector, Projection, ProjectionUnit, ProjectionEngine};
26//! use arc_core::read_model_store::{ReadModelStore, InMemoryReadModelStore};
27//! use arc_core::event::Event;
28//! use std::sync::Arc;
29//!
30//! struct UserListProjector;
31//!
32//! #[async_trait]
33//! impl Projector for UserListProjector {
34//!     fn name(&self) -> &str { "UserList" }
35//!
36//!     fn handles(&self) -> Vec<String> {
37//!         vec!["UserCreated".to_string(), "ProfileUpdated".to_string()]
38//!     }
39//!
40//!     async fn apply(&self, event: &Event, store: &dyn ReadModelStore) -> ProjectionResult<()> {
41//!         match event.event_type.as_str() {
42//!             "UserCreated" => {
43//!                 store.upsert(Upsert::new("users_view", &event.aggregate_id, event.payload.clone())).await
44//!                     .map_err(|e| ProjectionError::handle_failed("UserList", &event.event_type, &event.event_id.to_string(), e.to_string()))?;
45//!             }
46//!             _ => {}
47//!         }
48//!         Ok(())
49//!     }
50//! }
51//!
52//! // Compose: projector + store = projection
53//! let store = Arc::new(InMemoryReadModelStore::new());
54//! let projection = ProjectionUnit::new(Box::new(UserListProjector), store, "users_view");
55//!
56//! // Register with engine
57//! let mut engine = ProjectionEngine::new(event_store);
58//! engine.register(Box::new(projection));
59//! engine.process(&event).await?;
60//! ```

76/// Errors that can occur during projection operations.
77#[derive(Debug, Error)]
78pub enum ProjectionError {
79    /// Error handling an event
80    #[error(
81        "Projection '{name}' failed to handle event {event_type} (event_id: {event_id}): {message}"
82    )]
83    HandleFailed {
84        name: String,
85        event_type: String,
86        event_id: String,
87        message: String,
88    },
89
90    /// Error clearing projection state
91    #[error("Projection '{name}' failed to clear: {message}")]
92    ClearFailed { name: String, message: String },
93
94    /// Error rebuilding projection
95    #[error("Projection '{name}' failed to rebuild: {message}")]
96    RebuildFailed { name: String, message: String },
97
98    /// Event store error during rebuild
99    #[error("Failed to load events for rebuild: {0}")]
100    EventStoreError(String),
101
102    /// Read model store error
103    #[error("Read model store error in projection '{name}': {message}")]
104    ReadModelError { name: String, message: String },
105
106    /// Other errors
107    #[error("Projection error: {message}")]
108    Other { message: String },
109}
110
111impl ProjectionError {
112    /// Create a handle failed error.
113    pub fn handle_failed(
114        name: impl Into<String>,
115        event_type: impl Into<String>,
116        event_id: impl Into<String>,
117        message: impl Into<String>,
118    ) -> Self {
119        ProjectionError::HandleFailed {
120            name: name.into(),
121            event_type: event_type.into(),
122            event_id: event_id.into(),
123            message: message.into(),
124        }
125    }
126
127    /// Create a clear failed error.
128    pub fn clear_failed(name: impl Into<String>, message: impl Into<String>) -> Self {
129        ProjectionError::ClearFailed {
130            name: name.into(),
131            message: message.into(),
132        }
133    }
134
135    /// Create a rebuild failed error.
136    pub fn rebuild_failed(name: impl Into<String>, message: impl Into<String>) -> Self {
137        ProjectionError::RebuildFailed {
138            name: name.into(),
139            message: message.into(),
140        }
141    }
142
143    /// Create a read model error.
144    pub fn read_model_error(name: impl Into<String>, message: impl Into<String>) -> Self {
145        ProjectionError::ReadModelError {
146            name: name.into(),
147            message: message.into(),
148        }
149    }
150
151    /// Create a generic error.
152    pub fn other(message: impl Into<String>) -> Self {
153        ProjectionError::Other {
154            message: message.into(),
155        }
156    }
157}

159/// Result type for projection operations.
160pub type ProjectionResult<T> = Result<T, ProjectionError>;

162/// Failure reasons for events received through an external projection
163/// delivery boundary.
164#[derive(Debug, Error)]
165pub enum ExternalProjectionError {
166    /// The event identity does not exist in the authoritative event store.
167    #[error("event is not present in the authoritative event store")]
168    MissingEvent,
169
170    /// The submitted event differs from the event persisted under its identity.
171    #[error("submitted event differs from the authoritative persisted event")]
172    EventMismatch,
173
174    /// The store could not verify or retrieve the authoritative event.
175    #[error("authoritative event lookup failed: {0}")]
176    EventStore(String),
177
178    /// A registered projection failed after the stored event was verified.
179    #[error("projection failed: {0}")]
180    Projection(String),
181}

183// ---------------------------------------------------------------------------
184// Projector trait — the stateless event handler
185// ---------------------------------------------------------------------------
186
187/// A projector contains the pure event-handling logic for building a read model.
188///
189/// Projectors are stateless — they receive events and translate them into write
190/// operations against a [`ReadModelStore`]. They do not own the store or the
191/// read model state.
192///
193/// # Design
194///
195/// - **Stateless**: all state lives in the `ReadModelStore`
196/// - **Deterministic**: same events + empty store = same read model
197/// - **Composable**: one projector per read model concern
198/// - **`&self`**: safe to share across threads
199///
200/// # Idempotency
201///
202/// `apply()` should be idempotent — handling the same event twice must produce
203/// the same result. Use UPSERT, check event_id, or make operations naturally
204/// idempotent (SET vs INCREMENT).
205///
206/// # Example
207///
208/// ```rust,ignore
209/// struct UserListProjector;
210///
211/// #[async_trait]
212/// impl Projector for UserListProjector {
213///     fn name(&self) -> &str { "UserList" }
214///
215///     fn handles(&self) -> Vec<String> {
216///         vec!["UserCreated".to_string()]
217///     }
218///
219///     async fn apply(&self, event: &Event, store: &dyn ReadModelStore) -> ProjectionResult<()> {
220///         store.upsert(Upsert::new("users_view", &event.aggregate_id, event.payload.clone())).await
221///             .map_err(|e| ProjectionError::handle_failed(
222///                 "UserList", &event.event_type, &event.event_id.to_string(), e.to_string()
223///             ))?;
224///         Ok(())
225///     }
226/// }
227/// ```
228#[async_trait]
229pub trait Projector: Send + Sync {
230    /// Unique name identifying this projector.
231    ///
232    /// Used for logging, monitoring, position tracking, and rebuild targeting.
233    fn name(&self) -> &str;
234
235    /// Event types this projector handles.
236    ///
237    /// Only events whose `event_type` is in this list will be passed to `apply()`.
238    fn handles(&self) -> Vec<String>;
239
240    /// Apply a single event to the read model via the store.
241    ///
242    /// This method should be idempotent: applying the same event twice
243    /// must produce the same result.
244    async fn apply(&self, event: &Event, store: &dyn ReadModelStore) -> ProjectionResult<()>;
245
246    /// Apply with access to authoritative history. Projectors with partial event
247    /// payloads can reconcile full current state before one version-gated write,
248    /// avoiding lost fields under concurrent or reordered delivery.
249    async fn apply_authoritative(
250        &self,
251        event: &Event,
252        store: &dyn ReadModelStore,
253        _event_store: &dyn EventStore,
254    ) -> ProjectionResult<()> {
255        self.apply(event, store).await
256    }
257
258    /// Initialize the read model schema (CREATE TABLE IF NOT EXISTS, etc.).
259    ///
260    /// Called once when the projector is first registered and before rebuilds.
261    /// Default implementation does nothing (for stores that don't need schema setup).
262    async fn init(&self, _store: &dyn ReadModelStore) -> ProjectionResult<()> {
263        Ok(())
264    }
265}

267// ---------------------------------------------------------------------------
268// Projection trait — the composed read model unit
269// ---------------------------------------------------------------------------
270
271/// A projection is the composed unit of a projector + its read model store.
272///
273/// It represents a complete, self-contained read model: the logic that transforms
274/// events into state, paired with the storage where that state lives.
275///
276/// Most users don't implement this trait directly. Instead, implement [`Projector`]
277/// and compose it with a [`ReadModelStore`] via [`ProjectionUnit`].
278///
279/// # `&self` not `&mut self`
280///
281/// All methods take `&self`. Mutable state lives in the `ReadModelStore`, which
282/// handles interior mutability via connection pools, `Mutex`, etc.
283#[async_trait]
284pub trait Projection: Send + Sync {
285    /// Projection name (delegates to the projector).
286    fn name(&self) -> &str;
287
288    /// Event types this projection handles (delegates to the projector).
289    fn handles(&self) -> Vec<String>;
290
291    /// Handle a single event by applying it through the projector to the store.
292    async fn handle(&self, event: &Event) -> ProjectionResult<()>;
293
294    async fn handle_authoritative(
295        &self,
296        event: &Event,
297        _event_store: &dyn EventStore,
298    ) -> ProjectionResult<()> {
299        self.handle(event).await
300    }
301
302    /// Clear all read model state for this projection.
303    async fn clear(&self) -> ProjectionResult<()>;
304
305    /// Rebuild from a set of events: clear, then replay matching events.
306    async fn rebuild(&self, events: Vec<Event>) -> ProjectionResult<()> {
307        self.clear().await?;
308        for event in events {
309            if self.handles().contains(&event.event_type) {
310                self.handle(&event).await?;
311            }
312        }
313        Ok(())
314    }
315}

317// ---------------------------------------------------------------------------
318// ProjectionUnit — standard composition glue
319// ---------------------------------------------------------------------------
320
321/// Standard composition of a [`Projector`] and a [`ReadModelStore`].
322///
323/// This is the typical way to create a [`Projection`]: provide the event-handling
324/// logic (projector) and the storage backend (store), and `ProjectionUnit` wires
325/// them together.
326///
327/// # Example
328///
329/// ```rust,ignore
330/// let projector = Box::new(UserListProjector);
331/// let store: Arc<dyn ReadModelStore> = Arc::new(SqliteReadModelStore::new(pool));
332/// let projection = ProjectionUnit::new(projector, store, "users_view");
333/// engine.register(Box::new(projection));
334/// ```
335pub struct ProjectionUnit {
336    projector: Box<dyn Projector>,
337    store: Arc<dyn ReadModelStore>,
338    /// Table/collection name used for `clear()` (truncate target).
339    table: String,
340}
341
342impl ProjectionUnit {
343    /// Create a new projection unit.
344    ///
345    /// # Arguments
346    ///
347    /// - `projector`: The stateless event handler
348    /// - `store`: The read model storage backend
349    /// - `table`: Table/collection name to truncate on `clear()`
350    pub fn new(
351        projector: Box<dyn Projector>,
352        store: Arc<dyn ReadModelStore>,
353        table: impl Into<String>,
354    ) -> Self {
355        Self {
356            projector,
357            store,
358            table: table.into(),
359        }
360    }
361}
362
363#[async_trait]
364impl Projection for ProjectionUnit {
365    fn name(&self) -> &str {
366        self.projector.name()
367    }
368
369    fn handles(&self) -> Vec<String> {
370        self.projector.handles()
371    }
372
373    async fn handle(&self, event: &Event) -> ProjectionResult<()> {
374        self.projector.apply(event, self.store.as_ref()).await
375    }
376
377    async fn handle_authoritative(
378        &self,
379        event: &Event,
380        event_store: &dyn EventStore,
381    ) -> ProjectionResult<()> {
382        self.projector
383            .apply_authoritative(event, self.store.as_ref(), event_store)
384            .await
385    }
386
387    async fn clear(&self) -> ProjectionResult<()> {
388        self.store
389            .truncate(&self.table)
390            .await
391            .map_err(|e| ProjectionError::clear_failed(self.projector.name(), e.to_string()))
392    }
393}

395// ---------------------------------------------------------------------------
396// ProjectionEngine — orchestrates multiple projections
397// ---------------------------------------------------------------------------
398
399/// Engine for managing multiple projections.
400///
401/// The `ProjectionEngine`:
402/// - Registers fully composed [`Projection`] instances
403/// - Routes events to interested projections
404/// - Rebuilds projections from the event store
405/// - Provides convenience registration via [`register_projector`](Self::register_projector)
406///
407/// # Example
408///
409/// ```rust,ignore
410/// let event_store = Box::new(sqlite_event_store);
411/// let mut engine = ProjectionEngine::new(event_store);
412///
413/// // Option 1: register a pre-composed projection
414/// engine.register(Box::new(projection_unit));
415///
416/// // Option 2: convenience — register projector + store directly
417/// engine.register_projector(Box::new(UserListProjector), store.clone(), "users_view");
418///
419/// // Process events
420/// engine.process(&event).await?;
421///
422/// // Rebuild all projections from event store
423/// engine.rebuild_all().await?;
424/// ```
425pub struct ProjectionEngine {
426    projections: Vec<Box<dyn Projection>>,
427    event_store: Box<dyn EventStore>,
428}
429
430impl ProjectionEngine {
431    /// Create a new projection engine.
432    pub fn new(event_store: Box<dyn EventStore>) -> Self {
433        Self {
434            projections: Vec::new(),
435            event_store,
436        }
437    }
438
439    /// Register a fully composed projection.
440    pub fn register(&mut self, projection: Box<dyn Projection>) {
441        tracing::info!("Registering projection: {}", projection.name());
442        self.projections.push(projection);
443    }
444
445    /// Convenience: register a projector + store as a [`ProjectionUnit`].
446    pub fn register_projector(
447        &mut self,
448        projector: Box<dyn Projector>,
449        store: Arc<dyn ReadModelStore>,
450        table: impl Into<String>,
451    ) {
452        let unit = ProjectionUnit::new(projector, store, table);
453        self.register(Box::new(unit));
454    }
455
456    /// Process a single event through all interested projections.
457    ///
458    /// Routes the event to projections whose `handles()` includes the event type.
459    pub async fn process(&self, event: &Event) -> ProjectionResult<()> {
460        for projection in &self.projections {
461            if projection.handles().contains(&event.event_type) {
462                tracing::debug!(
463                    "Processing event {} ({}) in projection {}",
464                    event.event_type,
465                    event.event_id,
466                    projection.name()
467                );
468
469                projection
470                    .handle_authoritative(event, self.event_store.as_ref())
471                    .await
472                    .map_err(|e| {
473                        ProjectionError::handle_failed(
474                            projection.name(),
475                            &event.event_type,
476                            event.event_id.to_string(),
477                            e.to_string(),
478                        )
479                    })?;
480            }
481        }
482        Ok(())
483    }
484
485    /// Validate an externally delivered event against the exact persisted
486    /// stream event, then project the authoritative stored value.
487    ///
488    /// Unlike [`Self::process`], this method is intended for untrusted delivery
489    /// boundaries. Stores without bounded `load_event` support fail closed.
490    pub async fn process_external_delivery(
491        &self,
492        submitted: &Event,
493    ) -> Result<(), ExternalProjectionError> {
494        let stored = self
495            .event_store
496            .load_event(
497                &submitted.aggregate_type,
498                &submitted.aggregate_id,
499                submitted.sequence,
500            )
501            .await
502            .map_err(|error| ExternalProjectionError::EventStore(error.to_string()))?
503            .ok_or(ExternalProjectionError::MissingEvent)?;
504
505        if stored != *submitted {
506            return Err(ExternalProjectionError::EventMismatch);
507        }
508
509        self.process(&stored)
510            .await
511            .map_err(|error| ExternalProjectionError::Projection(error.to_string()))
512    }
513
514    /// Process multiple events in sequence.
515    pub async fn process_batch(&self, events: Vec<Event>) -> ProjectionResult<()> {
516        for event in events {
517            self.process(&event).await?;
518        }
519        Ok(())
520    }
521
522    /// Rebuild all registered projections from the event store.
523    pub async fn rebuild_all(&self) -> ProjectionResult<()> {
524        tracing::info!("Rebuilding all projections");
525
526        let events = self
527            .event_store
528            .stream_all(0)
529            .await
530            .map_err(|e| ProjectionError::EventStoreError(e.to_string()))?;
531
532        tracing::info!("Loaded {} events for rebuild", events.len());
533
534        for projection in &self.projections {
535            tracing::info!("Rebuilding projection: {}", projection.name());
536
537            projection
538                .rebuild(events.clone())
539                .await
540                .map_err(|e| ProjectionError::rebuild_failed(projection.name(), e.to_string()))?;
541
542            tracing::info!("Rebuilt projection: {}", projection.name());
543        }
544
545        Ok(())
546    }
547
548    /// Rebuild a specific projection by name.
549    pub async fn rebuild_projection(&self, name: &str) -> ProjectionResult<()> {
550        tracing::info!("Rebuilding projection: {}", name);
551
552        let projection = self
553            .projections
554            .iter()
555            .find(|p| p.name() == name)
556            .ok_or_else(|| ProjectionError::other(format!("Projection not found: {}", name)))?;
557
558        let events = self
559            .event_store
560            .stream_all(0)
561            .await
562            .map_err(|e| ProjectionError::EventStoreError(e.to_string()))?;
563
564        projection
565            .rebuild(events)
566            .await
567            .map_err(|e| ProjectionError::rebuild_failed(name, e.to_string()))?;
568
569        tracing::info!("Rebuilt projection: {}", name);
570        Ok(())
571    }
572
573    /// Get number of registered projections.
574    pub fn projection_count(&self) -> usize {
575        self.projections.len()
576    }
577
578    /// Get names of all registered projections.
579    pub fn projection_names(&self) -> Vec<String> {
580        self.projections
581            .iter()
582            .map(|p| p.name().to_string())
583            .collect()
584    }
585
586    /// Union of every event type any registered projection handles. Used by
587    /// [`ProjectionEngineHandler`] to declare its `handles()` set when
588    /// subscribing to an [`EventBus`](crate::event_bus::EventBus).
589    pub fn all_handled_event_types(&self) -> Vec<String> {
590        let mut all: Vec<String> = self.projections.iter().flat_map(|p| p.handles()).collect();
591        all.sort();
592        all.dedup();
593        all
594    }
595}

597// ---------------------------------------------------------------------------
598// EventBus adapter — drive the engine from an in-process bus
599// ---------------------------------------------------------------------------
600
601/// Adapter that lets a [`ProjectionEngine`] subscribe to an
602/// [`EventBus`](crate::event_bus::EventBus). Wraps the engine in an
603/// [`EventHandler`] that routes every relevant event through
604/// [`ProjectionEngine::process`]. The engine stays accessible from outside
605/// (e.g. for `rebuild_all`) via the same [`Arc`].
606///
607/// Lives in `arc-core` because the adapter needs nothing app-specific —
608/// any aggregate's projector can be driven through it.
609pub struct ProjectionEngineHandler {
610    engine: Arc<ProjectionEngine>,
611    handles: Vec<String>,
612}
613
614impl ProjectionEngineHandler {
615    pub fn new(engine: Arc<ProjectionEngine>) -> Self {
616        let handles = engine.all_handled_event_types();
617        Self { engine, handles }
618    }
619}
620
621#[async_trait]
622impl EventHandler for ProjectionEngineHandler {
623    fn handles(&self) -> Vec<String> {
624        self.handles.clone()
625    }
626
627    async fn handle(&self, event: &Event) -> Result<(), Box<dyn std::error::Error + Send + Sync>> {
628        self.engine
629            .process(event)
630            .await
631            .map_err(|e| -> Box<dyn std::error::Error + Send + Sync> { Box::new(e) })
632    }
633}

749        async fn apply(&self, event: &Event, store: &dyn ReadModelStore) -> ProjectionResult<()> {
750            use crate::read_model_store::Upsert;
751            store
752                .upsert(Upsert::new(
753                    "test_table",
754                    event.event_id.to_string(),
755                    serde_json::json!({
756                        "id": event.event_id.to_string(),
757                        "event_type": event.event_type,
758                        "version": event.sequence,
759                    }),
760                ))
761                .await
762                .map_err(|e| {
763                    ProjectionError::handle_failed(
764                        &self.name,
765                        &event.event_type,
766                        event.event_id.to_string(),
767                        e.to_string(),
768                    )
769                })?;
770            Ok(())
771        }
772    }
