1//! Shared connection policy for store-owned pools. Injected pools keep their policy.
2use diesel::connection::{AnsiTransactionManager, SimpleConnection, TransactionManager};
3use diesel::r2d2::{ConnectionManager, CustomizeConnection, Error};
4use diesel::SqliteConnection;

6/// Maximum SQLite lock wait per statement on every store-owned connection,
7/// including replacements. This changes neither journal mode nor durability.
8pub(crate) const BUSY_TIMEOUT_MS: u32 = 5_000;
9
10/// The builder of every store-owned pool. r2d2 would test each connection it
11/// lends with `SELECT 1`, before the store can queue for the file. On a new
12/// connection, including a replacement, that is the first statement, the one
13/// that loads the schema under a shared lock on the file: outside the queue
14/// it polls for the lock against one commit after another, can lose every
15/// time until the busy timeout, and then r2d2 discards the connection and
16/// tries another. Without the test the first statement is the store's own,
17/// under its turn. The test would catch nothing here: a SQLite connection
18/// does not go stale, and Diesel already has r2d2 discard a connection left
19/// in a broken transaction.
20pub(crate) fn pool_builder() -> diesel::r2d2::Builder<ConnectionManager<SqliteConnection>> {
21    diesel::r2d2::Pool::builder().test_on_check_out(false)
22}

25pub(crate) struct BusyTimeout;

27impl CustomizeConnection<SqliteConnection, Error> for BusyTimeout {
28    fn on_acquire(&self, conn: &mut SqliteConnection) -> Result<(), Error> {
29        conn.batch_execute(&format!("PRAGMA busy_timeout = {BUSY_TIMEOUT_MS}"))
30            .map_err(Error::QueryError)
31    }
32}

34/// Clean up without replacing the original append error. Diesel may already
35/// have rolled back a failed commit. If rollback fails, mark the connection
36/// broken so r2d2 disposes of it rather than lending dirty state to a caller.
37pub(crate) fn rollback_append(conn: &mut SqliteConnection) {
38    if matches!(
39        AnsiTransactionManager::transaction_manager_status_mut(conn).transaction_depth(),
40        Ok(None)
41    ) {
42        return;
43    }
44    if AnsiTransactionManager::rollback_transaction(conn).is_err() {
45        *AnsiTransactionManager::transaction_manager_status_mut(conn) =
46            diesel::connection::TransactionManagerStatus::InError;
47    }
48}

50/// Run `body` as one write transaction for the holder of the file's write
51/// slot. The writer is reserved up front (`BEGIN IMMEDIATE`): a deferred
52/// read-to-write upgrade can fail immediately even with a busy timeout, and
53/// while this waits for a writer outside the process the stores' readers
54/// carry on. They are held back only for the commit itself. The body or
55/// commit error is preserved; COMMIT can fail (for example while an external
56/// reader holds a shared lock) and leave a transaction open.
57pub(crate) fn write_transaction<T, E>(
58    conn: &mut SqliteConnection,
59    permit: &crate::file_gate::WritePermit,
60    database_error: impl Fn(String) -> E,
61    body: impl FnOnce(&mut SqliteConnection) -> Result<T, E>,
62) -> Result<T, E> {
63    let usual_timeout = spend_busy_timeout(conn, permit.queued_behind_refusals())
64        .map_err(|e| database_error(e.to_string()))?;
65    let begun = AnsiTransactionManager::begin_transaction_sql(conn, "BEGIN IMMEDIATE");
66    // Only the wait for the write lock is shortened: it is the one this
67    // writer already sat through in the queue. The body and COMMIT, which
68    // may wait for a reader, get the connection's usual timeout back.
69    if let Some(timeout) = usual_timeout {
70        if let Err(error) = set_busy_timeout(conn, timeout) {
71            // A connection that cannot get its timeout back must not be reused.
72            rollback_append(conn);
73            *AnsiTransactionManager::transaction_manager_status_mut(conn) =
74                diesel::connection::TransactionManagerStatus::InError;
75            return Err(database_error(error.to_string()));
76        }
77    }
78    let result = begun
79        .map_err(|e| database_error(e.to_string()))
80        .and_then(|()| body(conn))
81        .and_then(|value| {
82            permit.exclude_readers();
83            AnsiTransactionManager::commit_transaction(conn)
84                .map(|()| value)
85                .map_err(|e| database_error(e.to_string()))
86        });
87    if result.is_err() {
88        rollback_append(conn);
89    }
90    result
91}

93/// A writer that queued behind attempts refused by a lock outside the gate
94/// has already waited `spent` for that lock. Take it off this connection's
95/// busy timeout for the writer's own attempt at the write lock, which it
96/// still makes, so that queueing does not multiply the bounded wait. Returns
97/// the timeout to put back, if it was changed.
98fn spend_busy_timeout(
99    conn: &mut SqliteConnection,
100    spent: std::time::Duration,
101) -> diesel::QueryResult<Option<i32>> {
102    use diesel::prelude::*;
103    #[derive(QueryableByName)]
104    struct Timeout {
105        #[diesel(sql_type = diesel::sql_types::Integer)]
106        timeout: i32,
107    }
108    if spent.is_zero() {
109        return Ok(None);
110    }
111    let usual = diesel::sql_query("PRAGMA busy_timeout")
112        .get_result::<Timeout>(conn)?
113        .timeout;
114    if usual <= 0 {
115        return Ok(None);
116    }
117    let spent = i32::try_from(spent.as_millis()).unwrap_or(i32::MAX);
118    set_busy_timeout(conn, usual.saturating_sub(spent).max(0))?;
119    Ok(Some(usual))
120}
121
122fn set_busy_timeout(conn: &mut SqliteConnection, milliseconds: i32) -> diesel::QueryResult<()> {
123    conn.batch_execute(&format!("PRAGMA busy_timeout = {milliseconds}"))
124}

126#[cfg(test)]
127pub(crate) fn assert_timeout_on_replacement(
128    pool: &diesel::r2d2::Pool<diesel::r2d2::ConnectionManager<SqliteConnection>>,
129) {
130    use diesel::prelude::*;
131    #[derive(QueryableByName)]
132    struct Timeout {
133        #[diesel(sql_type = diesel::sql_types::Integer)]
134        timeout: i32,
135    }
136    let check = |conn: &mut SqliteConnection| {
137        let value = diesel::sql_query("PRAGMA busy_timeout")
138            .get_result::<Timeout>(conn)
139            .unwrap();
140        assert_eq!(value.timeout, BUSY_TIMEOUT_MS as i32);
141    };
142    // Hold all original connections; returning one with an open transaction
143    // forces Diesel/r2d2 to discard it and create a replacement.
144    let mut held: Vec<_> = (0..10).map(|_| pool.get().unwrap()).collect();
145    for conn in &mut held {
146        check(conn);
147    }
148    let mut discarded = held.pop().unwrap();
149    discarded
150        .batch_execute("PRAGMA busy_timeout = 123")
151        .unwrap();
152    AnsiTransactionManager::begin_transaction(&mut *discarded).unwrap();
153    drop(discarded);
154    let mut replacement = pool.get().unwrap();
155    check(&mut replacement);
156    assert!(
157        AnsiTransactionManager::transaction_manager_status_mut(&mut *replacement)
158            .transaction_depth()
159            .unwrap()
160            .is_none()
161    );
162}

192    fn busy_timeout(conn: &mut SqliteConnection) -> i32 {
193        #[derive(QueryableByName)]
194        struct Timeout {
195            #[diesel(sql_type = diesel::sql_types::Integer)]
196            timeout: i32,
197        }
198        diesel::sql_query("PRAGMA busy_timeout")
199            .get_result::<Timeout>(conn)
200            .unwrap()
201            .timeout
202    }

204    /// A connection with a 300 ms busy timeout that records its statements.
205    fn recorded(file: &File) -> (SqliteConnection, Arc<Mutex<Vec<String>>>) {
206        let mut conn = file.connect();
207        set_busy_timeout(&mut conn, 300).unwrap();
208        let statements = Arc::new(Mutex::new(Vec::new()));
209        let record = statements.clone();
210        conn.set_instrumentation(move |event: InstrumentationEvent<'_>| {
211            if let InstrumentationEvent::StartQuery { query, .. } = event {
212                let query = query.to_string();
213                let statement = query.split(" -- ").next().unwrap_or_default();
214                record.lock().unwrap().push(statement.to_string());
215            }
216        });
217        (conn, statements)
218    }

220    #[test]
221    fn time_spent_queueing_shortens_only_the_wait_for_the_write_lock() {
222        let file = File::new();
223        let (mut conn, statements) = recorded(&file);
224        // Queued 200 of its 300 ms behind refused attempts: 100 ms are left
225        // for BEGIN IMMEDIATE, and the full 300 for the body and COMMIT.
226        let permit = crate::file_gate::WritePermit::queued_behind(Duration::from_millis(200));
227        let during_body = write_transaction(
228            &mut conn,
229            &permit,
230            |e| e,
231            |conn| {
232                let timeout = busy_timeout(conn);
233                conn.batch_execute("INSERT INTO notes VALUES ('kept')")
234                    .map_err(|e| e.to_string())?;
235                Ok(timeout)
236            },
237        )
238        .unwrap();
239        assert_eq!(during_body, 300);
240        assert_eq!(busy_timeout(&mut conn), 300);
241        assert_eq!(
242            statements.lock().unwrap()[..6],
243            [
244                "PRAGMA busy_timeout",
245                "PRAGMA busy_timeout = 100",
246                "BEGIN IMMEDIATE",
247                "PRAGMA busy_timeout = 300",
248                "PRAGMA busy_timeout",
249                "INSERT INTO notes VALUES ('kept')",
250            ]
251        );
252
253        // A writer that queued its whole timeout still tries once.
254        let permit = crate::file_gate::WritePermit::queued_behind(Duration::from_secs(60));
255        statements.lock().unwrap().clear();
256        write_transaction(&mut conn, &permit, |e| e, |_| Ok(())).unwrap();
257        assert_eq!(
258            statements.lock().unwrap()[1..3],
259            ["PRAGMA busy_timeout = 0", "BEGIN IMMEDIATE"]
260        );
261        assert_eq!(busy_timeout(&mut conn), 300);
262    }

264    /// The first statement on a new connection loads the schema, which needs
265    /// a shared lock on the file. Outside the queue it polls for that lock
266    /// against one commit after another and can lose every time until the
267    /// busy timeout; in the queue it waits for one commit. Every connection of
268    /// a new store is new, and so is a replacement.
269    #[tokio::test(flavor = "multi_thread", worker_threads = 4)]
270    async fn new_connections_read_in_turn_while_others_commit_nonstop() {
271        use arc_core::event_store::EventStore;
272        use arc_core::read_model_store::ReadModelStore;
273        const HOLD: Duration = Duration::from_millis(100);
274        // A few commits; a read that polls lasts the 5-second busy timeout.
275        const IN_TURN: Duration = Duration::from_secs(2);
276        let file = File::new();
277        crate::test_support::migrate(&mut file.connect());
278        let url = file.0.to_str().unwrap();
279        let commits = crate::test_support::SlowCommits::start(url, 2, HOLD);
280        tokio::time::sleep(HOLD * 2).await;
281
282        let events = crate::SqliteEventStore::new(url).await.unwrap();
283        let views = crate::SqliteReadModelStore::new(url).await.unwrap();
284        let began = Instant::now();
285        let mut reads = tokio::task::JoinSet::new();
286        // As many at once as a pool has connections, so each takes a new one.
287        for _ in 0..10 {
288            let (events, views) = (events.clone(), views.clone());
289            reads.spawn(async move {
290                assert!(events.load("none").await.unwrap().is_empty());
291                assert!(views.get("users_view", "none").await.unwrap().is_none());
292            });
293        }
294        while let Some(read) = reads.join_next().await {
295            read.unwrap();
296        }
297        let first = began.elapsed();
298
299        // Returning a connection with an open transaction makes r2d2 discard
300        // it; with the others held, the next read gets its replacement.
301        let pool = events.pool.clone();
302        let held = tokio::task::spawn_blocking(move || {
303            let mut held: Vec<_> = (0..10).map(|_| pool.get().unwrap()).collect();
304            let mut discarded = held.pop().unwrap();
305            AnsiTransactionManager::begin_transaction(&mut *discarded).unwrap();
306            drop(discarded);
307            held
308        })
309        .await
310        .unwrap();
311        let began = Instant::now();
312        assert!(events.load("none").await.unwrap().is_empty());
313        let replacement = began.elapsed();
314        drop(held);
315
316        let made = commits.stop();
317        assert!(
318            first < IN_TURN,
319            "first reads polled for the file: {first:?}"
320        );
321        assert!(
322            replacement < IN_TURN,
323            "a replacement connection polled for the file: {replacement:?}"
324        );
325        assert!(made >= 4, "the commits stopped: {made}");
326    }

328    #[test]
329    fn the_usual_timeout_returns_when_the_write_lock_is_refused() {
330        let file = File::new();
331        let mut outside = file.connect();
332        outside.batch_execute("BEGIN IMMEDIATE").unwrap();
333        let (mut conn, _) = recorded(&file);
334        let permit = crate::file_gate::WritePermit::queued_behind(Duration::from_secs(60));
335        let refused = write_transaction(&mut conn, &permit, |e| e, |_| Ok(())).unwrap_err();
336        assert!(refused.contains(crate::file_gate::LOCKED), "{refused}");
337        assert_eq!(busy_timeout(&mut conn), 300);
338        // The connection is clean and usable once the lock is gone.
339        outside.batch_execute("ROLLBACK").unwrap();
340        write_transaction(&mut conn, &permit, |e| e, |_| Ok::<_, String>(())).unwrap();
341
342        // A writer that did not queue behind refusals is left alone.
343        let (mut conn, statements) = recorded(&file);
344        let permit = crate::file_gate::WritePermit::queued_behind(Duration::ZERO);
345        write_transaction(&mut conn, &permit, |e| e, |_| Ok::<_, String>(())).unwrap();
346        assert_eq!(
347            statements.lock().unwrap()[..2],
348            ["BEGIN IMMEDIATE", "COMMIT"]
349        );
350    }
