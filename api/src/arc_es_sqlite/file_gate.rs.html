1//! Fair in-process admission to one SQLite database file.
2//!
3//! SQLite arbitrates a database file by polling: a connection that finds the
4//! file locked sleeps and retries, and whichever polls first after a commit
5//! wins. That is not fair. Under sustained contention one caller can lose
6//! every retry until its busy timeout expires and report `database is
7//! locked`, although nobody ever held the file for long. It happens to
8//! writers racing each other for the write lock, and, in rollback-journal
9//! mode, to a reader polling against one commit after another.
10//!
11//! The stores of this crate therefore follow SQLite's own protocol among
12//! themselves before they enter it, with queues instead of polls:
13//!
14//! - one writer at a time holds the *write slot*, granted first-in,
15//!   first-out, from before `BEGIN IMMEDIATE` until after commit or rollback
16//!   (SQLite's `RESERVED` lock);
17//! - readers run together, also beside a writer that has not reached commit
18//!   (`SHARED`);
19//! - to commit, the writer stops admitting readers and gives the running
20//!   ones a short grace to finish (`PENDING`, `EXCLUSIVE`); readers that
21//!   arrive meanwhile are all admitted when the commit ends, before the next
22//!   one can begin.
23//!
24//! So nobody is overtaken indefinitely: a writer waits for the writers that
25//! arrived before it, a read for at most one commit. The gate is shared by
26//! everything in the process that uses the same file, whichever store, pool
27//! or spelling of the path it came from: it is found through the file name
28//! SQLite reports for the connection.
29//!
30//! The gate only orders callers; it never replaces SQLite's locks, and where
31//! it cannot know what SQLite will do it steps aside:
32//!
33//! - The wait for running readers is bounded ([`DRAIN_GRACE`]). A reader may
34//!   itself be waiting inside SQLite for this writer, which takes the
35//!   exclusive lock early when its transaction outgrows the page cache; and
36//!   a long read is not worth queueing behind. After the grace the writer
37//!   commits and SQLite arbitrates, as it would without the gate.
38//! - In WAL mode readers and writers do not block each other, so readers are
39//!   not gated and commits do not wait for them; writers still take turns.
40//!   The journal mode is read once, by a store's first operation. A database
41//!   switched to WAL afterwards keeps its readers gated until the store is
42//!   reopened (slower, not wrong); SQLite does not let a database leave WAL
43//!   while other connections are open.
44//! - Only this crate's stores take part. Another process, or a connection
45//!   opened outside these stores, still competes through SQLite's busy
46//!   handler, so the busy timeout stays the bound for that contention. Time a
47//!   writer spent queued behind attempts that such a lock refused is taken
48//!   off the busy timeout of its own attempt at the write lock, which it
49//!   always makes; its commit keeps the full timeout.
50//! - In-memory and temporary databases have no file name and are not gated.
51//! - A permit is taken on the blocking thread and released on it; it is never
52//!   held across an `await`. An operation nested inside another on the same
53//!   thread passes without queueing.

71/// How long a commit waits for this process's running reads before it goes
72/// ahead and leaves the rest to SQLite. Store reads take well under this; a
73/// read still running after it is either long, or stuck behind this very
74/// writer inside SQLite.
75pub(crate) const DRAIN_GRACE: Duration = Duration::from_millis(20);

77struct Waiter {
78    thread: Thread,
79    granted: AtomicBool,
80}
81
82impl Waiter {
83    fn new() -> Arc<Self> {
84        Arc::new(Self {
85            thread: std::thread::current(),
86            granted: AtomicBool::new(false),
87        })
88    }
89
90    /// `park` may wake spuriously; only the flag grants.
91    fn wait(&self) {
92        while !self.granted.load(Ordering::Acquire) {
93            std::thread::park();
94        }
95    }
96
97    /// False if `grace` passed first.
98    fn wait_for(&self, grace: Duration) -> bool {
99        let deadline = Instant::now() + grace;
100        loop {
101            if self.granted.load(Ordering::Acquire) {
102                return true;
103            }
104            let Some(left) = deadline.checked_duration_since(Instant::now()) else {
105                return false;
106            };
107            std::thread::park_timeout(left);
108        }
109    }
110
111    fn grant(&self) {
112        self.granted.store(true, Ordering::Release);
113        self.thread.unpark();
114    }
115}

117#[derive(Default)]
118struct State {
119    /// The write slot is taken.
120    writer: bool,
121    /// Writers waiting for the slot, in arrival order.
122    writers: VecDeque<Arc<Waiter>>,
123    /// Reads running.
124    readers: usize,
125    /// The writer is committing, or waiting to: no new reads start.
126    committing: bool,
127    /// The writer waiting for the running reads to finish.
128    draining: Option<Arc<Waiter>>,
129    /// Reads waiting for the commit to end.
130    parked: Vec<Arc<Waiter>>,
131    /// When the holder of the write slot got it.
132    attempt_since: Option<Instant>,
133    /// Total length of the attempts SQLite refused as locked. Attempts hold
134    /// the slot one at a time, so this is time, not overlapping intervals.
135    refused: Duration,
136}
137
138impl State {
139    /// What a writer arriving at `now` notes, to learn at its turn how long
140    /// it queued behind refused attempts (see [`State::queued_behind_refusals`]).
141    /// The part of the running attempt that is already over is not its wait.
142    fn refusals_before(&self, now: Instant) -> Duration {
143        let running = self
144            .attempt_since
145            .map(|since| now.saturating_duration_since(since));
146        self.refused + running.unwrap_or_default()
147    }
148
149    /// The time a writer that noted `before` on arrival has since spent
150    /// queued behind attempts refused as locked. Never more than it waited.
151    fn queued_behind_refusals(&self, before: Duration) -> Duration {
152        self.refused.saturating_sub(before)
153    }
154
155    fn record_refusal(&mut self, now: Instant) {
156        if let Some(since) = self.attempt_since {
157            self.refused += now.saturating_duration_since(since);
158        }
159    }
160}

162/// The queues described in the module documentation. `std::sync` locks
163/// promise no order, and a barging lock would only move the starvation here.
164#[derive(Default)]
165struct Gate {
166    state: Mutex<State>,
167}
168
169impl Gate {
170    /// The critical sections below cannot panic, and a holder that panics
171    /// releases on unwind, so there is nothing a poisoned flag could protect:
172    /// ignore it rather than fail every later caller.
173    fn state(&self) -> MutexGuard<'_, State> {
174        self.state.lock().unwrap_or_else(PoisonError::into_inner)
175    }
176
177    fn begin_read(&self) {
178        let waiter = {
179            let mut state = self.state();
180            if !state.committing {
181                state.readers += 1;
182                return;
183            }
184            let waiter = Waiter::new();
185            state.parked.push(waiter.clone());
186            waiter
187        };
188        waiter.wait();
189    }
190
191    fn end_read(&self) {
192        let mut state = self.state();
193        state.readers -= 1;
194        if state.readers == 0 {
195            if let Some(writer) = state.draining.take() {
196                writer.grant();
197            }
198        }
199    }
200
201    /// Returns how long the caller queued behind attempts refused as locked.
202    fn begin_write(&self) -> Duration {
203        let (waiter, before) = {
204            let mut state = self.state();
205            let now = Instant::now();
206            if !state.writer {
207                state.writer = true;
208                state.attempt_since = Some(now);
209                return Duration::ZERO;
210            }
211            let waiter = Waiter::new();
212            state.writers.push_back(waiter.clone());
213            (waiter, state.refusals_before(now))
214        };
215        waiter.wait();
216        self.state().queued_behind_refusals(before)
217    }
218
219    /// For the holder of the write slot, before it commits: stop admitting
220    /// readers and give the running ones `grace` to finish.
221    fn exclude_readers(&self, grace: Duration) {
222        let waiter = {
223            let mut state = self.state();
224            state.committing = true;
225            if state.readers == 0 {
226                return;
227            }
228            let waiter = Waiter::new();
229            state.draining = Some(waiter.clone());
230            waiter
231        };
232        if !waiter.wait_for(grace) {
233            self.state().draining = None;
234        }
235    }
236
237    fn end_write(&self) {
238        let mut state = self.state();
239        state.committing = false;
240        state.draining = None;
241        // Every read that waited out this commit runs before the next one.
242        state.readers += state.parked.len();
243        for reader in state.parked.drain(..) {
244            reader.grant();
245        }
246        // The slot passes on directly, in arrival order.
247        match state.writers.pop_front() {
248            Some(next) => {
249                state.attempt_since = Some(Instant::now());
250                next.grant();
251            }
252            None => {
253                state.writer = false;
254                state.attempt_since = None;
255            }
256        }
257    }
258}

260fn registry() -> MutexGuard<'static, HashMap<PathBuf, Weak<Gate>>> {
261    static REGISTRY: OnceLock<Mutex<HashMap<PathBuf, Weak<Gate>>>> = OnceLock::new();
262    REGISTRY
263        .get_or_init(Default::default)
264        .lock()
265        .unwrap_or_else(PoisonError::into_inner)
266}

268fn gate_for(file: PathBuf) -> Arc<Gate> {
269    let mut gates = registry();
270    if let Some(gate) = gates.get(&file).and_then(Weak::upgrade) {
271        return gate;
272    }
273    gates.retain(|_, gate| gate.strong_count() > 0);
274    let gate = Arc::new(Gate::default());
275    gates.insert(file, Arc::downgrade(&gate));
276    gate
277}

305/// Ask the connection which file it uses and how. SQLite reports an absolute
306/// name with symbolic links resolved, whatever spelling, relative path or
307/// `file:` URI opened it, and an empty name for in-memory and temporary
308/// databases. `Err` means the question could not be asked; `Ok(None)` means
309/// no file.
310fn describe(conn: &mut SqliteConnection) -> QueryResult<Option<Target>> {
311    let file = diesel::sql_query("PRAGMA database_list")
312        .load::<Attached>(conn)?
313        .into_iter()
314        .find(|database| database.name == "main")
315        .and_then(|database| database.file)
316        .filter(|file| !file.is_empty());
317    let Some(file) = file else {
318        return Ok(None);
319    };
320    let journal_mode = diesel::sql_query("PRAGMA journal_mode")
321        .get_result::<JournalMode>(conn)?
322        .journal_mode;
323    let file = PathBuf::from(file);
324    Ok(Some(Target {
325        gate: gate_for(std::fs::canonicalize(&file).unwrap_or(file)),
326        readers_gated: !journal_mode.eq_ignore_ascii_case("wal"),
327    }))
328}

332/// A store's handle on the gate of its database file. The file and its
333/// journal mode are resolved once, by the store's first operation, and kept
334/// for the life of the store.
335#[derive(Default)]
336pub(crate) struct FileGate {
337    target: OnceLock<Option<Target>>,
338}
339
340impl FileGate {
341    pub(crate) fn new() -> Arc<Self> {
342        Arc::default()
343    }
344
345    /// Admit a read on `conn`, and keep the permit until the read's last
346    /// statement has finished. The reader takes its connection first: one
347    /// that still waited for a pooled connection while counted as running
348    /// could be waiting for the very writer whose commit waits for it.
349    pub(crate) fn read(&self, conn: &mut SqliteConnection) -> ReadPermit {
350        if self.target.get().is_none() {
351            self.resolve(conn);
352        }
353        let target = self.target.get().and_then(|target| target.as_ref());
354        let Some(target) = target.filter(|target| target.readers_gated && !inside(&target.gate))
355        else {
356            return ReadPermit(None, PhantomData);
357        };
358        target.gate.begin_read();
359        enter(&target.gate);
360        ReadPermit(Some(target.gate.clone()), PhantomData)
361    }
362
363    /// Queue for the write slot of the database file behind `pool`. Call it
364    /// on the blocking thread before taking the connection the write will
365    /// use, so a queued writer does not keep a pooled connection from others.
366    pub(crate) fn write(&self, pool: &ConnectionPool) -> WritePermit {
367        if self.target.get().is_none() {
368            // If no connection can be had, the operation itself reports that.
369            // The connection goes back to the pool before queueing.
370            if let Ok(mut conn) = pool.get() {
371                self.resolve(&mut conn);
372            }
373        }
374        self.admit_writer()
375    }
376
377    /// As [`write`](Self::write), for a writer that owns its connection.
378    pub(crate) fn write_on(&self, conn: &mut SqliteConnection) -> WritePermit {
379        if self.target.get().is_none() {
380            self.resolve(conn);
381        }
382        self.admit_writer()
383    }
384
385    /// If the connection cannot be asked, this operation runs ungated and
386    /// the next one asks again.
387    fn resolve(&self, conn: &mut SqliteConnection) {
388        if let Ok(target) = describe(conn) {
389            let _ = self.target.set(target);
390        }
391    }
392
393    fn admit_writer(&self) -> WritePermit {
394        let Some(Some(target)) = self.target.get() else {
395            return WritePermit::ungated();
396        };
397        if inside(&target.gate) {
398            return WritePermit::ungated();
399        }
400        let queued_behind_refusals = target.gate.begin_write();
401        enter(&target.gate);
402        WritePermit {
403            held: Some(Held {
404                gate: target.gate.clone(),
405                readers_gated: target.readers_gated,
406                queued_behind_refusals,
407            }),
408            _on_this_thread: PhantomData,
409        }
410    }
411}
412
413fn inside(gate: &Arc<Gate>) -> bool {
414    INSIDE.with(|inside| inside.borrow().contains(&Arc::as_ptr(gate)))
415}
416
417fn enter(gate: &Arc<Gate>) {
418    INSIDE.with(|inside| inside.borrow_mut().push(Arc::as_ptr(gate)));
419}
420
421fn leave(gate: &Arc<Gate>) {
422    INSIDE.with(|inside| {
423        inside
424            .borrow_mut()
425            .retain(|held| *held != Arc::as_ptr(gate))
426    });
427}

436            leave(&gate);
437            gate.end_read();

442struct Held {
443    gate: Arc<Gate>,
444    readers_gated: bool,
445    queued_behind_refusals: Duration,
446}
447
448/// The write slot. Dropping it, on any path including a panic, passes the
449/// slot on; it cannot leave the thread that obtained it.
450pub(crate) struct WritePermit {
451    held: Option<Held>,
452    _on_this_thread: PhantomData<*const ()>,
453}
454
455impl WritePermit {
456    fn ungated() -> Self {
457        Self {
458            held: None,
459            _on_this_thread: PhantomData,
460        }
461    }
462
463    /// A permit on a gate of its own that queued `behind_refusals`.
464    #[cfg(test)]
465    pub(crate) fn queued_behind(behind_refusals: Duration) -> Self {
466        let gate = Arc::new(Gate::default());
467        gate.begin_write();
468        Self {
469            held: Some(Held {
470                gate,
471                readers_gated: true,
472                queued_behind_refusals: behind_refusals,
473            }),
474            _on_this_thread: PhantomData,
475        }
476    }
477
478    /// How long this writer queued behind attempts that a lock outside the
479    /// gate refused. It has waited out that lock for as long already, so its
480    /// own attempt should not wait a whole busy timeout again: queueing must
481    /// not multiply the bounded wait.
482    pub(crate) fn queued_behind_refusals(&self) -> Duration {
483        self.held
484            .as_ref()
485            .map_or(Duration::ZERO, |held| held.queued_behind_refusals)
486    }
487
488    /// Call before `COMMIT`: stop admitting this process's readers and give
489    /// the running ones a short grace, so that normally the commit does not
490    /// wait for them inside SQLite and they do not poll for the file against
491    /// it. Nothing to do in WAL mode.
492    pub(crate) fn exclude_readers(&self) {
493        if let Some(held) = self.held.as_ref().filter(|held| held.readers_gated) {
494            held.gate.exclude_readers(DRAIN_GRACE);
495        }
496    }
497
498    /// Release after the write has committed or rolled back, recording
499    /// whether SQLite refused it as locked.
500    pub(crate) fn finish<T, E: Display>(self, outcome: &Result<T, E>) {
501        if let (Some(held), Err(error)) = (&self.held, outcome) {
502            if error.to_string().contains(LOCKED) {
503                held.gate.state().record_refusal(Instant::now());
504            }
505        }
506    }
507}

512            leave(&held.gate);
513            held.gate.end_write();
514        }
515    }

563    fn the_write_slot_is_granted_in_arrival_order() {
564        let gate = Arc::new(Gate::default());
565        gate.begin_write();
566        let (order_tx, order_rx) = mpsc::channel();
567        let mut threads = Vec::new();

590    #[test]
591    fn reads_run_beside_a_writer_until_it_commits_and_wait_out_one_commit() {
592        let gate = Arc::new(Gate::default());
593        // Reads run together, and beside a writer that has not reached commit.
594        gate.begin_read();
595        gate.begin_write();
596        gate.begin_read();
597        assert_eq!(gate.state().readers, 2);
598
599        // The commit waits for both running reads.
600        let committing = gate.clone();
601        let (committed_tx, committed_rx) = mpsc::channel();
602        let writer = std::thread::spawn(move || {
603            committing.exclude_readers(NEVER);
604            committed_tx.send(committing.state().readers).unwrap();
605        });
606        until(&gate, |state| state.draining.is_some());
607
608        // Reads arriving now wait for the commit to end, however many. Each
609        // reports when it runs and keeps running until told to finish.
610        let (running_tx, running_rx) = mpsc::channel();
611        let (finish_tx, finish_rx) = mpsc::channel::<()>();
612        let finish_rx = Arc::new(Mutex::new(finish_rx));
613        let late: Vec<_> = (0..3)
614            .map(|_| {
615                let (reading, running_tx, finish_rx) =
616                    (gate.clone(), running_tx.clone(), finish_rx.clone());
617                std::thread::spawn(move || {
618                    reading.begin_read();
619                    running_tx.send(()).unwrap();
620                    finish_rx.lock().unwrap().recv_timeout(NEVER).unwrap();
621                    reading.end_read();
622                })
623            })
624            .collect();
625        until(&gate, |state| state.parked.len() == 3);
626        gate.end_read();
627        assert!(committed_rx.try_recv().is_err(), "one read still runs");
628        gate.end_read();
629        assert_eq!(committed_rx.recv_timeout(NEVER).unwrap(), 0);
630        writer.join().unwrap();
631        assert!(running_rx.try_recv().is_err(), "late reads still wait");
632
633        // A second writer is queued for the slot. When the first ends, all
634        // waiting reads run at once, and the second writer's commit then
635        // waits for them: no read waits out more than one commit.
636        let second = gate.clone();
637        let (second_tx, second_rx) = mpsc::channel();
638        let second_writer = std::thread::spawn(move || {
639            second.begin_write();
640            second.exclude_readers(NEVER);
641            second_tx.send(second.state().readers).unwrap();
642            second.end_write();
643        });
644        until(&gate, |state| state.writers.len() == 1);
645        gate.end_write();
646        for _ in 0..3 {
647            running_rx.recv_timeout(NEVER).unwrap();
648        }
649        assert_eq!(gate.state().readers, 3, "all three run together");
650        until(&gate, |state| state.draining.is_some());
651        assert!(second_rx.try_recv().is_err(), "the commit waits for them");
652        for _ in 0..3 {
653            finish_tx.send(()).unwrap();
654        }
655        assert_eq!(second_rx.recv_timeout(NEVER).unwrap(), 0);
656        for thread in late.into_iter().chain([second_writer]) {
657            thread.join().unwrap();
658        }

686    #[test]
687    fn a_writer_is_charged_only_the_refused_time_it_queued_through() {
688        // The arithmetic alone, on chosen instants.
689        let start = Instant::now();
690        let at = |milliseconds| start + Duration::from_millis(milliseconds);
691        let mut state = State {
692            writer: true,
693            attempt_since: Some(at(0)),
694            ..State::default()
695        };
696        // Two writers arrive during an attempt that is then refused at 300.
697        let early = state.refusals_before(at(10));
698        let late = state.refusals_before(at(280));
699        state.record_refusal(at(300));
700        assert_eq!(
701            state.queued_behind_refusals(early),
702            Duration::from_millis(290)
703        );
704        assert_eq!(
705            state.queued_behind_refusals(late),
706            Duration::from_millis(20)
707        );
708
709        // The next attempt succeeds: nobody is charged for queueing behind it.
710        state.attempt_since = Some(at(300));
711        let behind_success = state.refusals_before(at(350));
712        state.attempt_since = Some(at(900));
713        assert_eq!(state.queued_behind_refusals(behind_success), Duration::ZERO);
714        // Not even when a later attempt is refused for less than that wait.
715        state.record_refusal(at(940));
716        assert_eq!(state.queued_behind_refusals(behind_success), Duration::ZERO);
717        // Refusals add up for a writer that queued through several.
718        assert_eq!(
719            state.queued_behind_refusals(early),
720            Duration::from_millis(330)
721        );
722    }

724    #[test]
725    fn a_refused_attempt_is_charged_to_the_writers_queued_behind_it() {
726        let dir = Dir::new();
727        let pool = dir.pool();
728        let file_gate = FileGate::new();
729        let holder = file_gate.write(&pool);
730        assert_eq!(holder.queued_behind_refusals(), Duration::ZERO);
731        let gate = gate_of(&file_gate);
732        let queue = || {
733            let (file_gate, pool) = (file_gate.clone(), pool.clone());
734            let queued = gate.state().writers.len();
735            let writer = std::thread::spawn(move || {
736                let permit = file_gate.write(&pool);
737                let charged = permit.queued_behind_refusals();
738                permit.finish(&Ok::<(), &str>(()));
739                charged
740            });
741            until(&gate, |state| state.writers.len() > queued);
742            writer
743        };
744        let first = queue();
745        std::thread::sleep(Duration::from_millis(20));
746        holder.finish(&Err::<(), _>(LOCKED));
747        let charged = first.join().unwrap();
748        assert!(charged >= Duration::from_millis(20), "{charged:?}");
749
750        // Other errors and successes charge nobody.
751        let holder = file_gate.write(&pool);
752        let second = queue();
753        std::thread::sleep(Duration::from_millis(20));
754        holder.finish(&Err::<(), _>("constraint failed"));
755        assert_eq!(second.join().unwrap(), Duration::ZERO);
