42/// Storage backend selected from a database URL.
43#[derive(Clone, Copy, Debug, PartialEq, Eq)]
44enum Backend {
45    Sqlite,
46    Postgres,
47}
48impl Backend {
49    fn from_url(url: &str) -> Self {
50        let url = url.trim_start();
51        let scheme = url.split_once("://").map(|(scheme, _)| scheme);
52        match scheme.map(str::to_ascii_lowercase).as_deref() {
53            Some("postgres" | "postgresql") => Backend::Postgres,
54            _ => Backend::Sqlite,
55        }
56    }
57    fn as_str(self) -> &'static str {
58        match self {
59            Backend::Sqlite => "sqlite",
60            Backend::Postgres => "postgres",
61        }
62    }
63}
64
65/// Checks that `DATABASE_DRIVER` (when set) agrees with the URL's backend.
66fn check_driver(driver: Option<&str>, backend: Backend) -> io::Result<()> {
67    let Some(driver) = driver else {
68        return Ok(());
69    };
70    let named = match driver.trim().to_ascii_lowercase().as_str() {
71        "" | "sqlite" => Backend::Sqlite,
72        "postgres" | "postgresql" | "pg" => Backend::Postgres,
73        other => {
74            return Err(io::Error::new(
75                io::ErrorKind::Unsupported,
76                format!("arc-auth-db supports DATABASE_DRIVER=sqlite or postgres, not `{other}`"),
77            ))
78        }
79    };
80    if named != backend {
81        return Err(io::Error::new(
82            io::ErrorKind::InvalidInput,
83            format!(
84                "DATABASE_DRIVER={driver} does not match the {} database URL given to arc-auth-db",
85                backend.as_str()
86            ),
87        ));
88    }
89    Ok(())
90}

92/// Connection to whichever backend the URL names.
93enum AnyConnection {
94    Sqlite(SqliteConnection),
95    #[cfg(feature = "postgres")]
96    Postgres(diesel::PgConnection),
97}
98impl AnyConnection {
99    fn establish(url: &str) -> Result<Self, String> {
100        match Backend::from_url(url) {
101            Backend::Sqlite => SqliteConnection::establish(url)
102                .map(Self::Sqlite)
103                .map_err(|e| e.to_string()),
104            #[cfg(feature = "postgres")]
105            Backend::Postgres => diesel::PgConnection::establish(url)
106                .map(Self::Postgres)
107                .map_err(|e| e.to_string()),
108            #[cfg(not(feature = "postgres"))]
109            Backend::Postgres => Err(
110                "PostgreSQL database URLs require arc-auth-db's `postgres` cargo feature".into(),
111            ),
112        }
113    }
114    fn install_schema(&mut self) -> QueryResult<()> {
115        match self {
116            Self::Sqlite(c) => c.install_schema(),
117            #[cfg(feature = "postgres")]
118            Self::Postgres(c) => c.install_schema(),
119        }
120    }
121    #[cfg(test)]
122    fn batch_execute(&mut self, sql: &str) -> QueryResult<()> {
123        match self {
124            Self::Sqlite(c) => c.batch_execute(sql),
125            #[cfg(feature = "postgres")]
126            Self::Postgres(c) => c.batch_execute(sql),
127        }
128    }
129}

131/// Per-backend SQL dialect and transaction behaviour.
132trait Dialect: Connection + Sized {
133    const POSTGRES: bool;
134    /// A transaction that takes the write lock up front where the backend
135    /// distinguishes (SQLite `BEGIN IMMEDIATE`).
136    fn write_transaction<T, E, F>(&mut self, f: F) -> Result<T, E>
137    where
138        F: FnOnce(&mut Self) -> Result<T, E>,
139        E: From<diesel::result::Error>;
140    /// Idempotently create the identity and browser-session tables.
141    fn install_schema(&mut self) -> QueryResult<()>;
142}
143impl Dialect for SqliteConnection {
144    const POSTGRES: bool = false;
145    fn write_transaction<T, E, F>(&mut self, f: F) -> Result<T, E>
146    where
147        F: FnOnce(&mut Self) -> Result<T, E>,
148        E: From<diesel::result::Error>,
149    {
150        self.immediate_transaction(f)
151    }
152    fn install_schema(&mut self) -> QueryResult<()> {
153        self.batch_execute(IDENTITY_ROLES_MIGRATION)?;
154        self.batch_execute(BROWSER_SESSIONS_MIGRATION)
155    }
156}
157#[cfg(feature = "postgres")]
158impl Dialect for diesel::PgConnection {
159    const POSTGRES: bool = true;
160    fn write_transaction<T, E, F>(&mut self, f: F) -> Result<T, E>
161    where
162        F: FnOnce(&mut Self) -> Result<T, E>,
163        E: From<diesel::result::Error>,
164    {
165        self.transaction(f)
166    }
167    fn install_schema(&mut self) -> QueryResult<()> {
168        self.transaction(|c| {
169            sql_query("SELECT pg_advisory_xact_lock($1)")
170                .bind::<diesel::sql_types::BigInt, _>(POSTGRES_SETUP_LOCK)
171                .execute(c)?;
172            c.batch_execute(POSTGRES_IDENTITY_ROLES_MIGRATION)?;
173            c.batch_execute(POSTGRES_BROWSER_SESSIONS_MIGRATION)
174        })
175    }
176}

178/// Statement text for `C`: SQLite SQL as written, or the Postgres variant with
179/// `?` placeholders renumbered to `$1, $2, ...`.
180fn sql<C: Dialect>(sqlite: &'static str, postgres: &'static str) -> Cow<'static, str> {
181    if !C::POSTGRES {
182        return Cow::Borrowed(sqlite);
183    }
184    let mut out = String::with_capacity(postgres.len() + 8);
185    let mut n = 0;
186    for ch in postgres.chars() {
187        if ch == '?' {
188            n += 1;
189            out.push('$');
190            out.push_str(&n.to_string());
191        } else {
192            out.push(ch);
193        }
194    }
195    Cow::Owned(out)
196}
197/// Statement whose text is the same on both backends apart from placeholders.
198fn same<C: Dialect>(statement: &'static str) -> Cow<'static, str> {
199    sql::<C>(statement, statement)
200}

202struct BrowserStoreError(AuthError);
203impl From<AuthError> for BrowserStoreError {
204    fn from(error: AuthError) -> Self {
205        Self(error)

208impl From<diesel::result::Error> for BrowserStoreError {
209    fn from(error: diesel::result::Error) -> Self {
210        Self(AuthError::Store(error.to_string()))
211    }

253fn now_us() -> i64 {
254    std::time::SystemTime::now()
255        .duration_since(std::time::UNIX_EPOCH)
256        .unwrap_or_default()
257        .as_micros() as i64
258}
259fn validate(name: &str, email: &str) -> Result<String, AuthError> {
260    if name.trim().is_empty() {
261        return Err(AuthError::InvalidInput("name is required".into()));
262    }
263    let email = email.trim().to_ascii_lowercase();
264    if !email
265        .split_once('@')
266        .is_some_and(|(l, r)| !l.is_empty() && r.contains('.'))
267    {
268        return Err(AuthError::InvalidInput("valid email is required".into()));
269    }
270    Ok(email)
271}
272fn hash(password: &str) -> Result<String, AuthError> {
273    Argon2::default()
274        .hash_password(password.as_bytes(), &SaltString::generate(&mut OsRng))
275        .map(|h| h.to_string())
276        .map_err(|e| AuthError::Store(e.to_string()))
277}
278fn store_error(e: diesel::result::Error) -> AuthError {
279    AuthError::Store(e.to_string())
280}

282/// Store operations, instantiated once per backend connection type so each
283/// body is checked against a concrete diesel connection.
284macro_rules! identity_ops {
285    ($module:ident, $conn:ty) => {
286        mod $module {
287            use super::*;
288            type C = $conn;

290            const ACTIVE_USER_BY_EMAIL_SQLITE: &str = "SELECT id,name,email,password_hash,active FROM users WHERE email = ? COLLATE NOCASE AND active = 1";
291            const ACTIVE_USER_BY_EMAIL_POSTGRES: &str = "SELECT id,name,email,password_hash,active FROM users WHERE lower(email) = lower(?) AND active = 1";
292
293            fn roles(c: &mut C, id: &str) -> Result<Vec<String>, AuthError> {
294                sql_query(same::<C>("SELECT roles.name AS name FROM roles JOIN user_roles ON user_roles.role_id = roles.id WHERE user_roles.user_id = ? ORDER BY roles.name"))
295                    .bind::<diesel::sql_types::Text, _>(id)
296                    .load::<RoleRow>(c)
297                    .map(|rows| rows.into_iter().map(|r| r.name).collect())
298                    .map_err(store_error)
299            }
300            fn identity(c: &mut C, row: UserRow) -> Result<Identity, AuthError> {
301                let assigned = roles(c, &row.id)?;
302                Ok(Identity {
303                    id: row.id,
304                    name: row.name,
305                    email: row.email,
306                    active: row.active != 0,
307                    roles: assigned,
308                })
309            }
310            fn active_user_by_email(
311                c: &mut C,
312                email: &str,
313            ) -> diesel::QueryResult<Option<UserRow>> {
314                sql_query(sql::<C>(
315                    ACTIVE_USER_BY_EMAIL_SQLITE,
316                    ACTIVE_USER_BY_EMAIL_POSTGRES,
317                ))
318                .bind::<diesel::sql_types::Text, _>(email.trim())
319                .get_result::<UserRow>(c)
320                .optional()
321            }
322            fn verify(row: &UserRow, password: &str) -> Result<(), AuthError> {
323                let parsed = PasswordHash::new(&row.password_hash)
324                    .map_err(|_| AuthError::InvalidCredentials)?;
325                Argon2::default()
326                    .verify_password(password.as_bytes(), &parsed)
327                    .map_err(|_| AuthError::InvalidCredentials)
328            }
329            fn delete_user_sessions(c: &mut C, id: &str) -> diesel::QueryResult<usize> {
330                sql_query(same::<C>("DELETE FROM browser_sessions WHERE user_id=?"))
331                    .bind::<diesel::sql_types::Text, _>(id)
332                    .execute(c)
333            }
334            fn assign_roles(c: &mut C, id: &str, assigned: &[String]) -> diesel::QueryResult<()> {
335                for role in assigned {
336                    sql_query(same::<C>(
337                        "INSERT INTO user_roles (user_id,role_id) SELECT ?,id FROM roles WHERE name = ?",
338                    ))
339                    .bind::<diesel::sql_types::Text, _>(id)
340                    .bind::<diesel::sql_types::Text, _>(role)
341                    .execute(c)?;
342                }
343                Ok(())
344            }

346            pub(super) fn get(c: &mut C, id: &str) -> Result<Option<Identity>, AuthError> {
347                sql_query(same::<C>(
348                    "SELECT id,name,email,password_hash,active FROM users WHERE id = ?",
349                ))
350                .bind::<diesel::sql_types::Text, _>(id)
351                .get_result::<UserRow>(c)
352                .optional()
353                .map_err(store_error)?
354                .map(|row| identity(c, row))
355                .transpose()
356            }
357            pub(super) fn authenticate_browser(
358                c: &mut C,
359                email: &str,
360                password: &str,
361                ttl_seconds: u32,
362            ) -> Result<(Identity, String), AuthError> {
363                c.write_transaction::<_, BrowserStoreError, _>(|c| {
364                    let row = active_user_by_email(c, email)?.ok_or(AuthError::InvalidCredentials)?;
365                    verify(&row, password)?;
366                    let user = identity(c, row)?;
367                    let id = Uuid::new_v4().to_string();
368                    sql_query(same::<C>("DELETE FROM browser_sessions WHERE expires_at <= ?"))
369                        .bind::<diesel::sql_types::BigInt, _>(now_us())
370                        .execute(c)?;
371                    sql_query(same::<C>(
372                        "INSERT INTO browser_sessions (id,user_id,expires_at) VALUES (?,?,?)",
373                    ))
374                    .bind::<diesel::sql_types::Text, _>(&id)
375                    .bind::<diesel::sql_types::Text, _>(&user.id)
376                    .bind::<diesel::sql_types::BigInt, _>(
377                        now_us() + i64::from(ttl_seconds) * 1_000_000,
378                    )
379                    .execute(c)?;
380                    Ok((user, id))
381                })
382                .map_err(|e| e.0)
383            }
384            pub(super) fn browser_identity(
385                c: &mut C,
386                session_id: &str,
387            ) -> Result<Option<Identity>, AuthError> {
388                c.transaction::<_, BrowserStoreError, _>(|c| {
389                    let row = sql_query(same::<C>("SELECT users.id,name,email,password_hash,active FROM users JOIN browser_sessions ON users.id=browser_sessions.user_id WHERE browser_sessions.id=? AND expires_at>? AND active=1"))
390                        .bind::<diesel::sql_types::Text, _>(session_id)
391                        .bind::<diesel::sql_types::BigInt, _>(now_us())
392                        .get_result::<UserRow>(c)
393                        .optional()?;
394                    Ok(row.map(|row| identity(c, row)).transpose()?)
395                })
396                .map_err(|e| e.0)
397            }

428            pub(super) fn collection(
429                c: &mut C,
430                query: &CollectionQuery,
431            ) -> Result<CollectionPage<Identity>, AuthError> {
432                let filter = query.filter.trim().to_ascii_lowercase();
433                let rows = sql_query(sql::<C>(
434                    "SELECT id,name,email,'' AS password_hash,active FROM users WHERE instr(lower(name), ?) > 0 OR instr(lower(email), ?) > 0 ORDER BY email COLLATE BINARY, id COLLATE BINARY LIMIT ? OFFSET ?",
435                    "SELECT id,name,email,'' AS password_hash,active FROM users WHERE strpos(lower(name), lower(?)) > 0 OR strpos(lower(email), lower(?)) > 0 ORDER BY email COLLATE \"C\", id COLLATE \"C\" LIMIT ? OFFSET ?",

437                .bind::<diesel::sql_types::Text, _>(&filter)

460            pub(super) fn create_user(
461                c: &mut C,
462                name: &str,
463                email: &str,
464                password: &str,
465                assigned: &[String],
466            ) -> Result<Identity, AuthError> {
467                let email = validate(name, email)?;
468                let id = Uuid::new_v4().to_string();
469                let password_hash = hash(password)?;
470                let now = now_us();
471                c.transaction::<_, diesel::result::Error, _>(|c| {
472                    sql_query(same::<C>("INSERT INTO users (id,name,email,password_hash,active,created_at,updated_at) VALUES (?,?,?,?,1,?,?)"))
473                        .bind::<diesel::sql_types::Text, _>(&id)
474                        .bind::<diesel::sql_types::Text, _>(name.trim())
475                        .bind::<diesel::sql_types::Text, _>(&email)
476                        .bind::<diesel::sql_types::Text, _>(&password_hash)
477                        .bind::<diesel::sql_types::BigInt, _>(now)
478                        .bind::<diesel::sql_types::BigInt, _>(now)
479                        .execute(c)?;
480                    assign_roles(c, &id, assigned)
481                })
482                .map_err(|e| {
483                    if matches!(
484                        e,
485                        diesel::result::Error::DatabaseError(
486                            diesel::result::DatabaseErrorKind::UniqueViolation,
487                            _
488                        )
489                    ) {
490                        AuthError::DuplicateEmail
491                    } else {
492                        store_error(e)
493                    }
494                })?;
495                get(c, &id)?.ok_or(AuthError::NotFound)
496            }

499                id: &str,

532            pub(super) fn set_roles(
533                c: &mut C,
534                id: &str,
535                assigned: &[String],
536            ) -> Result<Identity, AuthError> {
537                if !assigned.iter().any(|role| role == "admin") {
538                    ensure_not_final_admin(c, id)?;
539                }
540                c.transaction::<_, diesel::result::Error, _>(|c| {
541                    delete_user_sessions(c, id)?;
542                    sql_query(same::<C>("DELETE FROM user_roles WHERE user_id=?"))
543                        .bind::<diesel::sql_types::Text, _>(id)
544                        .execute(c)?;
545                    assign_roles(c, id, assigned)
546                })
547                .map_err(store_error)?;
548                get(c, id)?.ok_or(AuthError::NotFound)
549            }
550            pub(super) fn set_active(
551                c: &mut C,
552                id: &str,
553                active: bool,
554            ) -> Result<Identity, AuthError> {
555                if !active {
556                    ensure_not_final_admin(c, id)?;
557                }
558                let changed = c
559                    .transaction::<_, diesel::result::Error, _>(|c| {
560                        let changed =
561                            sql_query(same::<C>("UPDATE users SET active=?,updated_at=? WHERE id=?"))
562                                .bind::<diesel::sql_types::Integer, _>(i32::from(active))
563                                .bind::<diesel::sql_types::BigInt, _>(now_us())
564                                .bind::<diesel::sql_types::Text, _>(id)
565                                .execute(c)?;
566                        delete_user_sessions(c, id)?;
567                        Ok(changed)
568                    })
569                    .map_err(store_error)?;
570                if changed == 0 {
571                    return Err(AuthError::NotFound);
572                }
573                get(c, id)?.ok_or(AuthError::NotFound)
574            }
575            fn ensure_not_final_admin(c: &mut C, id: &str) -> Result<(), AuthError> {
576                let target = sql_query(same::<C>("SELECT COUNT(*) AS count FROM users JOIN user_roles ON users.id=user_roles.user_id JOIN roles ON roles.id=user_roles.role_id WHERE users.id=? AND users.active=1 AND roles.name='admin'"))
577                    .bind::<diesel::sql_types::Text, _>(id)
578                    .get_result::<Count>(c)
579                    .map_err(store_error)?
580                    .count;
581                if target == 0 {
582                    return Ok(());
583                }
584                let admins = sql_query("SELECT COUNT(DISTINCT users.id) AS count FROM users JOIN user_roles ON users.id=user_roles.user_id JOIN roles ON roles.id=user_roles.role_id WHERE users.active=1 AND roles.name='admin'")
585                    .get_result::<Count>(c)
586                    .map_err(store_error)?
587                    .count;
588                if admins <= 1 {
589                    Err(AuthError::InvalidInput(
590                        "the final active administrator cannot be deactivated or lose the admin role"
591                            .into(),
592                    ))
593                } else {
594                    Ok(())
595                }
596            }
597        }
598    };
599}

602identity_ops!(postgres_ops, diesel::PgConnection);

604/// Connect and run one store operation on the backend the URL selects.
605macro_rules! dispatch {
606    ($store:expr, $op:ident($($arg:expr),* $(,)?)) => {
607        match $store.connect()? {
608            AnyConnection::Sqlite(mut c) => sqlite_ops::$op(&mut c $(, $arg)*),
609            #[cfg(feature = "postgres")]
610            AnyConnection::Postgres(mut c) => postgres_ops::$op(&mut c $(, $arg)*),
611        }
612    };
613}

692    /// `postgres` feature) a `postgres://` / `postgresql://` connection URL.

696        }

699#[async_trait]
700impl ArcPlugin for DbIdentityPlugin {
701    fn name(&self) -> &'static str {
702        "auth-db"
703    }
704    fn register(&self, builder: ArcAppBuilder) -> ArcAppBuilder {
705        builder.register_data(self.store.clone())
706    }
707    async fn setup(&self, context: &PluginSetupContext<'_>) -> io::Result<()> {
708        let backend = Backend::from_url(context.database_url);
709        check_driver(std::env::var("DATABASE_DRIVER").ok().as_deref(), backend)?;
710        let mut c = AnyConnection::establish(context.database_url).map_err(|e| {
711            let kind = if cfg!(feature = "postgres") || backend == Backend::Sqlite {
712                io::ErrorKind::Other
713            } else {
714                io::ErrorKind::Unsupported
715            };
716            io::Error::new(kind, e)
717        })?;
718        c.install_schema().map_err(io::Error::other)?;
719        if !self.store.has_users().await.map_err(io::Error::other)? {
720            let name = bootstrap_value("ARC_SETUP_ADMIN_NAME", "Administrator name", false)?;
721            let email = bootstrap_value("ARC_SETUP_ADMIN_EMAIL", "Administrator email", false)?;
722            let password =
723                bootstrap_value("ARC_SETUP_ADMIN_PASSWORD", "Administrator password", true)?;
724            self.store
725                .create_user(&name, &email, &password, &["admin".into()])
726                .await
727                .map_err(io::Error::other)?;
728        }
729        Ok(())
730    }
731}

733fn bootstrap_value(key: &str, label: &str, secret: bool) -> io::Result<String> {
734    if let Ok(value) = std::env::var(key) {
735        if !value.trim().is_empty() {
736            return Ok(value);
737        }
738    }
739    if !io::stdin().is_terminal() {
740        return Err(io::Error::other(format!(
741            "{key} is required for noninteractive first-administrator setup"
742        )));
743    }
744    let value = if secret {
745        rpassword::prompt_password(format!("{label}: "))?
746    } else {
747        print!("{label}: ");
748        io::stdout().flush()?;
749        let mut value = String::new();
750        io::stdin().read_line(&mut value)?;
751        value.trim().to_owned()
752    };
753    if value.trim().is_empty() {
754        Err(io::Error::other(format!("{label} cannot be empty")))
755    } else {
756        Ok(value)
757    }
758}

861    /// Shared browser-session checks: re-applying the session migration keeps
862    /// data, and role/active/password changes revoke every session.
863    pub(super) async fn exercise_browser_sessions(
864        url: &str,
865        identity_migration: &str,
866        browser_migration: &str,
867    ) {
868        let store = DbIdentityStore::new(url);
869        let mut c = store.connect().unwrap();

871        let user = store
872            .create_user("User", "user@example.test", "password", &["user".into()])
873            .await

877        }
878        assert_eq!(store.get(&user.id).await.unwrap(), Some(user.clone()));
879        for change in ["roles", "active", "password"] {
880            let (_, first) = store

891                .unwrap()
892                .is_some());

896                    store.set_roles(&user.id, &["user".into()]).await.unwrap();

898                "active" => {
