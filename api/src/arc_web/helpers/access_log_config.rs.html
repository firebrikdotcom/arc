1//! Startup selection for the opt-in durable read audit journal.
2use arc_core::access_log::{AccessLogger, NoOpAccessLogger};
3use arc_es_sqlite::{AccessLogPrivacy, SqliteAccessLogger};
4use std::{io, path::PathBuf, sync::Arc};
5
6/// Build from an injectable environment reader (also useful for startup tests).
7/// Invalid or unavailable configured sinks prevent startup; never fall back.
8pub async fn build_access_logger(
9    get: impl Fn(&str) -> Option<String>,
10) -> io::Result<Arc<dyn AccessLogger>> {
11    let invalid = || {
12        io::Error::new(
13            io::ErrorKind::InvalidInput,
14            "invalid access-log configuration",
15        )
16    };
17    let boolean = |key: &str| match get(key).as_deref() {
18        None | Some("false") => Ok(false),
19        Some("true") => Ok(true),
20        _ => Err(invalid()),
21    };
22    let required = boolean("ACCESS_LOG_REQUIRED")?;
23    match get("ACCESS_LOG_DRIVER").as_deref().unwrap_or("noop") {
24        "noop" if !required => {
25            tracing::warn!("access logging disabled: instrumented reads are not persisted");
26            Ok(Arc::new(NoOpAccessLogger))
27        }
28        "sqlite" => {
29            let path = PathBuf::from(get("ACCESS_LOG_SQLITE_PATH").ok_or_else(invalid)?);
30            let timeout = get("ACCESS_LOG_BUSY_TIMEOUT_MS")
31                .unwrap_or_else(|| "1000".into())
32                .parse::<u32>()
33                .map_err(|_| invalid())?;
34            let privacy = AccessLogPrivacy {
35                include_source_ip: boolean("ACCESS_LOG_INCLUDE_SOURCE_IP")?,
36                include_user_agent: boolean("ACCESS_LOG_INCLUDE_USER_AGENT")?,
37            };
38            let logger = SqliteAccessLogger::open(path, privacy, timeout)
39                .await
40                .map_err(|_| io::Error::other("could not initialize durable access journal"))?;
41            tracing::info!("durable SQLite access logging initialized");
42            Ok(Arc::new(logger))
43        }
44        _ => Err(invalid()),
45    }
46}

48#[cfg(test)]
49mod tests {
50    use super::*;
51    #[tokio::test]
52    async fn configuration_rejects_missing_required_unknown_and_unavailable() {
53        for settings in [
54            vec![("ACCESS_LOG_REQUIRED", "true")],
55            vec![("ACCESS_LOG_DRIVER", "typo")],
56            vec![("ACCESS_LOG_DRIVER", "sqlite")],
57            vec![("ACCESS_LOG_REQUIRED", "yes")],
58            vec![
59                ("ACCESS_LOG_DRIVER", "sqlite"),
60                ("ACCESS_LOG_SQLITE_PATH", "/missing-parent/audit.db"),
61            ],
62            vec![
63                ("ACCESS_LOG_DRIVER", "sqlite"),
64                ("ACCESS_LOG_SQLITE_PATH", ":memory:"),
65            ],
66        ] {
67            assert!(build_access_logger(|key| settings
68                .iter()
69                .find(|(k, _)| *k == key)
70                .map(|(_, v)| v.to_string()))
71            .await
72            .is_err());
73        }
74        assert!(build_access_logger(|_| None).await.is_ok());
75    }
