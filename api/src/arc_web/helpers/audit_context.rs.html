1//! Build a `CommandContext` from an Actix HTTP request.
2//!
3//! Pulls `source_ip`, `user_agent`, and an optional `X-Correlation-Id` header
4//! into the context so every event written by the request carries
5//! request-scoped audit metadata (HIPAA §164.312(b)).
6
7use actix_web::HttpRequest;
8use arc_core::audit::ANONYMOUS_ACTOR;
9use arc_core::command_bus::CommandContext;
10use uuid::Uuid;
11
12/// Build a `CommandContext` for an authenticated request.
13/// Pass the aggregate UUID resolved by the JWT middleware as `actor_id`.
14pub fn for_actor(req: &HttpRequest, actor_id: impl Into<String>) -> CommandContext {
15    CommandContext {
16        actor_id: actor_id.into(),
17        session_id: None,
18        source_ip: req
19            .connection_info()
20            .realip_remote_addr()
21            .map(str::to_string),
22        user_agent: req
23            .headers()
24            .get("User-Agent")
25            .and_then(|v| v.to_str().ok())
26            .map(str::to_string),
27        correlation_id: correlation_from(req),
28        causation_id: None,
29    }
30}
31
32/// Build a `CommandContext` for an unauthenticated request (e.g. self-registration).
33pub fn anonymous(req: &HttpRequest) -> CommandContext {
34    for_actor(req, ANONYMOUS_ACTOR)
35}
36
37/// Read `X-Correlation-Id` from the incoming request, falling back to a fresh UUID.
38fn correlation_from(req: &HttpRequest) -> Uuid {
39    req.headers()
40        .get("X-Correlation-Id")
41        .and_then(|v| v.to_str().ok())
42        .and_then(|s| Uuid::parse_str(s).ok())
43        .unwrap_or_else(Uuid::new_v4)
44}
