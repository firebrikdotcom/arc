1//! JWT bearer auth (HIPAA-4 aware).
2//!
3//! On every request:
4//! 1. Decode and signature-verify the bearer token.
5//! 2. Check `jti` against the server-side [`SessionStore`]. Revoked / unknown
6//!    → 401. Store unavailable → **fail closed** with 503.
7//! 3. Insert `(actor_id, jti)` into request extensions for handlers.
8//!
9//! Tokens minted before HIPAA-4 landed have no `jti`. Set
10//! `JWT_GRANDFATHER_LEGACY=true` to accept them during rollout; defaults to
11//! refusing such tokens.
12

26impl<S, B> Transform<S, ServiceRequest> for JwtMiddleware
27where
28    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
29    S::Future: 'static,
30    B: 'static,
31{
32    type Response = ServiceResponse<EitherBody<B>>;
33    type Error = Error;
34    type InitError = ();
35    type Transform = JwtCheck<S>;
36    type Future = Ready<Result<Self::Transform, Self::InitError>>;
37
38    fn new_transform(&self, service: S) -> Self::Future {
39        ready(Ok(JwtCheck {
40            service: Rc::new(service),
41        }))
42    }
43}

45pub struct JwtCheck<S> {
46    service: Rc<S>,
47}

49fn now_us() -> i64 {
50    SystemTime::now()
51        .duration_since(UNIX_EPOCH)
52        .map(|d| d.as_micros() as i64)
53        .unwrap_or(0)
54}
55
56fn legacy_grandfather_enabled() -> bool {
57    std::env::var("JWT_GRANDFATHER_LEGACY")
58        .map(|v| v == "true" || v == "1")
59        .unwrap_or(false)
60}

62fn unauthorized<B>(req: ServiceRequest, msg: &str) -> ServiceResponse<EitherBody<B>>
63where
64    B: 'static,
65{
66    let body = format!(r#"{{"error": "{}"}}"#, msg);
67    req.into_response(
68        HttpResponse::Unauthorized()
69            .content_type("application/json")
70            .body(body)
71            .map_into_right_body(),
72    )
73}
74
75fn service_unavailable<B>(req: ServiceRequest) -> ServiceResponse<EitherBody<B>>
76where
77    B: 'static,
78{
79    req.into_response(
80        HttpResponse::ServiceUnavailable()
81            .content_type("application/json")
82            .body(r#"{"error": "Authentication backend unavailable"}"#)
83            .map_into_right_body(),
84    )
85}

87impl<S, B> Service<ServiceRequest> for JwtCheck<S>
88where
89    S: Service<ServiceRequest, Response = ServiceResponse<B>, Error = Error> + 'static,
90    S::Future: 'static,
91    B: 'static,
92{
93    type Response = ServiceResponse<EitherBody<B>>;
94    type Error = Error;
95    type Future = LocalBoxFuture<'static, Result<Self::Response, Self::Error>>;
96
97    forward_ready!(service);
98
99    fn call(&self, req: ServiceRequest) -> Self::Future {
100        let auth_value = req.headers().get("Authorization");
101        let token = if let Some(auth) = auth_value {
102            auth.to_str()
103                .ok()
104                .and_then(|header| header.strip_prefix("Bearer "))
105                .map(str::to_string)
106        } else {
107            None
108        };
109
110        let token = match token {
111            Some(t) => t,
112            None => {
113                let resp = req.into_response(
114                    HttpResponse::Unauthorized()
115                        .content_type("application/json")
116                        .body(r#"{"error": "Missing or invalid Authorization header. Expected: Bearer <token>"}"#)
117                        .map_into_right_body(),
118                );
119                return Box::pin(async move { Ok(resp) });
120            }
121        };
122
123        let claims = match decode_token(&token) {
124            Ok(c) => c,
125            Err(_) => {
126                let resp = unauthorized(req, "Invalid or expired token");
127                return Box::pin(async move { Ok(resp) });
128            }
129        };
130
131        let Some(store) = req
132            .app_data::<actix_web::web::Data<dyn SessionStore>>()
133            .cloned()
134        else {
135            tracing::error!("JWT revocation store is not configured");
136            let response = service_unavailable(req);
137            return Box::pin(async move { Ok(response) });
138        };
139        if claims.jti.is_none() && !legacy_grandfather_enabled() {
140            let response = unauthorized(req, "Token has no session identifier");
141            return Box::pin(async move { Ok(response) });
142        }
143        let service = self.service.clone();
144        Box::pin(async move {
145            if let Some(jti) = claims.jti {
146                match store.is_valid(jti, now_us()).await {
147                    Ok(true) => {}
148                    Ok(false) => return Ok(unauthorized(req, "Session revoked")),
149                    Err(SessionStoreError::Sink(_)) => {
150                        tracing::error!(category = "sink_unavailable", "session store unavailable");
151                        return Ok(service_unavailable(req));
152                    }
153                    Err(SessionStoreError::NotFound(_)) => {
154                        tracing::error!(category = "record_not_found", "session store error");
155                        return Ok(service_unavailable(req));
156                    }
157                    Err(SessionStoreError::Validation(_)) => {
158                        tracing::error!(category = "validation", "session store error");
159                        return Ok(service_unavailable(req));
160                    }
161                }
162            } else {
163                tracing::warn!(reason = "legacy_jwt_no_jti", "accepting jwt without jti");
164            }
165            service
166                .call(req_with_extensions(req, claims.sub, claims.jti))
167                .await
168                .map(ServiceResponse::map_into_left_body)
169        })
170    }
171}

173/// Insert actor_id and jti into request extensions before forwarding.
174fn req_with_extensions(req: ServiceRequest, actor_id: String, jti: Option<Uuid>) -> ServiceRequest {
175    {
176        let mut ext = req.extensions_mut();
177        ext.insert(actor_id);
178        if let Some(j) = jti {
179            ext.insert(j);
180        }
181    }
182    req
