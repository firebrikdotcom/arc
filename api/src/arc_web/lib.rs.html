1//! Arc framework runtime.
2//!
3//! Holds the reusable web/runtime machinery — Actix middleware, Tera helpers,
4//! the event-sourced stack wiring, the websocket transport, and the server
5//! bootstrap — behind a builder seam. Applications depend on this crate by
6//! version and plug their own aggregate, projectors, and routes in through
7//! [`ArcApp::builder`]. No concrete domain type lives here: the runtime is
8//! generic over [`arc_core::aggregate::Aggregate`].
9

61/// A read-model projector plus the view (table) it maintains. Applications
62/// register these against their aggregate so the framework can drive
63/// synchronous, read-after-write projections in single-process mode.
64pub struct ProjectorReg {
65    pub projector: Box<dyn Projector>,
66    pub view: String,
67}
68
69impl ProjectorReg {
70    pub fn new(projector: impl Projector + 'static, view: impl Into<String>) -> Self {
71        Self {
72            projector: Box::new(projector),
73            view: view.into(),
74        }
75    }
76}

78/// Entry point to the framework. Applications register one or more aggregate
79/// types, their projectors, and routes, then drive the result with
80/// [`ArcAppBuilder::serve`].
81pub struct ArcApp;
82
83impl ArcApp {
84    pub fn builder() -> ArcAppBuilder {
85        ArcAppBuilder {
86            aggregates: Vec::new(),
87            routes: Vec::new(),
88            app_data: Vec::new(),
89            plugin_names: Vec::new(),
90            plugins: Vec::new(),
91            ui_host: None,
92            ui_contributions: Vec::new(),
93        }
94    }
95}

97type RoutesFn = dyn Fn(&mut ServiceConfig) + Send + Sync + 'static;
98type AppDataFn = dyn Fn(&mut ServiceConfig) + Send + Sync + 'static;
99type AggregateBuildFuture = Pin<
100    Box<
101        dyn Future<Output = std::io::Result<commands::serve::BuiltAggregateRuntime>>
102            + Send
103            + 'static,
104    >,
105>;
106type AggregateBuildFn =
107    dyn FnOnce(Vec<ProjectorReg>, Option<SnapshotPolicy>) -> AggregateBuildFuture + Send + 'static;
108
109struct AggregateRegistration {
110    type_id: TypeId,
111    aggregate_type: &'static str,
112    projectors: Vec<ProjectorReg>,
113    snapshot_policy: Option<SnapshotPolicy>,
114    build: Box<AggregateBuildFn>,
115}
116
117impl AggregateRegistration {
118    fn new<A: Aggregate + 'static>() -> Self {
119        Self {
120            type_id: TypeId::of::<A>(),
121            aggregate_type: A::aggregate_type(),
122            projectors: Vec::new(),
123            snapshot_policy: None,
124            build: Box::new(|projectors, snapshot_policy| {
125                Box::pin(commands::serve::build_aggregate_runtime::<A>(
126                    projectors,
127                    snapshot_policy,
128                ))
129            }),
130        }
131    }
132}

134/// Application builder holding type-erased registrations for every aggregate
135/// served by the process. Each registration produces its own typed
136/// `CommandBus<A>` while sharing the configured storage backend.
137pub struct ArcAppBuilder {
138    aggregates: Vec<AggregateRegistration>,
139    routes: Vec<Arc<RoutesFn>>,
140    app_data: Vec<Arc<AppDataFn>>,
141    plugin_names: Vec<&'static str>,
142    plugins: Vec<Arc<dyn ArcPlugin>>,
143    ui_host: Option<UiHost>,
144    ui_contributions: Vec<UiContribution>,
145}

153/// Compile-time extension package registered through [`ArcAppBuilder`].
154///
155/// Plugins remain ordinary Cargo dependencies: registration is explicit,
156/// typed, and visible in application code. Implementations may add routes,
157/// aggregates, projectors, and other builder-supported capabilities.
158#[async_trait::async_trait]
159pub trait ArcPlugin: Send + Sync + 'static {
160    /// Stable identifier used for collision detection and diagnostics.
161    fn name(&self) -> &'static str;
162
163    /// Contribute this package's capabilities to the application builder.
164    fn register(&self, builder: ArcAppBuilder) -> ArcAppBuilder;
165
166    /// Run package-owned migrations and idempotent setup work.
167    async fn setup(&self, _context: &PluginSetupContext<'_>) -> std::io::Result<()> {
168        Ok(())
169    }
170}
171
172impl ArcAppBuilder {
173    /// Register the application's sole browser UI host.
174    pub fn register_ui_host(mut self, host: UiHost) -> Self {
175        if self.ui_host.is_some() {
176            self.ui_contributions
177                .push(UiContribution::duplicate_host(host.owner));
178        } else {
179            self.ui_host = Some(host);
180        }
181        self
182    }
183
184    /// Register namespaced templates and shell metadata from a capability.
185    pub fn register_ui(mut self, contribution: UiContribution) -> Self {
186        self.ui_contributions.push(contribution);
187        self
188    }
189
190    /// Validate and build the immutable UI registry without starting a server.
191    pub fn build_ui_registry(&self) -> Result<Option<UiRegistry>, ui::UiError> {
192        UiRegistry::build(self.ui_host.as_ref(), &self.ui_contributions)
193    }
194    /// Register a writable aggregate type. Arc creates and injects a distinct
195    /// `CommandBus<A>` for every call.
196    pub fn register_aggregate<A: Aggregate + 'static>(mut self) -> Self {
197        assert!(
198            !self
199                .aggregates
200                .iter()
201                .any(|registration| registration.type_id == TypeId::of::<A>()),
202            "aggregate type {} is already registered",
203            A::aggregate_type()
204        );
205        self.aggregates.push(AggregateRegistration::new::<A>());
206        self
207    }
208
209    /// Register all projectors for the most recently registered aggregate.
210    pub fn register_projectors(mut self, projectors: Vec<ProjectorReg>) -> Self {
211        self.current_aggregate_mut().projectors.extend(projectors);
212        self
213    }
214
215    /// Register a single projector for the most recently registered aggregate.
216    pub fn register_projector(
217        mut self,
218        projector: impl Projector + 'static,
219        view: impl Into<String>,
220    ) -> Self {
221        self.current_aggregate_mut()
222            .projectors
223            .push(ProjectorReg::new(projector, view));
224        self
225    }
226
227    /// Mount the application's own routes. Framework handlers (static assets,
228    /// websocket, health) are exported from this crate for the closure to
229    /// reference; the application owns which routes it mounts.
230    pub fn register_routes<F>(mut self, f: F) -> Self
231    where
232        F: Fn(&mut ServiceConfig) + Send + Sync + 'static,
233    {
234        self.routes.push(Arc::new(f));
235        self
236    }
237
238    /// Inject shared typed state contributed by an application or plugin.
239    pub fn register_data<T>(mut self, data: Arc<T>) -> Self
240    where
241        T: ?Sized + Send + Sync + 'static,
242    {
243        let data = web::Data::from(data);
244        self.app_data.push(Arc::new(move |cfg| {
245            cfg.app_data(data.clone());
246        }));
247        self
248    }
249
250    /// Register a compile-time capability package.
251    pub fn register_plugin<P: ArcPlugin>(mut self, plugin: P) -> Self {
252        let name = plugin.name();
253        assert!(
254            !self.plugin_names.contains(&name),
255            "plugin {name} is already registered"
256        );
257        self.plugin_names.push(name);
258        let plugin = Arc::new(plugin);
259        let mut builder = plugin.register(self);
260        builder.plugins.push(plugin);
261        builder
262    }
263
264    /// Run setup hooks for every registered capability package in order.
265    pub async fn setup_plugins(&self, context: &PluginSetupContext<'_>) -> std::io::Result<()> {
266        for plugin in &self.plugins {
267            plugin.setup(context).await.map_err(|error| {
268                std::io::Error::new(
269                    error.kind(),
270                    format!("plugin {} setup failed: {error}", plugin.name()),
271                )
272            })?;
273        }
274        Ok(())
275    }
276
277    /// Snapshot policy for the most recently registered aggregate.
278    pub fn snapshot_policy(mut self, policy: Option<SnapshotPolicy>) -> Self {
279        self.current_aggregate_mut().snapshot_policy = policy;
280        self
281    }
282
283    fn current_aggregate_mut(&mut self) -> &mut AggregateRegistration {
284        self.aggregates.last_mut().unwrap_or_else(|| {
285            panic!(
286                "register_aggregate::<A>() must be called before registering projectors or a snapshot policy"
287            )
288        })
289    }
290
291    /// Boot the HTTP server with every registered aggregate and the
292    /// application's routes.
293    pub async fn serve(self, app_url: String, app_port: u16) -> std::io::Result<()> {
294        if self.aggregates.is_empty() {
295            return Err(std::io::Error::new(
296                std::io::ErrorKind::InvalidInput,
297                "ArcAppBuilder::serve requires at least one register_aggregate::<A>() call",
298            ));
299        }
300        if self.routes.is_empty() {
301            return Err(std::io::Error::new(
302                std::io::ErrorKind::InvalidInput,
303                "ArcAppBuilder::serve requires at least one route registration",
304            ));
305        }
306        let ui_registry = self.build_ui_registry().map_err(std::io::Error::other)?;
307        commands::serve::run(
308            app_url,
309            app_port,
310            self.aggregates,
311            self.routes,
312            self.app_data,
313            ui_registry.map(Arc::new),
314        )
315        .await
316    }
317}

319/// Verifies database availability for the configured driver. For file-backed
320/// drivers (SQLite) this checks the `DATABASE_URL` file exists and exits with
321/// code 1 if missing. Connection-string drivers (Postgres) skip the check.
322pub fn check_database_health() {
323    info!("Checking database health");
324    let driver = helpers::config::DatabaseDriver::from_env();
325
326    if !driver.is_file_backed() {
327        debug!(
328            driver = driver.as_str(),
329            "Database driver uses a connection string; skipping filesystem check"
330        );
331        return;
332    }
333
334    let database: String = helpers::config::database_url();
335    if !fs::exists(PathBuf::from(&database)).unwrap() {
336        error!("Database file not found at: {}", database);
337        error!("Please run `cargo run migrate` to create the database");
338        std::process::exit(1);
339    }
340    debug!("Database file found at: {}", database);
341}

343/// Copies `.env.example` to `.env` when no `.env` file is present.
344pub fn check_app_health() {
345    info!("Checking app health");
346    if !fs::exists(PathBuf::from(".env")).unwrap() {
347        info!("Creating .env file from .env.example");
348        fs::copy(PathBuf::from(".env.example"), PathBuf::from(".env"))
349            .expect("Failed to copy .env.example to .env");
350    }
351}

353/// Fails fast if required environment variables are missing.
354pub fn validate_environment() {
355    let required_vars = ["APP_URL", "SECRET_KEY", "DATABASE_URL"];
356    let mut missing = Vec::new();
357    for var in required_vars {
358        if env::var(var).is_err() {
359            missing.push(var);
360        }
361    }
362    if !missing.is_empty() {
363        error!(
364            "Missing required environment variables: {}. Check your .env file.",
365            missing.join(", ")
366        );
367        std::process::exit(1);
368    }
369    debug!("All required environment variables present");
370}
