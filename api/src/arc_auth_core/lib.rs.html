21#[derive(Debug, Error)]
22pub enum AuthError {
23    #[error("invalid credentials")]
24    InvalidCredentials,
25    #[error("identity not found")]
26    NotFound,
27    #[error("email is already registered")]
28    DuplicateEmail,
29    #[error("invalid input: {0}")]
30    InvalidInput(String),
31    #[error("identity store unavailable: {0}")]
32    Store(String),
33}
