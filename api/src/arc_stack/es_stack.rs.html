1//! Shared assembly of the event-sourced stack — `EventStore`, in-process
2//! `EventBus`, `ReadModelStore`, `ProjectionEngine`, and `CommandBus`.
3//!
4//! Generic over the domain aggregate `A`: the framework never names a concrete
5//! aggregate. Applications supply `A` and their projectors; the runtime server
6//! (`commands::serve`) and CLI utilities drive the same wiring.
7

22/// Bundle of constructed components — the parts external code keeps a
23/// handle on after wiring.
24pub struct EsStack<A: Aggregate> {
25    pub command_bus: CommandBus<A>,
26    pub read_model_store: Arc<dyn ReadModelStore>,
27    /// Held so callers that want to drive `rebuild_all()` can do so. CLI
28    /// utilities ignore it; the runtime server keeps a clone.
29    #[allow(dead_code)]
30    pub projection_engine: Arc<ProjectionEngine>,
31}
32
33/// Share one constructed stack with multiple adapters. The caller owns startup
34/// projection rebuilding before making the stack available to those adapters.
35pub struct SharedStack<A: Aggregate> {
36    /// The shared command bus.
37    pub command_bus: Arc<CommandBus<A>>,
38    /// The shared projection read store.
39    pub read_model_store: Arc<dyn ReadModelStore>,
40    /// The projection engine, including the caller-owned rebuild operation.
41    pub projection_engine: Arc<ProjectionEngine>,
42}
43
44impl<A: Aggregate> Clone for SharedStack<A> {
45    fn clone(&self) -> Self {
46        Self {
47            command_bus: self.command_bus.clone(),
48            read_model_store: self.read_model_store.clone(),
49            projection_engine: self.projection_engine.clone(),
50        }
51    }
52}
53
54impl<A: Aggregate> From<EsStack<A>> for SharedStack<A> {
55    fn from(stack: EsStack<A>) -> Self {
56        Self {
57            command_bus: Arc::new(stack.command_bus),
58            read_model_store: stack.read_model_store,
59            projection_engine: stack.projection_engine,
60        }
61    }
62}

64/// Boxed storage seam shared by every entry point. The command bus and the
65/// projection engine each own an independent handle to the same backend so the
66/// write path and replay path do not contend on one connection wrapper.
67pub struct EsStores {
68    pub command_event_store: Box<dyn EventStore>,
69    pub projection_event_store: Box<dyn EventStore>,
70    pub read_model_store: Arc<dyn ReadModelStore>,
71}

73/// Construct the event store and read-model store for the selected driver.
74/// Domain code never sees the concrete type — only `Box<dyn EventStore>` and
75/// `Arc<dyn ReadModelStore>`.
76pub async fn build_stores(
77    driver: DatabaseDriver,
78    database_url: &str,
79) -> Result<EsStores, Box<dyn std::error::Error>> {
80    match driver {
81        DatabaseDriver::Sqlite => build_sqlite_stores(database_url).await,
82        DatabaseDriver::Postgres => build_postgres_stores(database_url).await,
83    }
84}
85
86#[cfg(feature = "sqlite")]
87async fn build_sqlite_stores(database_url: &str) -> Result<EsStores, Box<dyn std::error::Error>> {
88    let event_store = build_sqlite_event_store(database_url).await?;
89    let read_model_store: Arc<dyn ReadModelStore> =
90        Arc::new(SqliteReadModelStore::new(database_url).await?);
91
92    Ok(EsStores {
93        command_event_store: Box::new(event_store.clone()),
94        projection_event_store: Box::new(event_store),
95        read_model_store,
96    })
97}
98
99#[cfg(feature = "sqlite")]
100async fn build_sqlite_event_store(
101    database_url: &str,
102) -> Result<SqliteEventStore, Box<dyn std::error::Error>> {
103    match config::event_integrity_key() {
104        Some(key) => Ok(SqliteEventStore::new_with_integrity_key(
105            database_url,
106            key,
107            config::event_integrity_key_id(),
108        )
109        .await?),
110        None => Ok(SqliteEventStore::new(database_url).await?),
111    }
112}

116    Err("DATABASE_DRIVER=sqlite requires building with the `sqlite` cargo feature".into())
117}

119#[cfg(feature = "postgres")]
120async fn build_postgres_stores(database_url: &str) -> Result<EsStores, Box<dyn std::error::Error>> {
121    use arc_es_postgres::{PostgresEventStore, PostgresReadModelStore};
122
123    let event_store = PostgresEventStore::new(database_url).await?;
124    event_store.initialize_schema().await?;
125
126    let read_model_store_impl = PostgresReadModelStore::new(database_url).await?;
127    read_model_store_impl.initialize_schema().await?;
128    let read_model_store: Arc<dyn ReadModelStore> = Arc::new(read_model_store_impl);
129
130    Ok(EsStores {
131        command_event_store: Box::new(event_store.clone()),
132        projection_event_store: Box::new(event_store),
133        read_model_store,
134    })
135}

144/// Build the in-process stack for the configured driver, registering the
145/// supplied projectors against the in-process bus so writes drive their views
146/// synchronously. Used by CLI utilities (`migrate --seed`, `seed`) and tests.
147pub async fn build<A: Aggregate>(
148    database_url: &str,
149    projectors: Vec<ProjectorReg>,
150    snapshot_policy: Option<SnapshotPolicy>,
151) -> Result<EsStack<A>, Box<dyn std::error::Error>> {
152    build_with_driver(
153        DatabaseDriver::from_env(),
154        database_url,
155        projectors,
156        snapshot_policy,
157    )
158    .await
159}
160
161/// Build an in-process stack using an explicitly selected database driver.
162/// The caller is responsible for rebuilding projections when required.
163///
164/// Only the driver is explicit. The SQLite event store still reads
165/// `EVENT_INTEGRITY_KEY` and `EVENT_INTEGRITY_KEY_ID` from the process
166/// environment; when they are unset, events are stored unsigned.
167pub async fn build_with_driver<A: Aggregate>(
168    driver: DatabaseDriver,
169    database_url: &str,
170    projectors: Vec<ProjectorReg>,
171    snapshot_policy: Option<SnapshotPolicy>,
172) -> Result<EsStack<A>, Box<dyn std::error::Error>> {
173    let stores = build_stores(driver, database_url).await?;
174    let read_model_store = stores.read_model_store.clone();
175
176    let mut engine = ProjectionEngine::new(stores.projection_event_store);
177    for reg in projectors {
178        engine.register_projector(reg.projector, read_model_store.clone(), reg.view);
179    }
180    let engine = Arc::new(engine);
181
182    let mut bus = InProcessEventBus::new();
183    bus.subscribe(Box::new(ProjectionEngineHandler::new(engine.clone())))
184        .await?;
185
186    let command_bus = apply_snapshot_policy(
187        CommandBus::<A>::new(stores.command_event_store, Box::new(bus)),
188        snapshot_policy,
189    );
190
191    Ok(EsStack {
192        command_bus,
193        read_model_store,
194        projection_engine: engine,
195    })
196}

198/// Apply a snapshot policy to a command bus if one is configured.
199pub fn apply_snapshot_policy<A: Aggregate>(
200    command_bus: CommandBus<A>,
201    policy: Option<SnapshotPolicy>,
202) -> CommandBus<A> {
203    match policy {
204        Some(p) => command_bus.with_snapshot_policy(p),
205        None => command_bus,
206    }
207}
