1//! Database selection and event integrity configuration.
2use std::env;
3
4/// Storage backend selected at startup behind the `EventStore` /
5/// `ReadModelStore` traits. Parsing is always available; constructing the
6/// Postgres stores additionally requires the `postgres` cargo feature.
7#[derive(Clone, Copy, Debug, PartialEq, Eq)]
8pub enum DatabaseDriver {
9    Sqlite,
10    Postgres,
11}
12
13impl DatabaseDriver {
14    /// Read `DATABASE_DRIVER` from the environment, defaulting to SQLite.
15    pub fn from_env() -> Self {
16        Self::parse(&env::var("DATABASE_DRIVER").unwrap_or_else(|_| "sqlite".to_string()))
17    }
18
19    /// Parse a driver name. Anything other than a recognized Postgres alias
20    /// falls back to SQLite, matching the historical default.
21    pub fn parse(value: &str) -> Self {
22        match value.trim().to_ascii_lowercase().as_str() {
23            "postgres" | "postgresql" | "pg" => DatabaseDriver::Postgres,
24            _ => DatabaseDriver::Sqlite,
25        }
26    }
27
28    pub fn as_str(self) -> &'static str {
29        match self {
30            DatabaseDriver::Sqlite => "sqlite",
31            DatabaseDriver::Postgres => "postgres",
32        }
33    }
34
35    /// Whether `DATABASE_URL` names a local filesystem path the startup health
36    /// check can stat. Postgres uses a connection string, not a file.
37    pub fn is_file_backed(self) -> bool {
38        matches!(self, DatabaseDriver::Sqlite)
39    }
40}
41
42/// Optional HMAC key enabling HIPAA-5 event integrity signing/enforcement.
43/// The storage layer validates length and rejects keys shorter than 32 bytes.
44pub fn event_integrity_key() -> Option<Vec<u8>> {
45    env::var("EVENT_INTEGRITY_KEY")
46        .ok()
47        .filter(|value| !value.trim().is_empty())
48        .map(|value| value.into_bytes())
49}
50
51/// Identifier stored with each event signature so future key rotation can
52/// distinguish which key signed a row.
53pub fn event_integrity_key_id() -> String {
54    env::var("EVENT_INTEGRITY_KEY_ID").unwrap_or_else(|_| "default".to_string())
55}
